#include <hip/hip_runtime.h>
#include <cstdint>
extern "C" __global__ void tsvc_2_s252_fp64_kernel(double* a, const double* b, const double* c, int64_t n) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n) {
        double prod = b[i] * c[i];
        double prev = (i > 0) ? b[i-1] * c[i-1] : 0.0;
        a[i] = prod + prev;
    }
}
