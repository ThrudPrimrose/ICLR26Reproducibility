#include <hip/hip_runtime.h>
#include <cstdint>
#include <cmath>

extern "C" __global__ void argmax_reduce_kernel(const double *a, int64_t len,
                                                double *block_max, int64_t *block_idx) {
    const int tid = threadIdx.x;
    const int blockSize = blockDim.x;
    const int totalThreads = blockSize * gridDim.x;
    double local_max = -INFINITY;
    int64_t local_idx = -1;
    for (int64_t i = static_cast<int64_t>(blockIdx.x) * blockSize + tid; i < len; i += totalThreads) {
        double val = a[i];
        if (val > local_max) {
            local_max = val;
            local_idx = i;
        }
    }
    __shared__ double s_val[256];
    __shared__ int64_t s_idx[256];
    s_val[tid] = local_max;
    s_idx[tid] = local_idx;
    __syncthreads();
    for (int step = blockSize / 2; step > 0; step >>= 1) {
        if (tid < step) {
            double other_val = s_val[tid + step];
            int64_t other_idx = s_idx[tid + step];
            if (other_val > s_val[tid]) {
                s_val[tid] = other_val;
                s_idx[tid] = other_idx;
            }
        }
        __syncthreads();
    }
    if (tid == 0) {
        block_max[blockIdx.x] = s_val[0];
        block_idx[blockIdx.x] = s_idx[0];
    }
}

extern "C" __global__ void argmax_final_reduce_kernel(const double *vals, const int64_t *idxs,
                                                      int64_t n, double *out_val, int64_t *out_idx) {
    const int tid = threadIdx.x;
    const int blockSize = blockDim.x;
    double local_max = -INFINITY;
    int64_t local_idx = -1;
    for (int64_t i = tid; i < n; i += blockSize) {
        double val = vals[i];
        if (val > local_max) {
            local_max = val;
            local_idx = idxs[i];
        }
    }
    __shared__ double s_val[256];
    __shared__ int64_t s_idx[256];
    s_val[tid] = local_max;
    s_idx[tid] = local_idx;
    __syncthreads();
    for (int step = blockSize / 2; step > 0; step >>= 1) {
        if (tid < step) {
            double other_val = s_val[tid + step];
            int64_t other_idx = s_idx[tid + step];
            if (other_val > s_val[tid]) {
                s_val[tid] = other_val;
                s_idx[tid] = other_idx;
            }
        }
        __syncthreads();
    }
    if (tid == 0) {
        *out_val = s_val[0];
        *out_idx = s_idx[0];
    }
}
