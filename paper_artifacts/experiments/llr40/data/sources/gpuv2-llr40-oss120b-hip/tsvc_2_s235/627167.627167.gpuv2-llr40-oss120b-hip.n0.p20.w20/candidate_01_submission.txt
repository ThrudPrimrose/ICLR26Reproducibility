#include <hip/hip_runtime.h>
#include <cstdint>

extern "C" __global__ void tsvc_2_s235_kernel(double* __restrict__ a, double* __restrict__ aa,
                                             const double* __restrict__ b, const double* __restrict__ bb,
                                             const double* __restrict__ c, const int64_t LEN_2D) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i >= LEN_2D) return;

    double ai = a[i] + b[i] * c[i];
    a[i] = ai;
    double* aa_i = aa + i;
    const double* bb_i = bb + i;
    double prev = aa_i[0];
    int stride = (int)LEN_2D;
    double* aa_dest = aa_i + stride; // aa[1, i]
    const double* bb_src = bb_i + stride; // bb[1, i]
    #pragma unroll 4
    for (int j = 1; j < stride; ++j) {
        double val = prev + (*bb_src) * ai;
        *aa_dest = val;
        prev = val;
        aa_dest += stride;
        bb_src += stride;
    }
}
