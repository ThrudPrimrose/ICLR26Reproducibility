#include <cstdint>
#ifdef __HIP_PLATFORM_AMD__
#include <hip/hip_runtime.h>
__global__ void tsvc_2_vpvts_kernel(double* __restrict__ a, const double* __restrict__ b, const int64_t LEN_1D, const double s) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int totalThreads = gridDim.x * blockDim.x;
    for (int64_t i = idx; i < LEN_1D; i += totalThreads) {
        a[i] += b[i] * s;
    }
}
#else
inline void tsvc_2_vpvts_kernel(double* a, const double* b, const int64_t LEN_1D, const double s) {}
#endif
