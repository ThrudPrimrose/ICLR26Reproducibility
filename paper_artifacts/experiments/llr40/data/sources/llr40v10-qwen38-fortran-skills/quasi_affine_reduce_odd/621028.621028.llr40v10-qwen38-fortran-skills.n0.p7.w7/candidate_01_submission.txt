subroutine quasi_affine_reduce_odd_fp64(a, out, len_1d) bind(C)
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: len_1d
  real(c_double), intent(in) :: a(len_1d)
  real(c_double), intent(out) :: out(1)
  double precision :: acc
  integer(c_int64_t) :: j

  if (len_1d < 65536) then
    acc = 0.0d0
    !$omp simd reduction(+:acc)
    do j = 2, len_1d, 2
      acc = acc + a(j)
    end do
  else
    acc = 0.0d0
    !$omp parallel do simd reduction(+:acc)
    do j = 2, len_1d, 2
      acc = acc + a(j)
    end do
  end if
  out(1) = acc
end subroutine quasi_affine_reduce_odd_fp64
