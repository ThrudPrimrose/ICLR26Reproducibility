#include <stdint.h>
#include <immintrin.h>

static inline __m512d work_a(__m512d ai, __m512d bi, __m512d di) {
    return _mm512_fmadd_pd(bi, di, ai);
}
static inline __m512d work_b(__m512d ai, __m512d ei2) {
    return _mm512_add_pd(ai, ei2);
}

void tsvc_2_s2710_fp64(double *restrict a, double *restrict b, double *restrict c,
                       const double *restrict d, const double *restrict e,
                       const double *restrict x, const int64_t LEN_1D) {
    const int len_gt_10 = LEN_1D > 10;
    const int x_pos = x[0] > 0.0;
    const int64_t nvec = LEN_1D & ~7;
    int64_t i = 0;
    if (nvec > 0) {
        const __m512d vone = _mm512_set1_pd(1.0);
        if (LEN_1D > 4096) {
            const int64_t nvec2 = nvec & ~15;
            #pragma omp parallel for schedule(static)
            for (i = 0; i < nvec2; i += 16) {
                __m512d ai0 = _mm512_loadu_pd(&a[i]);
                __m512d bi0 = _mm512_loadu_pd(&b[i]);
                __m512d ci0 = _mm512_loadu_pd(&c[i]);
                __m512d di0 = _mm512_loadu_pd(&d[i]);
                __m512d ei0 = _mm512_loadu_pd(&e[i]);

                __m512d ai1 = _mm512_loadu_pd(&a[i+8]);
                __m512d bi1 = _mm512_loadu_pd(&b[i+8]);
                __m512d ci1 = _mm512_loadu_pd(&c[i+8]);
                __m512d di1 = _mm512_loadu_pd(&d[i+8]);
                __m512d ei1 = _mm512_loadu_pd(&e[i+8]);

                __mmask8 m0 = _mm512_cmp_pd_mask(ai0, bi0, _CMP_GT_OQ);
                __mmask8 m1 = _mm512_cmp_pd_mask(ai1, bi1, _CMP_GT_OQ);

                __m512d di02 = _mm512_mul_pd(di0, di0);
                __m512d ei02 = _mm512_mul_pd(ei0, ei0);
                __m512d di12 = _mm512_mul_pd(di1, di1);
                __m512d ei12 = _mm512_mul_pd(ei1, ei1);

                __m512d a_new0 = _mm512_fmadd_pd(bi0, di0, ai0);
                __m512d a_new1 = _mm512_fmadd_pd(bi1, di1, ai1);
                __m512d b_new0 = _mm512_add_pd(ai0, ei02);
                __m512d b_new1 = _mm512_add_pd(ai1, ei12);

                __m512d c_true0, c_false0, c_true1, c_false1;
                if (len_gt_10) {
                    c_true0 = _mm512_add_pd(ci0, di02);
                    c_true1 = _mm512_add_pd(ci1, di12);
                } else {
                    c_true0 = _mm512_fmadd_pd(di0, ei0, vone);
                    c_true1 = _mm512_fmadd_pd(di1, ei1, vone);
                }
                if (x_pos) {
                    c_false0 = _mm512_add_pd(ai0, di02);
                    c_false1 = _mm512_add_pd(ai1, di12);
                } else {
                    c_false0 = _mm512_add_pd(ci0, ei02);
                    c_false1 = _mm512_add_pd(ci1, ei12);
                }

                __m512d a_out0 = _mm512_mask_blend_pd(m0, ai0, a_new0);
                __m512d a_out1 = _mm512_mask_blend_pd(m1, ai1, a_new1);
                __m512d b_out0 = _mm512_mask_blend_pd(m0, b_new0, bi0);
                __m512d b_out1 = _mm512_mask_blend_pd(m1, b_new1, bi1);
                __m512d c_out0 = _mm512_mask_blend_pd(m0, c_false0, c_true0);
                __m512d c_out1 = _mm512_mask_blend_pd(m1, c_false1, c_true1);

                _mm512_storeu_pd(&a[i], a_out0);
                _mm512_storeu_pd(&a[i+8], a_out1);
                _mm512_storeu_pd(&b[i], b_out0);
                _mm512_storeu_pd(&b[i+8], b_out1);
                _mm512_storeu_pd(&c[i], c_out0);
                _mm512_storeu_pd(&c[i+8], c_out1);
            }
            for (i = nvec2; i < nvec; i += 8) {
                __m512d ai = _mm512_loadu_pd(&a[i]);
                __m512d bi = _mm512_loadu_pd(&b[i]);
                __m512d ci = _mm512_loadu_pd(&c[i]);
                __m512d di = _mm512_loadu_pd(&d[i]);
                __m512d ei = _mm512_loadu_pd(&e[i]);

                __mmask8 m = _mm512_cmp_pd_mask(ai, bi, _CMP_GT_OQ);
                __m512d di2 = _mm512_mul_pd(di, di);
                __m512d ei2 = _mm512_mul_pd(ei, ei);

                __m512d a_new = _mm512_fmadd_pd(bi, di, ai);
                __m512d b_new = _mm512_add_pd(ai, ei2);

                __m512d c_true, c_false;
                if (len_gt_10) {
                    c_true = _mm512_add_pd(ci, di2);
                } else {
                    c_true = _mm512_fmadd_pd(di, ei, vone);
                }
                if (x_pos) {
                    c_false = _mm512_add_pd(ai, di2);
                } else {
                    c_false = _mm512_add_pd(ci, ei2);
                }

                __m512d a_out = _mm512_mask_blend_pd(m, ai, a_new);
                __m512d b_out = _mm512_mask_blend_pd(m, b_new, bi);
                __m512d c_out = _mm512_mask_blend_pd(m, c_false, c_true);

                _mm512_storeu_pd(&a[i], a_out);
                _mm512_storeu_pd(&b[i], b_out);
                _mm512_storeu_pd(&c[i], c_out);
            }
        } else {
            for (i = 0; i < nvec; i += 8) {
                __m512d ai = _mm512_loadu_pd(&a[i]);
                __m512d bi = _mm512_loadu_pd(&b[i]);
                __m512d ci = _mm512_loadu_pd(&c[i]);
                __m512d di = _mm512_loadu_pd(&d[i]);
                __m512d ei = _mm512_loadu_pd(&e[i]);

                __mmask8 m = _mm512_cmp_pd_mask(ai, bi, _CMP_GT_OQ);
                __m512d di2 = _mm512_mul_pd(di, di);
                __m512d ei2 = _mm512_mul_pd(ei, ei);

                __m512d a_new = _mm512_fmadd_pd(bi, di, ai);
                __m512d b_new = _mm512_add_pd(ai, ei2);

                __m512d c_true, c_false;
                if (len_gt_10) {
                    c_true = _mm512_add_pd(ci, di2);
                } else {
                    c_true = _mm512_fmadd_pd(di, ei, vone);
                }
                if (x_pos) {
                    c_false = _mm512_add_pd(ai, di2);
                } else {
                    c_false = _mm512_add_pd(ci, ei2);
                }

                __m512d a_out = _mm512_mask_blend_pd(m, ai, a_new);
                __m512d b_out = _mm512_mask_blend_pd(m, b_new, bi);
                __m512d c_out = _mm512_mask_blend_pd(m, c_false, c_true);

                _mm512_storeu_pd(&a[i], a_out);
                _mm512_storeu_pd(&b[i], b_out);
                _mm512_storeu_pd(&c[i], c_out);
            }
        }
        i = nvec;
    }
    for (; i < LEN_1D; ++i) {
        if (a[i] > b[i]) {
            a[i] += b[i] * d[i];
            if (len_gt_10)
                c[i] += d[i] * d[i];
            else
                c[i] = d[i] * e[i] + 1.0;
        } else {
            b[i] = a[i] + e[i] * e[i];
            if (x_pos)
                c[i] = a[i] + d[i] * d[i];
            else
                c[i] += e[i] * e[i];
        }
    }
}
