subroutine argmax_with_index_fp64(a, out_index, out_value, LEN_1D) bind(C, name="argmax_with_index_fp64")
    use iso_c_binding
    use omp_lib
    implicit none
    real(c_double), intent(in) :: a(*)
    integer(c_int64_t), intent(out) :: out_index(*)
    real(c_double), intent(out) :: out_value(*)
    integer(c_int64_t), intent(in), value :: LEN_1D

    real(c_double) :: x, bx
    integer(c_int64_t) :: idx, bidx, i, t, nthreads, start, end
    real(c_double), allocatable :: block_max(:)
    integer(c_int64_t), allocatable :: block_idx(:)

    if (LEN_1D <= 0) then
        out_value(1) = a(1)
        out_index(1) = 1
        return
    end if

    nthreads = 1
    !$ nthreads = omp_get_max_threads()

    if (LEN_1D < 4096 .or. nthreads <= 1) then
        x = a(1)
        idx = 1
        do i = 2, LEN_1D
            if (a(i) > x) then
                x = a(i)
                idx = i
            end if
        end do
        out_value(1) = x
        out_index(1) = idx
        return
    end if

    allocate(block_max(nthreads), block_idx(nthreads))

    !$omp parallel private(t, start, end, i, bx, bidx)
    t = omp_get_thread_num() + 1
    start = ((t - 1) * LEN_1D) / nthreads + 1
    end = (t * LEN_1D) / nthreads
    bx = a(start)
    bidx = start
    do i = start + 1, end
        if (a(i) > bx) then
            bx = a(i)
            bidx = i
        end if
    end do
    block_max(t) = bx
    block_idx(t) = bidx
    !$omp end parallel

    x = block_max(1)
    idx = block_idx(1)
    do i = 2, nthreads
        bx = block_max(i)
        if (bx > x) then
            x = bx
            idx = block_idx(i)
        end if
    end do

    out_value(1) = x
    out_index(1) = idx
end subroutine argmax_with_index_fp64
