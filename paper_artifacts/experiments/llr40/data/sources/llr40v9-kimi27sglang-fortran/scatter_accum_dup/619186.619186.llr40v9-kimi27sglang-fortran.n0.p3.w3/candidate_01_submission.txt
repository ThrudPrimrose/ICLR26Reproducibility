subroutine scatter_accum_dup_fp64(bins, ip, src, LEN_1D) bind(C, name='scatter_accum_dup_fp64')
  use, intrinsic :: iso_c_binding, only: c_double, c_int64_t, c_int
  implicit none
  integer(c_int64_t), value, intent(in) :: LEN_1D
  real(c_double), intent(inout) :: bins(*)
  integer(c_int), intent(in) :: ip(*)
  real(c_double), intent(in) :: src(*)
  integer(c_int64_t) :: i, idx

  if (LEN_1D <= 0) return

  if (LEN_1D < 2048_c_int64_t) then
    do i = 1, LEN_1D
      idx = ip(i)
      bins(idx) = bins(idx) + src(i)
    end do
    return
  end if

  !$omp parallel do private(i, idx) schedule(static)
  do i = 1, LEN_1D
    idx = ip(i)
    !$omp atomic update
    bins(idx) = bins(idx) + src(i)
    !$omp end atomic
  end do
  !$omp end parallel do
end subroutine scatter_accum_dup_fp64
