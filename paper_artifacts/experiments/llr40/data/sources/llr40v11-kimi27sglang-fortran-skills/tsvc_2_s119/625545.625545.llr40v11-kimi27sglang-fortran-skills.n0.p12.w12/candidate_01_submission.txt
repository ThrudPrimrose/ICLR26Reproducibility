subroutine tsvc_2_s119(aa, bb, len_2d) bind(C, name="tsvc_2_s119_fp64")
  use iso_c_binding
  use omp_lib
  implicit none
  integer(c_int64_t), value, intent(in) :: len_2d
  real(c_double), intent(inout) :: aa(len_2d, len_2d)
  real(c_double), intent(in) :: bb(len_2d, len_2d)
  integer(c_int64_t) :: i, j, n

  n = len_2d
  if (n < 64 .or. omp_get_max_threads() < 2) then
    do i = 2, n
      do j = 2, n
        aa(j, i) = aa(j - 1, i - 1) + bb(j, i)
      end do
    end do
    return
  end if

  !$omp parallel private(i, j)
  do i = 2, n
    !$omp do simd schedule(static)
    do j = 2, n
      aa(j, i) = aa(j - 1, i - 1) + bb(j, i)
    end do
  end do
  !$omp end parallel
end subroutine tsvc_2_s119
