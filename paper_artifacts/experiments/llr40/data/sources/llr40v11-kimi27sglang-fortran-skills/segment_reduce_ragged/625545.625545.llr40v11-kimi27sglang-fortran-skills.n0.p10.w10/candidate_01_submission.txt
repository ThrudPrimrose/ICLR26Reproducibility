subroutine segment_reduce_ragged_fp64(out, row_ptr, val, w, NSEG) bind(C, name="segment_reduce_ragged_fp64")
  use, intrinsic :: iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: NSEG
  real(c_double), intent(inout) :: out(NSEG)
  integer(c_int64_t), intent(in) :: row_ptr(NSEG + 1)
  real(c_double), intent(in) :: val(NSEG * 24)
  real(c_double), intent(in) :: w(NSEG * 24)
  integer(c_int64_t) :: s, e, lo, hi
  real(c_double) :: acc

  !$omp parallel do schedule(dynamic, 128) private(s, e, lo, hi, acc)
  do s = 1, NSEG
    lo = row_ptr(s)
    hi = row_ptr(s + 1)
    acc = 0.0_c_double
    do e = lo, hi - 1
      acc = acc + val(e + 1) * w(e + 1)
    end do
    out(s) = acc
  end do
end subroutine segment_reduce_ragged_fp64
