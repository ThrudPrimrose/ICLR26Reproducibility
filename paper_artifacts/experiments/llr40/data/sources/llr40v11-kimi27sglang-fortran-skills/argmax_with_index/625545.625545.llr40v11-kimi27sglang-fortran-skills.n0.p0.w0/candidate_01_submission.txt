subroutine argmax_with_index_fp64(a, out_index, out_value, n, workspace, workspace_size) bind(C)
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: n
  integer(c_int64_t), value, intent(in) :: workspace_size
  real(c_double), intent(in) :: a(n)
  integer(c_int64_t), intent(out) :: out_index(1)
  real(c_double), intent(out) :: out_value(1)
  integer(c_int8_t), intent(inout) :: workspace(workspace_size)

  real(c_double) :: m
  integer(c_int64_t) :: i, first

  m = a(1)
  !$omp parallel do reduction(max:m) schedule(static)
  do i = 2, n
    if (a(i) > m) m = a(i)
  end do

  first = n + 1
  !$omp parallel do reduction(min:first) schedule(static)
  do i = 1, n
    if (a(i) == m .and. i < first) first = i
  end do

  out_value(1) = m
  out_index(1) = first
end subroutine argmax_with_index_fp64
