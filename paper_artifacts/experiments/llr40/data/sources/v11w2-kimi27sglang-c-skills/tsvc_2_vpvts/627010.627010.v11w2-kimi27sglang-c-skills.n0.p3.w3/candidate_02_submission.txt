#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

void tsvc_2_vpvts_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D, const int64_t S) {
  if (LEN_1D < 4096) {
    #pragma omp simd
    for (int64_t i = 0; i < LEN_1D; ++i) {
      a[i] += b[i] * S;
    }
    return;
  }
  const double s = (double)S;
  const __m512d vs = _mm512_set1_pd(s);
  #pragma omp parallel for schedule(static)
  for (int64_t i = 0; i < LEN_1D - 15; i += 16) {
    __m512d a0 = _mm512_loadu_pd(&a[i + 0]);
    __m512d a1 = _mm512_loadu_pd(&a[i + 8]);
    __m512d b0 = _mm512_loadu_pd(&b[i + 0]);
    __m512d b1 = _mm512_loadu_pd(&b[i + 8]);
    a0 = _mm512_fmadd_pd(b0, vs, a0);
    a1 = _mm512_fmadd_pd(b1, vs, a1);
    _mm512_storeu_pd(&a[i + 0], a0);
    _mm512_storeu_pd(&a[i + 8], a1);
  }
  for (int64_t i = LEN_1D & ~15; i < LEN_1D; ++i) {
    a[i] += b[i] * S;
  }
}
