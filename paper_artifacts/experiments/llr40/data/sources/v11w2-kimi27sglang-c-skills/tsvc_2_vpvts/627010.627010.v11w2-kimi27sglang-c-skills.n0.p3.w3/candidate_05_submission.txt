#include <stdint.h>
#include <omp.h>

void tsvc_2_vpvts_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D, const int64_t S) {
  if (LEN_1D < 4096) {
    #pragma omp simd
    for (int64_t i = 0; i < LEN_1D; ++i) {
      a[i] += b[i] * S;
    }
    return;
  }
  const int max_threads = omp_get_max_threads();
  const int nthreads = max_threads < 1 ? 1 : max_threads;
  const int desired = (int)(LEN_1D / 256);
  const int use_threads = desired < 1 ? 1 : (desired > nthreads ? nthreads : desired);
  #pragma omp parallel for simd schedule(static) num_threads(use_threads)
  for (int64_t i = 0; i < LEN_1D; ++i) {
    a[i] += b[i] * S;
  }
}
