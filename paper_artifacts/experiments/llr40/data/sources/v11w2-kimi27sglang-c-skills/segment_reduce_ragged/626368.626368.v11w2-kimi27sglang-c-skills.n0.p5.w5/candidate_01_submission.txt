#include <stdint.h>
#include <omp.h>

static int64_t lower_bound(const int64_t *row_ptr, int64_t n, int64_t target)
{
    int64_t lo = 0;
    int64_t hi = n;
    while (lo < hi) {
        const int64_t mid = (lo + hi) >> 1;
        if (row_ptr[mid] < target) {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    return lo;
}

void segment_reduce_ragged_fp64(double *restrict out,
                                int64_t *restrict row_ptr,
                                double *restrict val,
                                double *restrict w,
                                int64_t NSEG,
                                uint8_t *restrict workspace,
                                int64_t workspace_size)
{
    (void)workspace;
    (void)workspace_size;

    const int64_t total = row_ptr[NSEG];

    #pragma omp parallel shared(row_ptr, val, w, out, NSEG, total)
    {
        const int tid = omp_get_thread_num();
        const int nthreads = omp_get_num_threads();
        const int64_t start = ((int64_t)tid * total) / nthreads;
        const int64_t end = ((int64_t)(tid + 1) * total) / nthreads;
        const int64_t s_begin = lower_bound(row_ptr, NSEG + 1, start);
        const int64_t s_end = lower_bound(row_ptr, NSEG + 1, end);

        for (int64_t s = s_begin; s < s_end; ++s) {
            double acc = 0.0;
            const int64_t e0 = row_ptr[s];
            const int64_t e1 = row_ptr[s + 1];
            #pragma omp simd reduction(+:acc)
            for (int64_t e = e0; e < e1; ++e) {
                acc += val[e] * w[e];
            }
            out[s] = acc;
        }
    }
}
