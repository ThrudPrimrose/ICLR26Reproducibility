#include <stdint.h>
#include <omp.h>

void versioned_distance_update_fp64(double *restrict a, const double *restrict b, const double *restrict c, const int64_t K, const int64_t LEN_1D, uint8_t *restrict workspace, const int64_t workspace_bytes) {
    (void)workspace;
    (void)workspace_bytes;
    if (K <= 0 || LEN_1D <= K) return;
    const double decay = 0.75;
    if (K >= 64) {
        #pragma omp parallel
        {
            const int64_t nt = omp_get_num_threads();
            const int64_t tid = omp_get_thread_num();
            // chunk rounded up to multiple of 8 doubles (one cache line)
            int64_t chunk = (K + nt - 1) / nt;
            chunk = (chunk + 7) & ~((int64_t)7);
            const int64_t j0 = tid * chunk;
            const int64_t j1 = (j0 + chunk < K) ? j0 + chunk : K;
            if (j0 < K) {
                for (int64_t base = K; base < LEN_1D; base += K) {
                    const int64_t end = base + K < LEN_1D ? base + K : LEN_1D;
                    const int64_t istart = base + j0;
                    if (istart >= end) continue;
                    const int64_t iend = base + j1 < end ? base + j1 : end;
                    #pragma omp simd
                    for (int64_t i = istart; i < iend; ++i) {
                        a[i] = decay * a[i - K] + b[i] * c[i];
                    }
                }
            }
        }
    } else {
        for (int64_t base = K; base < LEN_1D; base += K) {
            const int64_t end = base + K < LEN_1D ? base + K : LEN_1D;
            #pragma omp simd
            for (int64_t i = base; i < end; ++i) {
                a[i] = decay * a[i - K] + b[i] * c[i];
            }
        }
    }
}
