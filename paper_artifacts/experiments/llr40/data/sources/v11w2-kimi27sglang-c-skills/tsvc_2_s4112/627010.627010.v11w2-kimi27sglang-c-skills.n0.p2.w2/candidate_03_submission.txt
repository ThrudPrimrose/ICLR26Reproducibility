#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

void tsvc_2_s4112_fp64(double *restrict a, const double *restrict b,
                       const int32_t *restrict ip, const int64_t LEN_1D) {
    const int64_t VL = 8;
    const __m512d vtwo = _mm512_set1_pd(2.0);
    const int64_t PF1 = 5;
    const int64_t PF2 = 6;

    int64_t head = 0;
    if (LEN_1D > 0) {
        uintptr_t ap = (uintptr_t)a;
        uintptr_t ipp = (uintptr_t)ip;
        int64_t need_a  = (ap  & 63ULL) ? ((64ULL - (ap  & 63ULL)) / sizeof(double)) : 0;
        int64_t need_ip = (ipp & 63ULL) ? ((64ULL - (ipp & 63ULL)) / sizeof(int32_t)) : 0;
        int64_t need = need_a > need_ip ? need_a : need_ip;
        if (need > LEN_1D) need = LEN_1D;
        for (; head < need; ++head) {
            a[head] += b[ip[head]] * 2.0;
        }
    }

    const int64_t main_k = (LEN_1D - head) / VL;
    const int64_t tail   = head + main_k * VL;

    #pragma omp parallel for schedule(static) if(LEN_1D > 4096)
    for (int64_t k = 0; k < main_k; ++k) {
        const int64_t i = head + k * VL;
        if (k + PF2 < main_k) {
            const int64_t pi1 = head + (k + PF1) * VL;
            const int64_t pi2 = head + (k + PF2) * VL;
            __m256i idx_pf1 = _mm256_stream_load_si256((__m256i *)(void *)(ip + pi1));
            __m256i idx_pf2 = _mm256_stream_load_si256((__m256i *)(void *)(ip + pi2));
            int32_t pidx1[8] __attribute__((aligned(32)));
            int32_t pidx2[8] __attribute__((aligned(32)));
            _mm256_store_si256((__m256i *)pidx1, idx_pf1);
            _mm256_store_si256((__m256i *)pidx2, idx_pf2);
            _mm_prefetch((const char *)(b + pidx1[0]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx1[1]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx1[2]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx1[3]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx1[4]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx1[5]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx1[6]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx1[7]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[0]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[1]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[2]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[3]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[4]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[5]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[6]), _MM_HINT_T1);
            _mm_prefetch((const char *)(b + pidx2[7]), _MM_HINT_T1);
        }
        __m256i idx = _mm256_stream_load_si256((__m256i *)(void *)(ip + i));
        __m512d bv  = _mm512_i32gather_pd(idx, (const void *)b, 8);
        __m512d av  = _mm512_castsi512_pd(
                          _mm512_stream_load_si512((__m512i *)(void *)(a + i)));
        __m512d out = _mm512_fmadd_pd(vtwo, bv, av);
        _mm512_stream_pd(a + i, out);
    }

    _mm_sfence();

    for (int64_t i = tail; i < LEN_1D; ++i) {
        a[i] += b[ip[i]] * 2.0;
    }
}
