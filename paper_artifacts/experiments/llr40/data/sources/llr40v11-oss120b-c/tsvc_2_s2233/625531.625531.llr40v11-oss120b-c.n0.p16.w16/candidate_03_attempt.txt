#include <stdint.h>

void tsvc_2_s2233_fp64(double *restrict aa, double *restrict bb, const double *restrict cc, const int64_t LEN_2D) {
    // Parallelize across columns for "aa" prefix sums. Each column i can be processed
    // independently because the recurrence only touches elements in that column.
    #pragma omp parallel for schedule(static)
    for (int64_t i = 8; i < LEN_2D; ++i) {
        // Pointers to the start of the i-th column (row index 8) and the previous row (7).
        const double *prev = aa + (8 - 1) * LEN_2D + i;   // aa[7, i]
        const double *cptr = cc + 8 * LEN_2D + i;       // cc[8, i]
        double *aptr = aa + 8 * LEN_2D + i;             // aa[8, i]
        // Compute the column prefix sum across rows j.
        for (int64_t j = 8; j < LEN_2D; ++j) {
            *aptr = *prev + *cptr;
            prev += LEN_2D;
            cptr += LEN_2D;
            aptr += LEN_2D;
        }
    }

    // Parallelize across columns for "bb" prefix sums. Here each column j is independent.
    #pragma omp parallel for schedule(static)
    for (int64_t j = 8; j < LEN_2D; ++j) {
        const double *prev = bb + (8 - 1) * LEN_2D + j;   // bb[7, j]
        const double *cptr = cc + 8 * LEN_2D + j;       // cc[8, j]
        double *bptr = bb + 8 * LEN_2D + j;             // bb[8, j]
        for (int64_t i = 8; i < LEN_2D; ++i) {
            *bptr = *prev + *cptr;
            prev += LEN_2D;
            cptr += LEN_2D;
            bptr += LEN_2D;
        }
    }
}
