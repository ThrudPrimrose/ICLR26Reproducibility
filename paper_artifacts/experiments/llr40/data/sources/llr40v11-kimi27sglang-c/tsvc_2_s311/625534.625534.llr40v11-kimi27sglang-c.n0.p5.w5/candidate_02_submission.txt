#include <stdint.h>
#include <omp.h>

void tsvc_2_s311_fp64(const double *restrict a, double *restrict sum_out, const int64_t LEN_1D) {
    const int64_t n4 = LEN_1D & ~((int64_t)3);
    double sum0 = 0.0, sum1 = 0.0, sum2 = 0.0, sum3 = 0.0;
    
    #pragma omp parallel for reduction(+:sum0,sum1,sum2,sum3) schedule(static)
    for (int64_t i = 0; i < n4; i += 4) {
        sum0 += a[i];
        sum1 += a[i+1];
        sum2 += a[i+2];
        sum3 += a[i+3];
    }
    
    double sum = sum0 + sum1 + sum2 + sum3;
    for (int64_t i = n4; i < LEN_1D; i++) {
        sum += a[i];
    }
    sum_out[0] = sum;
}
