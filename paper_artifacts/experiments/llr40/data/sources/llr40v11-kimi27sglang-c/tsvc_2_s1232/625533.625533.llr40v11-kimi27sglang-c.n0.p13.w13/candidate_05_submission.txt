#include <stdint.h>
#include <omp.h>

void tsvc_2_s1232_fp64(double *restrict aa, const double *restrict bb, const double *restrict cc,
                       const int64_t LEN_2D, const int64_t VLEN) {
    if (LEN_2D <= 0 || VLEN <= 0) {
        return;
    }

    #pragma omp parallel for schedule(guided,1)
    for (int64_t i = 0; i < LEN_2D; ++i) {
        const int64_t jmax = (VLEN == 1) ? i : (i / VLEN);
        const int64_t jlim = (jmax < LEN_2D) ? jmax : LEN_2D - 1;

        double *restrict a_row = aa + i * LEN_2D;
        const double *restrict b_row = bb + i * LEN_2D;
        const double *restrict c_row = cc + i * LEN_2D;

        #pragma omp simd simdlen(8)
        for (int64_t j = 0; j <= jlim; ++j) {
            a_row[j] = b_row[j] + c_row[j];
        }
    }
}
