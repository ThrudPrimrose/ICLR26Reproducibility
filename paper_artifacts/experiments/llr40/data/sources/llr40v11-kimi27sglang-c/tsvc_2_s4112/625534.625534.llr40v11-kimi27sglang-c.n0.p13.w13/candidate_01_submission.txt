#include <stdint.h>
#include <immintrin.h>

void tsvc_2_s4112_fp64(double *restrict a, const double *restrict b, const int32_t *restrict ip, const int64_t LEN_1D) {
    const __m512d vtwo = _mm512_set1_pd(2.0);

    #pragma omp parallel for
    for (int64_t i = 0; i < LEN_1D; i += 8) {
        int64_t remaining = LEN_1D - i;
        if (remaining >= 8) {
            __m256i idx = _mm256_loadu_si256((const __m256i *)(ip + i));
            __m512d bv = _mm512_i32gather_pd(idx, b, 8);
            __m512d av = _mm512_loadu_pd(a + i);
            av = _mm512_fmadd_pd(bv, vtwo, av);
            _mm512_storeu_pd(a + i, av);
        } else {
            __mmask8 mask = (1 << remaining) - 1;
            __m256i idx = _mm256_maskz_loadu_epi32(mask, ip + i);
            __m512d bv = _mm512_mask_i32gather_pd(_mm512_setzero_pd(), mask, idx, b, 8);
            __m512d av = _mm512_maskz_loadu_pd(mask, a + i);
            av = _mm512_fmadd_pd(bv, vtwo, av);
            _mm512_mask_storeu_pd(a + i, mask, av);
        }
    }
}
