#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

void quasi_affine_reduce_odd_fp64(const double *restrict a, double *restrict out, const int64_t LEN_1D) {
    const __mmask8 odd_mask = 0xAA; /* lanes 1,3,5,7 */
    double sum = 0.0;

    if (LEN_1D > 1) {
        #pragma omp parallel reduction(+:sum) if(LEN_1D > 4096)
        {
            const int tid = omp_get_thread_num();
            const int nt  = omp_get_num_threads();
            const int64_t chunk = LEN_1D / nt;
            const int64_t rem   = LEN_1D % nt;
            int64_t start = tid * chunk + (tid < rem ? tid : rem);
            int64_t end   = start + chunk + (tid < rem ? 1 : 0);

            double partial = 0.0;
            int64_t i = start;

            /* process elements until the next 8-element boundary */
            for (; i < end && (i & 7LL); ++i) {
                if (i & 1LL) partial += a[i];
            }

            __m512d acc0 = _mm512_setzero_pd();
            __m512d acc1 = _mm512_setzero_pd();
            __m512d acc2 = _mm512_setzero_pd();
            __m512d acc3 = _mm512_setzero_pd();

            for (; i + 32 <= end; i += 32) {
                acc0 = _mm512_add_pd(acc0, _mm512_maskz_loadu_pd(odd_mask, a + i));
                acc1 = _mm512_add_pd(acc1, _mm512_maskz_loadu_pd(odd_mask, a + i + 8));
                acc2 = _mm512_add_pd(acc2, _mm512_maskz_loadu_pd(odd_mask, a + i + 16));
                acc3 = _mm512_add_pd(acc3, _mm512_maskz_loadu_pd(odd_mask, a + i + 24));
            }
            for (; i + 8 <= end; i += 8) {
                acc0 = _mm512_add_pd(acc0, _mm512_maskz_loadu_pd(odd_mask, a + i));
            }

            __m512d acc01 = _mm512_add_pd(acc0, acc1);
            __m512d acc23 = _mm512_add_pd(acc2, acc3);
            __m512d acc   = _mm512_add_pd(acc01, acc23);
            partial += _mm512_reduce_add_pd(acc);

            /* scalar tail for this thread */
            for (int64_t j = i + 1; j < end; j += 2) {
                partial += a[j];
            }

            sum += partial;
        }
    }

    out[0] = sum;
}
