#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

void ext_break_capture_fp64(const double *restrict a, int64_t *restrict out_index,
                            double *restrict out_value, const int64_t LEN_1D) {
    const double k = 1.0;
    const __m512d vk = _mm512_set1_pd(k);

    int64_t best_idx = LEN_1D;

    /* For small inputs a serial SIMD scan is cheaper than spawning threads. */
    if (LEN_1D < 65536) {
        int64_t i = 0;
        for (; i + 8 <= LEN_1D; i += 8) {
            __m512d va = _mm512_loadu_pd(a + i);
            __mmask8 mask = _mm512_cmp_pd_mask(va, vk, _CMP_GT_OQ);
            if (__builtin_expect(mask != 0, 0)) {
                int j = __builtin_ctz((unsigned)mask);
                out_index[0] = i + j;
                out_value[0] = a[i + j];
                return;
            }
        }
        for (; i < LEN_1D; ++i) {
            if (a[i] > k) {
                out_index[0] = i;
                out_value[0] = a[i];
                return;
            }
        }
        out_index[0] = -1;
        out_value[0] = -1.0;
        return;
    }

    const int64_t chunk = 4096;
    const int64_t nchunks = (LEN_1D + chunk - 1) / chunk;

    #pragma omp parallel for schedule(dynamic, 1)
    for (int64_t c = 0; c < nchunks; ++c) {
        int64_t start = c * chunk;
        int64_t end = start + chunk;
        if (end > LEN_1D) end = LEN_1D;

        int64_t i = start;
        int found = 0;
        for (; i + 8 <= end; i += 8) {
            __m512d va = _mm512_loadu_pd(a + i);
            __mmask8 mask = _mm512_cmp_pd_mask(va, vk, _CMP_GT_OQ);
            if (__builtin_expect(mask != 0, 0)) {
                int j = __builtin_ctz((unsigned)mask);
                int64_t fidx = i + j;
                #pragma omp critical
                {
                    if (fidx < best_idx) best_idx = fidx;
                }
                found = 1;
                break;
            }
        }
        if (!found) {
            for (; i < end; ++i) {
                if (a[i] > k) {
                    #pragma omp critical
                    {
                        if (i < best_idx) best_idx = i;
                    }
                    break;
                }
            }
        }
    }

    if (best_idx < LEN_1D) {
        out_index[0] = best_idx;
        out_value[0] = a[best_idx];
    } else {
        out_index[0] = -1;
        out_value[0] = -1.0;
    }
}
