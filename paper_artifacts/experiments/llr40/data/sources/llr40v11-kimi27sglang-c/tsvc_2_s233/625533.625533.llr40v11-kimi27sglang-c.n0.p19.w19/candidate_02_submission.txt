#include <stdint.h>
#include <omp.h>

void tsvc_2_s233_fp64(double *restrict aa, double *restrict bb, const double *restrict cc, const int64_t LEN_2D) {
    const int64_t n = LEN_2D;
    const int64_t m = n - 8;
    if (m <= 0) return;

    aa = __builtin_assume_aligned(aa, 64);
    bb = __builtin_assume_aligned(bb, 64);
    cc = __builtin_assume_aligned(cc, 64);

    const int maxt = omp_get_max_threads();
    int nthreads;
    if (m < 64) {
        nthreads = 1;
    } else if (m < 256) {
        nthreads = (maxt < 4) ? maxt : 4;
    } else if (m < 1024) {
        nthreads = (maxt < 16) ? maxt : 16;
    } else {
        nthreads = maxt;
    }
    if (nthreads > (int)m) nthreads = (int)m;
    if (nthreads < 1) nthreads = 1;

    #pragma omp parallel if(nthreads > 1) num_threads(nthreads)
    {
        const int tid = omp_get_thread_num();
        const int nt = omp_get_num_threads();

        /* aa: columns are independent.  Each thread owns a contiguous column slab. */
        const int64_t i0 = 8 + (tid * m) / nt;
        const int64_t i1 = 8 + ((tid + 1) * m) / nt;
        for (int64_t j = 8; j < n; ++j) {
            double *restrict const a_row = aa + j * n;
            const double *restrict const a_prev = aa + (j - 1) * n;
            const double *restrict const c_row = cc + j * n;
            #pragma omp simd
            for (int64_t i = i0; i < i1; ++i) {
                a_row[i] = a_prev[i] + c_row[i];
            }
        }

        /* bb: rows are independent.  Each thread owns a contiguous row slab. */
        const int64_t j0 = 8 + (tid * m) / nt;
        const int64_t j1 = 8 + ((tid + 1) * m) / nt;
        for (int64_t j = j0; j < j1; ++j) {
            double *restrict const b_row = bb + j * n;
            const double *restrict const c_row = cc + j * n;
            for (int64_t i = 8; i < n; ++i) {
                b_row[i] = b_row[i - 1] + c_row[i];
            }
        }
    }
}
