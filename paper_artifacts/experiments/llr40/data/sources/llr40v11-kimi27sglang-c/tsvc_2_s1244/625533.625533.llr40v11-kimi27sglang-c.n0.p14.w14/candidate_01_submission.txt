#include <stdint.h>
#include <omp.h>

#define TSVC_MAX_THREADS 256

void tsvc_2_s1244_fp64(double *restrict a, const double *restrict b,
                       const double *restrict c, double *restrict d,
                       const int64_t LEN_1D) {
    const int64_t n = LEN_1D - 1;
    if (n <= 0) return;

    /* In the reference loop d[i] uses the freshly-written a[i] and the
       *original* a[i+1] (the loop has not reached i+1 yet).  We can fuse the
       two statements into one forward pass provided we read a[i+1] before
       any thread overwrites it.  To keep OpenMP safe, each thread snapshots the
       original value at the right edge of its chunk before any writes. */

    const int max_threads = omp_get_max_threads();
    if (max_threads > TSVC_MAX_THREADS) {
        /* Fallback: two parallel passes, recomputing the polynomial. */
        #pragma omp parallel for schedule(static) if(n > 4096)
        for (int64_t i = 0; i < n; ++i) {
            double bi = b[i];
            double ci = c[i];
            d[i] = (bi * bi + bi + ci * ci + ci) + a[i + 1];
        }
        #pragma omp parallel for schedule(static) if(n > 4096)
        for (int64_t i = 0; i < n; ++i) {
            double bi = b[i];
            double ci = c[i];
            a[i] = bi * bi + bi + ci * ci + ci;
        }
        return;
    }

    double right[TSVC_MAX_THREADS];

    #pragma omp parallel if(n > 4096)
    {
        const int tid = omp_get_thread_num();
        const int nthreads = omp_get_num_threads();
        const int64_t chunk = (n + nthreads - 1) / nthreads;
        const int64_t lo = tid * chunk;
        int64_t hi = lo + chunk;
        if (hi > n) hi = n;

        if (lo < hi) {
            right[tid] = a[hi];          /* original boundary value */
        }
        #pragma omp barrier

        for (int64_t i = lo; i < hi; ++i) {
            double bi = b[i];
            double ci = c[i];
            double t = bi * bi + bi + ci * ci + ci;
            double ap1 = (i + 1 < hi) ? a[i + 1] : right[tid];
            d[i] = t + ap1;
            a[i] = t;
        }
    }
}
