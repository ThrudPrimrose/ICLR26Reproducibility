#include <stdint.h>
#include <math.h>

typedef struct {
    double value;
    int64_t index;
} argmax_pair_t;

#pragma omp declare reduction(argmax_pair_reduction: argmax_pair_t: \
    omp_out = (omp_in.value > omp_out.value) ? omp_in : omp_out) \
    initializer(omp_priv = omp_orig)

void argmax_with_index_fp64(const double *restrict a, int64_t *restrict out_index,
                            double *restrict out_value, const int64_t LEN_1D) {
    argmax_pair_t best = {a[0], 0};

    #pragma omp parallel for reduction(argmax_pair_reduction: best) schedule(static)
    for (int64_t i = 1; i < LEN_1D; ++i) {
        double v = a[i];
        if (v > best.value) {
            best.value = v;
            best.index = i;
        }
    }

    out_value[0] = best.value;
    out_index[0] = best.index;
}
