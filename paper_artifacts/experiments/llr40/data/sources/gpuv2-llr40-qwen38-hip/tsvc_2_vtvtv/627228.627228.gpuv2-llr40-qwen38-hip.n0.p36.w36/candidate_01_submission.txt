// Device half for tsvc_2_vtvtv: a[i] = a[i] * b[i] * c[i], fp64 elementwise.
#include <hip/hip_runtime.h>
#include <cstdint>

namespace {

struct alignas(32) d4 {
    double x, y, z, w;
};

__global__ void vtvtv_vec4(d4* __restrict__ a, const d4* __restrict__ b,
                           const d4* __restrict__ c,
                           double* __restrict__ an, const double* __restrict__ bn,
                           const double* __restrict__ cn, int64_t nv, int64_t rem) {
    const int64_t stride = (int64_t)gridDim.x * blockDim.x;
    const int64_t gid = (int64_t)blockIdx.x * blockDim.x + threadIdx.x;
    for (int64_t i = gid; i < nv; i += stride) {
        d4 av = a[i];
        d4 bv = b[i];
        d4 cv = c[i];
        d4 r;
        r.x = av.x * bv.x * cv.x;
        r.y = av.y * bv.y * cv.y;
        r.z = av.z * bv.z * cv.z;
        r.w = av.w * bv.w * cv.w;
        a[i] = r;
    }
    if (gid < rem) {
        an[gid] = an[gid] * bn[gid] * cn[gid];
    }
}

__global__ void vtvtv_scalar(double* __restrict__ a, const double* __restrict__ b,
                             const double* __restrict__ c, int64_t n) {
    const int64_t stride = (int64_t)gridDim.x * blockDim.x;
    for (int64_t i = (int64_t)blockIdx.x * blockDim.x + threadIdx.x; i < n; i += stride)
        a[i] = a[i] * b[i] * c[i];
}

}  // namespace

extern "C" void tsvc_2_vtvtv_launch(double* a, const double* b, const double* c, int64_t n) {
    if (n <= 0) return;
    const int block = 256;
    uint64_t mis = (reinterpret_cast<uint64_t>(a) | reinterpret_cast<uint64_t>(b) |
                    reinterpret_cast<uint64_t>(c)) & 31;
    if (mis == 0) {
        int64_t nv = n >> 2;
        int64_t rem = n & 3;
        int64_t want = nv < 163840 ? nv : 163840;
        int grid = (int)((want + block - 1) / block);
        if (grid < 1) grid = 1;
        vtvtv_vec4<<<grid, block, 0, 0>>>((d4*)a, (const d4*)b, (const d4*)c,
                                          a + (nv << 2), b + (nv << 2), c + (nv << 2),
                                          nv, rem);
    } else {
        int64_t want = n < 163840 ? n : 163840;
        int grid = (int)((want + block - 1) / block);
        if (grid < 1) grid = 1;
        vtvtv_scalar<<<grid, block, 0, 0>>>(a, b, c, n);
    }
    hipStreamSynchronize(0);
}
