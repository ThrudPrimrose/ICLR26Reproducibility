#include <hip/hip_runtime.h>
#include <stdint.h>

__device__ __forceinline__ double warp_reduce(double s) {
  #pragma unroll
  for (int off = 16; off > 0; off >>= 1)
    s += __shfl_down_sync(__activemask(), s, off);
  return s;
}
__device__ __forceinline__ void masked_add4(double4& acc, double4 v) {
  acc.x += (v.x > 0.0) ? v.x : 0.0;
  acc.y += (v.y > 0.0) ? v.y : 0.0;
  acc.z += (v.z > 0.0) ? v.z : 0.0;
  acc.w += (v.w > 0.0) ? v.w : 0.0;
}

// Pass 1: per-block partial sum -> ws[blockIdx.x].  ILP=4 grid-stride.
__global__ void k_partials(const double4* __restrict__ a4,
                           const double*  __restrict__ ascal,
                           double* __restrict__ ws,
                           long long n) {
  const int tid = (int)threadIdx.x;
  const long long nthread = (long long)gridDim.x * blockDim.x;
  const long long n4 = n >> 2, tail0 = n4 << 2, step = nthread;
  double4 a0 = make_double4(0,0,0,0), a1 = a0, a2 = a0, a3 = a0;
  long long g = (long long)blockIdx.x*blockDim.x + tid;
  for (; g + 3*step < n4; g += 4*step) {
    masked_add4(a0, a4[g]);
    masked_add4(a1, a4[g+step]);
    masked_add4(a2, a4[g+2*step]);
    masked_add4(a3, a4[g+3*step]);
  }
  for (; g < n4; g += step) masked_add4(a0, a4[g]);
  for (long long t = tail0 + (long long)blockIdx.x*blockDim.x + tid; t < n; t += step) {
    double v = ascal[t]; a0.x += (v > 0.0) ? v : 0.0;
  }
  double s = ((a0.x+a0.y)+(a0.z+a0.w)) + ((a1.x+a1.y)+(a1.z+a1.w))
           + ((a2.x+a2.y)+(a2.z+a2.w)) + ((a3.x+a3.y)+(a3.z+a3.w));
  s = warp_reduce(s);
  __shared__ double smem[32];
  int warp = tid >> 5, lane = tid & 31;
  if (lane == 0) smem[warp] = s;
  __syncthreads();
  int nwarp = (blockDim.x + 31) >> 5;
  if (warp == 0) {
    s = (lane < nwarp) ? smem[lane] : 0.0;
    s = warp_reduce(s);
    if (lane == 0) ws[blockIdx.x] = s;
  }
}

// Pass 2: reduce per-block partials into b[0].
__global__ void k_final(const double* __restrict__ ws, double* __restrict__ dst, int nblk) {
  const int tid = (int)threadIdx.x;
  const int nth = (int)gridDim.x * blockDim.x;
  double s = 0.0;
  for (int i = tid; i < nblk; i += nth) s += ws[i];
  s = warp_reduce(s);
  __shared__ double smem[32];
  int warp = tid >> 5, lane = tid & 31;
  if (lane == 0) smem[warp] = s;
  __syncthreads();
  int nwarp = (blockDim.x + 31) >> 5;
  if (warp == 0) {
    s = (lane < nwarp) ? smem[lane] : 0.0;
    s = warp_reduce(s);
    if (lane == 0) dst[0] = s;
  }
}

extern "C" void launch_tsvc(const double* a, double* ws, double* b, long long n) {
  const int block = 256;
  const long long n4 = n >> 2;
  // Adaptive grid: ~256 double4 per thread at the cap; cap 1216 blocks.
  long long grid = (n4 + (long long)block*256 - 1) / ((long long)block*256);
  if (grid < 1) grid = 1;
  if (grid > 1216) grid = 1216;
  hipLaunchKernelGGL(k_partials, dim3(grid), dim3(block), 0, 0,
                     (const double4*)a, a, ws, n);
  hipLaunchKernelGGL(k_final, dim3(1), dim3(256), 0, 0,
                     (const double*)ws, b, (int)grid);
}
