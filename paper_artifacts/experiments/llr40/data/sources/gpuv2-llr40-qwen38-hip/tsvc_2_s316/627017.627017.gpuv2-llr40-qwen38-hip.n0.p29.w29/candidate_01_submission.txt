#include <hip/hip_runtime.h>
#include <cstdint>
#include <cfloat>

#define NTHREADS 256

__global__ void tsvc_min_kernel(const double* __restrict__ a,
                                double* __restrict__ result,
                                long long n,
                                double* __restrict__ partials,
                                unsigned int* __restrict__ counter) {
    __shared__ double sdata[NTHREADS];
    long long tid = (long long)blockIdx.x * NTHREADS + threadIdx.x;
    long long stride = (long long)gridDim.x * NTHREADS;
    double m = INFINITY;
    for (long long i = tid; i < n; i += stride) {
        double v = a[i];
        if (v < m) m = v;
    }
    sdata[threadIdx.x] = m;
    __syncthreads();
    for (unsigned s = NTHREADS/2; s > 0; s >>= 1) {
        if (threadIdx.x < s) {
            double o = sdata[threadIdx.x + s];
            if (o < sdata[threadIdx.x]) sdata[threadIdx.x] = o;
        }
        __syncthreads();
    }
    __shared__ unsigned int s_islast;
    if (threadIdx.x == 0) {
        partials[blockIdx.x] = sdata[0];
        __threadfence();
        unsigned int done = atomicAdd(counter, 1u);
        s_islast = (done == gridDim.x - 1);
    }
    __syncthreads();
    if (s_islast) {
        double g = INFINITY;
        for (unsigned i = threadIdx.x; i < gridDim.x; i += NTHREADS) {
            double v = partials[i];
            if (v < g) g = v;
        }
        sdata[threadIdx.x] = g;
        __syncthreads();
        for (unsigned s = NTHREADS/2; s > 0; s >>= 1) {
            if (threadIdx.x < s) {
                double o = sdata[threadIdx.x + s];
                if (o < sdata[threadIdx.x]) sdata[threadIdx.x] = o;
            }
            __syncthreads();
        }
        if (threadIdx.x == 0) { result[0] = sdata[0]; *counter = 0; }
    }
}

extern "C" void tsvc_min_launch(const double* a, double* result, long long n,
                     double* partials, unsigned int* counter, int nblocks) {
    tsvc_min_kernel<<<nblocks, NTHREADS>>>(a, result, n, partials, counter);
}
