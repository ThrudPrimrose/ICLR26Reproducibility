'''Optimized Python implementation of the ``tsvc_2`` kernel ``s1232`` using cached NumPy ``where`` mask.

The reference double loop is slow.  A NumPy ``add`` with a boolean ``where`` mask computes the
result in compiled C code in a single pass.  To avoid the cost of allocating the ``LEN_2D``
by ``LEN_2D`` mask on every call we cache it per ``(LEN_2D, VLEN)`` pair.

The function modifies ``aa`` in place and returns ``None``, matching the C ABI expected by the
benchmark harness.
''' 

from __future__ import annotations

import numpy as np

# Cache mapping (LEN_2D, VLEN) -> boolean mask of shape (LEN_2D, LEN_2D)
_mask_cache: dict[tuple[int, int], np.ndarray] = {}


def s1232(aa: np.ndarray, bb: np.ndarray, cc: np.ndarray, LEN_2D: int, VLEN: int) -> None:
    """In-place compute ``aa[i, j] = bb[i, j] + cc[i, j]`` for ``i >= j * VLEN``.

    Parameters
    ----------
    aa, bb, cc:
        2-D ``float64`` (or compatible) arrays of shape ``(LEN_2D, LEN_2D)``. ``aa`` is
        overwritten with the result.
    LEN_2D:
        The dimension of the square matrices.
    VLEN:
        Vector length controlling the start index of each column.
    """
    if LEN_2D == 0:
        return
    key = (LEN_2D, VLEN)
    mask = _mask_cache.get(key)
    if mask is None:
        # Build a broadcasted boolean mask where the condition i >= j * VLEN holds.
        i = np.arange(LEN_2D)[:, None]
        j = np.arange(LEN_2D)[None, :]
        mask = i >= j * VLEN
        _mask_cache[key] = mask
    # Perform the addition only where the mask is true; ``aa`` retains its original values elsewhere.
    np.add(bb, cc, out=aa, where=mask)
    # No explicit return – ``aa`` now holds the result.
