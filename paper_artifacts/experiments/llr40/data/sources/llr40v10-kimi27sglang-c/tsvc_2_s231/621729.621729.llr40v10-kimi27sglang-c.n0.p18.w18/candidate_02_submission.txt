#include <stdint.h>
#include <omp.h>

void tsvc_2_s231_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
    const int64_t n = LEN_2D;
    if (n <= 1) return;

    const int nth = omp_get_max_threads();
    int64_t BS = (n + nth - 1) / nth;
    BS = ((BS + 7) / 8) * 8;
    if (BS > 384) BS = 384;
    if (BS > n) BS = n;

    #pragma omp parallel for schedule(static, 1)
    for (int64_t ib = 0; ib < n; ib += BS) {
        const int64_t len = (ib + BS < n) ? BS : (n - ib);
        double *restrict cur = &aa[n + ib];
        const double *restrict prev = &aa[ib];
        const double *restrict b = &bb[n + ib];
        for (int64_t j = 1; j < n; ++j) {
            #pragma omp simd
            for (int64_t i = 0; i < len; ++i) {
                cur[i] = prev[i] + b[i];
            }
            prev = cur;
            cur += n;
            b += n;
        }
    }
}
