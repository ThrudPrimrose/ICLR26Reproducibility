#include <stdint.h>
#include <omp.h>
#include <immintrin.h>

void tsvc_2_vtvtv_fp64(double *restrict a, const double *restrict b, const double *restrict c, const int64_t LEN_1D) {
    if (LEN_1D <= 0) return;

    int64_t n = LEN_1D;
    if (n < 512) {
        for (int64_t i = 0; i < n; ++i) {
            a[i] = a[i] * b[i] * c[i];
        }
        return;
    }

    int64_t n64 = n & ~63;
    const int64_t PF = 64; // prefetch 64 doubles (512 bytes) ahead

    #pragma omp parallel for schedule(static)
    for (int64_t j = 0; j < n64; j += 64) {
        _mm_prefetch((const char *)&a[j + PF], _MM_HINT_T0);
        _mm_prefetch((const char *)&b[j + PF], _MM_HINT_T0);
        _mm_prefetch((const char *)&c[j + PF], _MM_HINT_T0);

        __m512d va0 = _mm512_loadu_pd(&a[j + 0]);
        __m512d va1 = _mm512_loadu_pd(&a[j + 8]);
        __m512d va2 = _mm512_loadu_pd(&a[j + 16]);
        __m512d va3 = _mm512_loadu_pd(&a[j + 24]);
        __m512d va4 = _mm512_loadu_pd(&a[j + 32]);
        __m512d va5 = _mm512_loadu_pd(&a[j + 40]);
        __m512d va6 = _mm512_loadu_pd(&a[j + 48]);
        __m512d va7 = _mm512_loadu_pd(&a[j + 56]);

        __m512d vb0 = _mm512_loadu_pd(&b[j + 0]);
        __m512d vb1 = _mm512_loadu_pd(&b[j + 8]);
        __m512d vb2 = _mm512_loadu_pd(&b[j + 16]);
        __m512d vb3 = _mm512_loadu_pd(&b[j + 24]);
        __m512d vb4 = _mm512_loadu_pd(&b[j + 32]);
        __m512d vb5 = _mm512_loadu_pd(&b[j + 40]);
        __m512d vb6 = _mm512_loadu_pd(&b[j + 48]);
        __m512d vb7 = _mm512_loadu_pd(&b[j + 56]);

        __m512d vc0 = _mm512_loadu_pd(&c[j + 0]);
        __m512d vc1 = _mm512_loadu_pd(&c[j + 8]);
        __m512d vc2 = _mm512_loadu_pd(&c[j + 16]);
        __m512d vc3 = _mm512_loadu_pd(&c[j + 24]);
        __m512d vc4 = _mm512_loadu_pd(&c[j + 32]);
        __m512d vc5 = _mm512_loadu_pd(&c[j + 40]);
        __m512d vc6 = _mm512_loadu_pd(&c[j + 48]);
        __m512d vc7 = _mm512_loadu_pd(&c[j + 56]);

        va0 = _mm512_mul_pd(va0, vb0);
        va1 = _mm512_mul_pd(va1, vb1);
        va2 = _mm512_mul_pd(va2, vb2);
        va3 = _mm512_mul_pd(va3, vb3);
        va4 = _mm512_mul_pd(va4, vb4);
        va5 = _mm512_mul_pd(va5, vb5);
        va6 = _mm512_mul_pd(va6, vb6);
        va7 = _mm512_mul_pd(va7, vb7);

        va0 = _mm512_mul_pd(va0, vc0);
        va1 = _mm512_mul_pd(va1, vc1);
        va2 = _mm512_mul_pd(va2, vc2);
        va3 = _mm512_mul_pd(va3, vc3);
        va4 = _mm512_mul_pd(va4, vc4);
        va5 = _mm512_mul_pd(va5, vc5);
        va6 = _mm512_mul_pd(va6, vc6);
        va7 = _mm512_mul_pd(va7, vc7);

        _mm512_storeu_pd(&a[j + 0], va0);
        _mm512_storeu_pd(&a[j + 8], va1);
        _mm512_storeu_pd(&a[j + 16], va2);
        _mm512_storeu_pd(&a[j + 24], va3);
        _mm512_storeu_pd(&a[j + 32], va4);
        _mm512_storeu_pd(&a[j + 40], va5);
        _mm512_storeu_pd(&a[j + 48], va6);
        _mm512_storeu_pd(&a[j + 56], va7);
    }

    for (int64_t j = n64; j < n; ++j) {
        a[j] = a[j] * b[j] * c[j];
    }
}
