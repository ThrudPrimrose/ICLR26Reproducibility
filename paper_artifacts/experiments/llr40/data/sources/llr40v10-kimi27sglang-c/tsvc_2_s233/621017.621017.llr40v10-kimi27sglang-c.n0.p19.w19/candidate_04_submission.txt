#include <stdint.h>

#ifndef AA_BLOCK
#define AA_BLOCK 128
#endif

void tsvc_2_s233_fp64(double *restrict aa, double *restrict bb, const double *restrict cc, const int64_t LEN_2D) {

    #pragma omp parallel
    {
        // aa recurrence: aa[j,i] depends on aa[j-1,i] (same column)
        // Process columns in blocks; each block is contiguous in memory and the inner i loop vectorizes.
        #pragma omp for schedule(static) nowait
        for (int64_t ib = 8; ib < LEN_2D; ib += AA_BLOCK) {
            int64_t iend = ib + AA_BLOCK < LEN_2D ? ib + AA_BLOCK : LEN_2D;
            const double *aa_prev = &aa[7 * LEN_2D + ib];
            const double *cc_base = &cc[8 * LEN_2D + ib];
            double *aa_base = &aa[8 * LEN_2D + ib];
            for (int64_t j = 8; j < LEN_2D; ++j) {
                const double *cc_row = cc_base + (j - 8) * LEN_2D;
                const double *aa_prev_row = aa_prev + (j - 8) * LEN_2D;
                double *aa_row = aa_base + (j - 8) * LEN_2D;
                #pragma omp simd
                for (int64_t k = 0; k < iend - ib; ++k) {
                    aa_row[k] = aa_prev_row[k] + cc_row[k];
                }
            }
        }

        // bb recurrence: bb[j,i] depends on bb[j,i-1] (same row)
        // Each row is independent; process rows in parallel with sequential recurrence along contiguous columns.
        #pragma omp for schedule(static) nowait
        for (int64_t j = 8; j < LEN_2D; ++j) {
            const double *cc_row = &cc[j * LEN_2D + 8];
            double *bb_row = &bb[j * LEN_2D + 8];
            double prev = bb_row[-1];
            for (int64_t i = 0; i < LEN_2D - 8; ++i) {
                prev += cc_row[i];
                bb_row[i] = prev;
            }
        }
    }
}
