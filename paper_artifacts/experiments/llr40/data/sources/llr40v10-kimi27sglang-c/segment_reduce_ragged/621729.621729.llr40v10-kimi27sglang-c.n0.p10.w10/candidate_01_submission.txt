#include <stdint.h>

void segment_reduce_ragged_fp64(double *out, int64_t *row_ptr, double *val, double *w, int64_t NSEG, uint8_t *workspace, int64_t workspace_bytes) {
    (void)workspace;
    (void)workspace_bytes;
    #pragma omp parallel for schedule(guided)
    for (int64_t s = 0; s < NSEG; s++) {
        int64_t start = row_ptr[s];
        int64_t end = row_ptr[s + 1];
        double acc = 0.0;
        #pragma omp simd reduction(+:acc)
        for (int64_t e = start; e < end; e++) {
            acc += val[e] * w[e];
        }
        out[s] = acc;
    }
}
