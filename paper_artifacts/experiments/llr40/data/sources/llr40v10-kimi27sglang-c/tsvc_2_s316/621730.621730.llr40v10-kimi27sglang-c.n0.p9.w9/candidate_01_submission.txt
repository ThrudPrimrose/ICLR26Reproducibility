#include <stdint.h>

void tsvc_2_s316_fp64(const double *restrict a, double *restrict result, const int64_t LEN_1D) {
    double x = a[0];

    #pragma omp parallel for reduction(min:x)
    for (int64_t i = 1; i < LEN_1D; ++i) {
        if (a[i] < x) {
            x = a[i];
        }
    }

    result[0] = x;
}
