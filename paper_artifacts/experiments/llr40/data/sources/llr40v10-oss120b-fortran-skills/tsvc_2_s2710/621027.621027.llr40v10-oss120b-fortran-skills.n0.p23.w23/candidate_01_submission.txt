
subroutine tsvc_2_s2710_fp64(a, b, c, d, e, x, len1d) bind(C)
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: len1d
  real(c_double), intent(inout) :: a(len1d), b(len1d), c(len1d)
  real(c_double), intent(in) :: d(len1d), e(len1d), x(len1d)
  logical :: len_gt_10, x_positive
  integer(c_int64_t) :: i
  len_gt_10 = len1d > 10_c_int64_t
  x_positive = x(1) > 0.0_c_double
  !$omp parallel do simd schedule(static) shared(a,b,c,d,e,x,len1d,len_gt_10,x_positive) private(i)
  do i = 1, len1d
    if (a(i) > b(i)) then
      a(i) = a(i) + b(i) * d(i)
      if (len_gt_10) then
        c(i) = c(i) + d(i) * d(i)
      else
        c(i) = d(i) * e(i) + 1.0_c_double
      end if
    else
      b(i) = a(i) + e(i) * e(i)
      if (x_positive) then
        c(i) = a(i) + d(i) * d(i)
      else
        c(i) = c(i) + e(i) * e(i)
      end if
    end if
  end do
  !$omp end parallel do simd
end subroutine tsvc_2_s2710_fp64
