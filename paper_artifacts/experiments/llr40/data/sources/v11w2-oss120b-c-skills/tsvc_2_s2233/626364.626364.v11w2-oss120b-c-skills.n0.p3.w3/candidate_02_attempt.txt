#include <stdint.h>
#include <omp.h>

void tsvc_2_s2233_fp64(double *restrict aa, double *restrict bb, const double *restrict cc, const int64_t LEN_2D) {
    // Parallel region to reuse threads for both phases.
    #pragma omp parallel
    {
        // Phase 1: Update AA. Each thread processes a subset of columns (i).
        #pragma omp for schedule(static)
        for (int64_t i = 8; i < LEN_2D; ++i) {
            // Unrolled inner loop over rows (j) to reduce loop overhead while preserving the recurrence.
            int64_t j = 8;
            for (; j + 3 < LEN_2D; j += 4) {
                aa[(j) * LEN_2D + i] = aa[(j - 1) * LEN_2D + i] + cc[(j) * LEN_2D + i];
                aa[(j + 1) * LEN_2D + i] = aa[(j) * LEN_2D + i] + cc[(j + 1) * LEN_2D + i];
                aa[(j + 2) * LEN_2D + i] = aa[(j + 1) * LEN_2D + i] + cc[(j + 2) * LEN_2D + i];
                aa[(j + 3) * LEN_2D + i] = aa[(j + 2) * LEN_2D + i] + cc[(j + 3) * LEN_2D + i];
            }
            for (; j < LEN_2D; ++j) {
                aa[j * LEN_2D + i] = aa[(j - 1) * LEN_2D + i] + cc[j * LEN_2D + i];
            }
        }
        // Implicit barrier here ensures AA phase is complete before BB phase begins.

        // Phase 2: Update BB. This is a row‑wise prefix sum; the outer loop over rows (i) is sequential,
        // but the inner loop over columns (j) can be executed in parallel by the team.
        for (int64_t i = 8; i < LEN_2D; ++i) {
            #pragma omp for schedule(static)
            for (int64_t j = 8; j < LEN_2D; ++j) {
                bb[i * LEN_2D + j] = bb[(i - 1) * LEN_2D + j] + cc[i * LEN_2D + j];
            }
            // Implicit barrier after the omp for ensures the row is fully computed before the next i.
        }
    }
}
