/* TSVC s119: aa[i][j] = aa[i-1][j-1] + bb[i][j]  for i,j in 1..N-1.
 *
 * Dependence: a row depends only on the previous row (shifted by one column).
 * Parallelize by column strips: each thread owns a contiguous set of columns
 * for ALL rows. The only cross-thread value needed is the left neighbour's
 * last column of the previous row, guarded by a per-thread release/acquire
 * counter (no barriers). Per-element arithmetic is the exact same single FADD
 * in the same order as the reference, so results are bit-identical.
 *
 * Small N: serial AVX-512 path (thread launch would dominate).
 */
#include <stdint.h>
#include <string.h>
#include <omp.h>
#include <immintrin.h>

static inline void row_zmm(double *restrict o, const double *restrict a,
                           const double *restrict b, int64_t m) {
    int64_t k = 0;
    for (; k + 32 <= m; k += 32) {
        __m512d x0 = _mm512_add_pd(_mm512_loadu_pd(a + k),     _mm512_loadu_pd(b + k));
        __m512d x1 = _mm512_add_pd(_mm512_loadu_pd(a + k + 8), _mm512_loadu_pd(b + k + 8));
        __m512d x2 = _mm512_add_pd(_mm512_loadu_pd(a + k + 16),_mm512_loadu_pd(b + k + 16));
        __m512d x3 = _mm512_add_pd(_mm512_loadu_pd(a + k + 24),_mm512_loadu_pd(b + k + 24));
        _mm512_storeu_pd(o + k,     x0);
        _mm512_storeu_pd(o + k + 8, x1);
        _mm512_storeu_pd(o + k + 16, x2);
        _mm512_storeu_pd(o + k + 24, x3);
    }
    for (; k < m; ++k) o[k] = a[k] + b[k];
}

#define MAX_STRIPS 256

void tsvc_2_s119_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
    const int64_t N = LEN_2D;
    if (N < 2) return;
    const int64_t W = N - 1; /* number of interior columns j = 1..N-1 */

    int P = (N < 2000) ? 1 : omp_get_max_threads();
    if (P < 1) P = 1;
    if (P > W) P = (int)W;
    if (P > MAX_STRIPS) P = MAX_STRIPS;

    const int64_t base = W / P;
    const int64_t rem  = W % P;

    if (P == 1) {
        const double *a = aa;                 /* row 0, col j0-1 = 0 */
        const double *b = bb + N + 1;         /* row 1, col 1   */
        double *o       = aa + N + 1;         /* row 1, col 1   */
        for (int64_t i = 1; i < N; ++i) {
            row_zmm(o, a, b, W);
            a += N; b += N; o += N;
        }
        return;
    }

    _Alignas(64) int64_t done[MAX_STRIPS];
    memset(done, 0, sizeof(done));

    #pragma omp parallel num_threads(P)
    {
        const int c = omp_get_thread_num();
        const int64_t j0 = 1 + c * base + ((c < rem) ? c : rem);
        const int64_t j1 = 1 + (c + 1) * base + ((c + 1 < rem) ? c + 1 : rem);
        const int64_t m  = j1 - j0;
        const double *a  = aa + j0 - 1;              /* row 0, col j0-1 */
        const double *b  = bb + N + j0;              /* row 1, col j0   */
        double *o        = aa + N + j0;              /* row 1, col j0   */
        for (int64_t i = 1; i < N; ++i) {
            if (c > 0) {
                while (__atomic_load_n(&done[c - 1], __ATOMIC_ACQUIRE) < i - 1)
                    __builtin_ia32_pause();
            }
            row_zmm(o, a, b, m);
            __atomic_store_n(&done[c], i, __ATOMIC_RELEASE);
            a += N; b += N; o += N;
        }
    }
}
