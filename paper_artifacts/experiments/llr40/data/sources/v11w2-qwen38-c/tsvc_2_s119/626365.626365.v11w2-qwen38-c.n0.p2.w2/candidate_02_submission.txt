/* TSVC s119: aa[i][j] = aa[i-1][j-1] + bb[i][j]  for i,j in 1..N-1.
 *
 * Column-strip parallelism: thread c owns contiguous columns for all rows;
 * row i depends only on row i-1, so strips are independent chains with a
 * single release/acquire counter per thread guarding the left edge column.
 * Exact same FADD order as the reference -> bit-identical results.
 *
 * Memory details:
 *  - stores are made 64B aligned (scalar prologue) so full-line stores
 *    do not trigger read-for-ownership partial-line fetches;
 *  - 16 loads kept in flight (8x ZMM unroll, load phase before add phase);
 *  - next row of bb prefetched while processing current row.
 */
#include <stdint.h>
#include <string.h>
#include <omp.h>
#include <immintrin.h>

static inline void row_zmm(double *restrict o, const double *restrict a,
                           const double *restrict b, int64_t m,
                           const double *restrict bnx) {
    int64_t k = 0;
    /* scalar prologue: align stores to 64B */
    while (k < m && ((uintptr_t)(o + k) & 63)) { o[k] = a[k] + b[k]; ++k; }
    for (; k + 64 <= m; k += 64) {
        __builtin_prefetch(bnx + k, 0, 0);
        const double *pa = a + k, *pb = b + k;
        __m512d la[8], lb[8];
        #pragma GCC unroll 8
        for (int t = 0; t < 8; ++t) la[t] = _mm512_loadu_pd(pa + 8*t);
        #pragma GCC unroll 8
        for (int t = 0; t < 8; ++t) lb[t] = _mm512_loadu_pd(pb + 8*t);
        #pragma GCC unroll 8
        for (int t = 0; t < 8; ++t) _mm512_storeu_pd(o + k + 8*t, _mm512_add_pd(la[t], lb[t]));
    }
    for (; k < m; ++k) o[k] = a[k] + b[k];
}

#define MAX_STRIPS 256

void tsvc_2_s119_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
    const int64_t N = LEN_2D;
    if (N < 2) return;
    const int64_t W = N - 1;

    int P = (N < 2000) ? 1 : omp_get_max_threads();
    if (P < 1) P = 1;
    if (P > W) P = (int)W;
    if (P > MAX_STRIPS) P = MAX_STRIPS;

    const int64_t base = W / P;
    const int64_t rem  = W % P;

    if (P == 1) {
        const double *a = aa;
        const double *b = bb + N + 1;
        double *o       = aa + N + 1;
        for (int64_t i = 1; i < N; ++i) {
            row_zmm(o, a, b, W, b + N);
            a += N; b += N; o += N;
        }
        return;
    }

    _Alignas(64) int64_t done[MAX_STRIPS];
    memset(done, 0, sizeof(done));

    #pragma omp parallel num_threads(P)
    {
        const int c = omp_get_thread_num();
        const int64_t j0 = 1 + c * base + ((c < rem) ? c : rem);
        const int64_t j1 = 1 + (c + 1) * base + ((c + 1 < rem) ? c + 1 : rem);
        const int64_t m  = j1 - j0;
        const double *a  = aa + j0 - 1;
        const double *b  = bb + N + j0;
        double *o        = aa + N + j0;
        for (int64_t i = 1; i < N; ++i) {
            if (c > 0) {
                while (__atomic_load_n(&done[c - 1], __ATOMIC_ACQUIRE) < i - 1)
                    __builtin_ia32_pause();
            }
            row_zmm(o, a, b, m, b + N);
            __atomic_store_n(&done[c], i, __ATOMIC_RELEASE);
            a += N; b += N; o += N;
        }
    }
}
