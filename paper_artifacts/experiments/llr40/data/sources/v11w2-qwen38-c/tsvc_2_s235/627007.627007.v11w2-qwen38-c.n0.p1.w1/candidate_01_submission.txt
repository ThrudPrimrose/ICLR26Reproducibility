#include <omp.h>
#include <stdint.h>

void tsvc_2_s235_fp64(double *restrict a, double *restrict aa, const double *restrict b, const double *restrict bb,
                      const double *restrict c, const int64_t LEN_2D) {
  if (LEN_2D <= 0) return;
  #pragma omp parallel
  {
    int nt  = omp_get_num_threads();
    int tid = omp_get_thread_num();
    int64_t base = LEN_2D / nt;
    int64_t rem  = LEN_2D % nt;
    int64_t i0 = base * tid + (tid < rem ? tid : rem);
    int64_t i1 = i0 + base + (tid < rem ? 1 : 0);

    /* step 1: a[i] += b[i]*c[i] for my block */
    for (int64_t i = i0; i < i1; ++i)
      a[i] += b[i] * c[i];

    /* step 2: j serial, i contiguous & vectorized within block */
    for (int64_t j = 1; j < LEN_2D; ++j) {
      double *aa_j   = aa + j * LEN_2D;
      const double *aa_jm1 = aa + (j - 1) * LEN_2D;
      const double *bb_j   = bb + j * LEN_2D;
      for (int64_t i = i0; i < i1; ++i)
        aa_j[i] = aa_jm1[i] + bb_j[i] * a[i];
    }
  }
}
