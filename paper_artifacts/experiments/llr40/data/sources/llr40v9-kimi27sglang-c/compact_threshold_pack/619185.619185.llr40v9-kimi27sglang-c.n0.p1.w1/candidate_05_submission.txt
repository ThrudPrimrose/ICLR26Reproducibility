#include <stdint.h>
#include <stdlib.h>
#include <string.h>
#include <omp.h>
#include <immintrin.h>

typedef struct {
    int64_t c;
    char pad[56];
} count_pad_t;

static count_pad_t g_counts[2048];

static inline void serial_pack(int64_t *restrict out_count,
                                 double *restrict packed,
                                 const double *restrict src,
                                 const double *restrict weight,
                                 const int64_t LEN_1D)
{
    int64_t n = 0;
    for (int64_t i = 0; i < LEN_1D; ++i) {
        const double v = src[i];
        if (v > 0.0) {
            packed[n++] = v * weight[i];
        }
    }
    out_count[0] = n;
}

void compact_threshold_pack_fp64(int64_t *restrict out_count,
                                   double *restrict packed,
                                   const double *restrict src,
                                   const double *restrict weight,
                                   const int64_t LEN_1D,
                                   uint8_t *restrict workspace,
                                   const int64_t workspace_bytes)
{
    (void)workspace;
    (void)workspace_bytes;

    if (LEN_1D <= 0) {
        out_count[0] = 0;
        return;
    }

    if (LEN_1D < 8192) {
        serial_pack(out_count, packed, src, weight, LEN_1D);
        return;
    }

    const int nt = omp_get_max_threads();
    if (nt > 2048) {
        serial_pack(out_count, packed, src, weight, LEN_1D);
        return;
    }

    const int64_t base = (LEN_1D / nt) & ~((int64_t)7);
    const int64_t last_lo = (int64_t)(nt - 1) * base;

    int64_t offset = 0;
    const __m512d zero = _mm512_setzero_pd();

    #pragma omp parallel
    {
        const int tid = omp_get_thread_num();
        int64_t lo, hi;
        if (tid < nt - 1) {
            lo = (int64_t)tid * base;
            hi = lo + base;
        } else {
            lo = last_lo;
            hi = LEN_1D;
        }

        int64_t c = 0;
        int64_t i = lo;
        const int64_t hi_vec = hi - (hi - lo) % 8;
        for (; i < hi_vec; i += 8) {
            __m512d v = _mm512_loadu_pd(src + i);
            __mmask8 m = _mm512_cmp_pd_mask(v, zero, _CMP_GT_OQ);
            c += (int64_t)__builtin_popcount((unsigned)m);
        }
        for (; i < hi; ++i) {
            if (src[i] > 0.0) ++c;
        }
        g_counts[tid].c = c;

        #pragma omp barrier
        #pragma omp masked
        {
            offset = 0;
            for (int j = 0; j < nt; ++j) {
                const int64_t cc = g_counts[j].c;
                g_counts[j].c = offset;
                offset += cc;
            }
            g_counts[nt].c = offset;
        }
        #pragma omp barrier

        double tmp[8] __attribute__((aligned(64)));
        int64_t pos = g_counts[tid].c;
        i = lo;
        for (; i < hi_vec; i += 8) {
            __m512d s = _mm512_loadu_pd(src + i);
            __m512d w = _mm512_loadu_pd(weight + i);
            __mmask8 m = _mm512_cmp_pd_mask(s, zero, _CMP_GT_OQ);
            __m512d p = _mm512_mul_pd(s, w);
            __m512d comp = _mm512_maskz_compress_pd(m, p);
            int k = __builtin_popcount((unsigned)m);
            _mm512_store_pd(tmp, comp);
            memcpy(packed + pos, tmp, (size_t)k * sizeof(double));
            pos += k;
        }
        for (; i < hi; ++i) {
            const double v = src[i];
            if (v > 0.0) {
                packed[pos++] = v * weight[i];
            }
        }
    }

    out_count[0] = offset;
}
