
#include <stdint.h>
#include <math.h>
#include <float.h>
#include <immintrin.h>
#include <omp.h>
#include <limits.h>

typedef struct {
    double value;
    int64_t index;
} maxloc_t;

static inline maxloc_t maxloc_init(void) {
    maxloc_t m;
    m.value = -DBL_MAX;
    m.index = INT64_MAX;
    return m;
}

static inline maxloc_t maxloc_combine(maxloc_t a, maxloc_t b) {
    if (b.value > a.value || (b.value == a.value && b.index < a.index))
        return b;
    return a;
}

#pragma omp declare reduction(argmax_reduce : maxloc_t : \
    omp_out = maxloc_combine(omp_out, omp_in)) \
    initializer(omp_priv = maxloc_init())

static inline maxloc_t argmax_chunk(const double *restrict a,
                                    int64_t lo, int64_t hi) {
    double cur_val = a[lo];
    int64_t cur_idx = lo;
    int64_t i = lo + 1;

    /* scalar prefix to reach an 8-element (64-byte) boundary */
    for (; i < hi && (i & 7); ++i) {
        double ai = a[i];
        if (ai > cur_val) {
            cur_val = ai;
            cur_idx = i;
        }
    }

#if defined(__AVX512F__)
    /* 8x unrolled aligned AVX-512 scan. */
    if (i + 63 < hi) {
        __m512d vmax0 = _mm512_set1_pd(cur_val);
        __m512d vmax1 = _mm512_set1_pd(cur_val);
        __m512d vmax2 = _mm512_set1_pd(cur_val);
        __m512d vmax3 = _mm512_set1_pd(cur_val);
        __m512d vmax4 = _mm512_set1_pd(cur_val);
        __m512d vmax5 = _mm512_set1_pd(cur_val);
        __m512d vmax6 = _mm512_set1_pd(cur_val);
        __m512d vmax7 = _mm512_set1_pd(cur_val);
        __m512i vidx0 = _mm512_set1_epi64(cur_idx);
        __m512i vidx1 = _mm512_set1_epi64(cur_idx);
        __m512i vidx2 = _mm512_set1_epi64(cur_idx);
        __m512i vidx3 = _mm512_set1_epi64(cur_idx);
        __m512i vidx4 = _mm512_set1_epi64(cur_idx);
        __m512i vidx5 = _mm512_set1_epi64(cur_idx);
        __m512i vidx6 = _mm512_set1_epi64(cur_idx);
        __m512i vidx7 = _mm512_set1_epi64(cur_idx);
        __m512i base0 = _mm512_set_epi64(i + 7, i + 6, i + 5, i + 4, i + 3, i + 2, i + 1, i + 0);
        __m512i base1 = _mm512_set_epi64(i + 15, i + 14, i + 13, i + 12, i + 11, i + 10, i + 9, i + 8);
        __m512i base2 = _mm512_set_epi64(i + 23, i + 22, i + 21, i + 20, i + 19, i + 18, i + 17, i + 16);
        __m512i base3 = _mm512_set_epi64(i + 31, i + 30, i + 29, i + 28, i + 27, i + 26, i + 25, i + 24);
        __m512i base4 = _mm512_set_epi64(i + 39, i + 38, i + 37, i + 36, i + 35, i + 34, i + 33, i + 32);
        __m512i base5 = _mm512_set_epi64(i + 47, i + 46, i + 45, i + 44, i + 43, i + 42, i + 41, i + 40);
        __m512i base6 = _mm512_set_epi64(i + 55, i + 54, i + 53, i + 52, i + 51, i + 50, i + 49, i + 48);
        __m512i base7 = _mm512_set_epi64(i + 63, i + 62, i + 61, i + 60, i + 59, i + 58, i + 57, i + 56);
        const __m512i inc64 = _mm512_set1_epi64(64);
        for (; i + 63 < hi; i += 64) {
            {
                __m512d va = _mm512_loadu_pd(a + i + 0);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax0, _CMP_GT_OQ);
                vmax0 = _mm512_mask_blend_pd(m, vmax0, va);
                vidx0 = _mm512_mask_blend_epi64(m, vidx0, base0);
            }
            {
                __m512d va = _mm512_loadu_pd(a + i + 8);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax1, _CMP_GT_OQ);
                vmax1 = _mm512_mask_blend_pd(m, vmax1, va);
                vidx1 = _mm512_mask_blend_epi64(m, vidx1, base1);
            }
            {
                __m512d va = _mm512_loadu_pd(a + i + 16);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax2, _CMP_GT_OQ);
                vmax2 = _mm512_mask_blend_pd(m, vmax2, va);
                vidx2 = _mm512_mask_blend_epi64(m, vidx2, base2);
            }
            {
                __m512d va = _mm512_loadu_pd(a + i + 24);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax3, _CMP_GT_OQ);
                vmax3 = _mm512_mask_blend_pd(m, vmax3, va);
                vidx3 = _mm512_mask_blend_epi64(m, vidx3, base3);
            }
            {
                __m512d va = _mm512_loadu_pd(a + i + 32);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax4, _CMP_GT_OQ);
                vmax4 = _mm512_mask_blend_pd(m, vmax4, va);
                vidx4 = _mm512_mask_blend_epi64(m, vidx4, base4);
            }
            {
                __m512d va = _mm512_loadu_pd(a + i + 40);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax5, _CMP_GT_OQ);
                vmax5 = _mm512_mask_blend_pd(m, vmax5, va);
                vidx5 = _mm512_mask_blend_epi64(m, vidx5, base5);
            }
            {
                __m512d va = _mm512_loadu_pd(a + i + 48);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax6, _CMP_GT_OQ);
                vmax6 = _mm512_mask_blend_pd(m, vmax6, va);
                vidx6 = _mm512_mask_blend_epi64(m, vidx6, base6);
            }
            {
                __m512d va = _mm512_loadu_pd(a + i + 56);
                __mmask8 m = _mm512_cmp_pd_mask(va, vmax7, _CMP_GT_OQ);
                vmax7 = _mm512_mask_blend_pd(m, vmax7, va);
                vidx7 = _mm512_mask_blend_epi64(m, vidx7, base7);
            }
            base0 = _mm512_add_epi64(base0, inc64);
            base1 = _mm512_add_epi64(base1, inc64);
            base2 = _mm512_add_epi64(base2, inc64);
            base3 = _mm512_add_epi64(base3, inc64);
            base4 = _mm512_add_epi64(base4, inc64);
            base5 = _mm512_add_epi64(base5, inc64);
            base6 = _mm512_add_epi64(base6, inc64);
            base7 = _mm512_add_epi64(base7, inc64);
        }
        _Alignas(64) double vals[64];
        _Alignas(64) int64_t idxs[64];
        _mm512_store_pd(vals + 0, vmax0);
        _mm512_store_pd(vals + 8, vmax1);
        _mm512_store_pd(vals + 16, vmax2);
        _mm512_store_pd(vals + 24, vmax3);
        _mm512_store_pd(vals + 32, vmax4);
        _mm512_store_pd(vals + 40, vmax5);
        _mm512_store_pd(vals + 48, vmax6);
        _mm512_store_pd(vals + 56, vmax7);
        _mm512_store_epi64(idxs + 0, vidx0);
        _mm512_store_epi64(idxs + 8, vidx1);
        _mm512_store_epi64(idxs + 16, vidx2);
        _mm512_store_epi64(idxs + 24, vidx3);
        _mm512_store_epi64(idxs + 32, vidx4);
        _mm512_store_epi64(idxs + 40, vidx5);
        _mm512_store_epi64(idxs + 48, vidx6);
        _mm512_store_epi64(idxs + 56, vidx7);
        for (int k = 0; k < 64; ++k) {
            double v = vals[k];
            int64_t idx = idxs[k];
            if (v > cur_val || (v == cur_val && idx < cur_idx)) {
                cur_val = v;
                cur_idx = idx;
            }
        }
    }

    if (i + 7 < hi) {
        __m512d vmax = _mm512_set1_pd(cur_val);
        __m512i vidx = _mm512_set1_epi64(cur_idx);
        __m512i base = _mm512_set_epi64(i + 7, i + 6, i + 5, i + 4,
                                        i + 3, i + 2, i + 1, i);
        const __m512i inc8 = _mm512_set1_epi64(8);
        for (; i + 7 < hi; i += 8) {
            __m512d va = _mm512_loadu_pd(a + i);
            __mmask8 m = _mm512_cmp_pd_mask(va, vmax, _CMP_GT_OQ);
            vmax = _mm512_mask_blend_pd(m, vmax, va);
            vidx = _mm512_mask_blend_epi64(m, vidx, base);
            base = _mm512_add_epi64(base, inc8);
        }
        _Alignas(64) double vals[8];
        _Alignas(64) int64_t idxs[8];
        _mm512_store_pd(vals, vmax);
        _mm512_store_epi64(idxs, vidx);
        for (int k = 0; k < 8; ++k) {
            double v = vals[k];
            int64_t idx = idxs[k];
            if (v > cur_val || (v == cur_val && idx < cur_idx)) {
                cur_val = v;
                cur_idx = idx;
            }
        }
    }
#endif

    for (; i < hi; ++i) {
        double ai = a[i];
        if (ai > cur_val) {
            cur_val = ai;
            cur_idx = i;
        }
    }

    maxloc_t res;
    res.value = cur_val;
    res.index = cur_idx;
    return res;
}

void argmax_with_index_fp64(const double *restrict a,
                            int64_t *restrict out_index,
                            double *restrict out_value,
                            const int64_t LEN_1D) {
    if (LEN_1D <= 0) {
        *out_value = 0.0;
        *out_index = 0;
        return;
    }
    if (LEN_1D == 1) {
        *out_value = a[0];
        *out_index = 0;
        return;
    }

    if (LEN_1D <= 256) {
        double cur_val = a[0];
        int64_t cur_idx = 0;
        for (int64_t i = 1; i < LEN_1D; ++i) {
            double ai = a[i];
            if (ai > cur_val) {
                cur_val = ai;
                cur_idx = i;
            }
        }
        *out_value = cur_val;
        *out_index = cur_idx;
        return;
    }

    if (__builtin_isnan(a[0])) {
        *out_value = a[0];
        *out_index = 0;
        return;
    }

#if defined(__AVX512F__)
    if (LEN_1D <= 65536) {
        maxloc_t r = argmax_chunk(a, 0, LEN_1D);
        *out_value = r.value;
        *out_index = r.index;
        return;
    }

    maxloc_t global = maxloc_init();
    const int64_t nthreads = omp_get_max_threads();
    const int64_t chunk = (LEN_1D + nthreads - 1) / nthreads;

    #pragma omp parallel for schedule(static) \
        reduction(argmax_reduce : global)
    for (int64_t t = 0; t < nthreads; ++t) {
        int64_t lo = t * chunk;
        int64_t hi = lo + chunk;
        if (hi > LEN_1D) hi = LEN_1D;
        if (lo < hi)
            global = maxloc_combine(global, argmax_chunk(a, lo, hi));
    }

    *out_value = global.value;
    *out_index = global.index;
#else
    maxloc_t r = argmax_chunk(a, 0, LEN_1D);
    *out_value = r.value;
    *out_index = r.index;
#endif
}
