#include <stdint.h>
#include <stddef.h>

#ifdef _OPENMP
#include <omp.h>
#endif

void scan_affine_decay_fp64(const double *restrict c,
                            const double *restrict x,
                            double *restrict y,
                            int64_t n,
                            uint8_t *restrict workspace,
                            int64_t workspace_size)
{
    if (n <= 0) {
        return;
    }

    int nt = 1;
#ifdef _OPENMP
    nt = omp_get_max_threads();
    if (nt < 1) nt = 1;
#endif
    if (nt > n) nt = (int)n;

    const int64_t chunk = (n + nt - 1) / nt;       /* each thread gets one contiguous chunk */
    const int blocks = (int)((n + chunk - 1) / chunk);

    /* Workspace: blocks * 2 doubles */
    double *restrict AB;
    double local_ab[64];
    int64_t need = (int64_t)blocks * 2 * (int64_t)sizeof(double);
    if (workspace != (uint8_t *)0 && workspace_size >= need) {
        AB = (double *restrict)workspace;
    } else {
        if ((size_t)need <= sizeof(local_ab)) {
            AB = local_ab;
        } else {
            AB = (double *restrict)__builtin_alloca((size_t)need);
        }
    }

    /* Pass 1: each thread computes the affine transform of its chunk.
       No writes to y yet. */
    #pragma omp parallel for schedule(static, 1)
    for (int b = 0; b < blocks; ++b) {
        int64_t l = (int64_t)b * chunk;
        int64_t r = l + chunk;
        if (r > n) r = n;
        double acc = x[l];
        double a = c[l];
        for (int64_t i = l + 1; i < r; ++i) {
            acc = c[i] * acc + x[i];
            a *= c[i];
        }
        AB[2 * b] = a;
        AB[2 * b + 1] = acc;
    }

    /* Pass 2: serial prefix of the block transforms. */
    double carry = 0.0;
    for (int b = 0; b < blocks; ++b) {
        double a = AB[2 * b];
        double bval = AB[2 * b + 1];
        AB[2 * b + 1] = carry;          /* incoming carry for block b */
        carry = a * carry + bval;
    }

    /* Pass 3: apply carries and write final y.
       With static scheduling the same thread revisits its chunk, so c/x are
       likely still in cache. */
    #pragma omp parallel for schedule(static, 1)
    for (int b = 0; b < blocks; ++b) {
        int64_t l = (int64_t)b * chunk;
        int64_t r = l + chunk;
        if (r > n) r = n;
        double carry_in = AB[2 * b + 1];
        double acc = c[l] * carry_in + x[l];
        y[l] = acc;
        for (int64_t i = l + 1; i < r; ++i) {
            acc = c[i] * acc + x[i];
            y[i] = acc;
        }
    }
}
