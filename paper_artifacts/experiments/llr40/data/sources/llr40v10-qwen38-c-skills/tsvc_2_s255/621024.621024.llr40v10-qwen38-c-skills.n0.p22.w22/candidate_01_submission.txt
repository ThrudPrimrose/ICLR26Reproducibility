#include <stdint.h>
#include <omp.h>

/*
 * Reference (serial):
 *   x = b[N-1]; y = b[N-2];
 *   for i in 0..N-1: a[i] = (b[i] + x + y) * 0.333;  y = x;  x = b[i];
 *
 * The scalar carry (x, y) is a sliding window over b, so after i >= 2 it holds
 * exactly x = b[i-1], y = b[i-2] (plain copies, no rounding).  Only the first
 * two iterations touch the wrap-around values b[N-1], b[N-2].  Peeling them
 * leaves a fully independent loop: a[i] = (b[i] + b[i-1] + b[i-2]) * 0.333.
 * The add order (left to right) matches the reference bit-for-bit.
 */
void tsvc_2_s255_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D) {
  if (LEN_1D <= 0) return;

  const double x0 = b[LEN_1D - 1];
  const double y0 = b[LEN_1D - 2];

  /* i = 0: x = b[N-1], y = b[N-2] */
  a[0] = (b[0] + x0 + y0) * 0.333;
  if (LEN_1D == 1) return;

  /* i = 1: x = b[0], y = b[N-1] */
  a[1] = (b[1] + b[0] + x0) * 0.333;
  if (LEN_1D == 2) return;

  /* i >= 2: independent across i -- thread + vectorize. */
  #pragma omp parallel for schedule(static)
  for (int64_t i = 2; i < LEN_1D; i++) {
    a[i] = (b[i] + b[i - 1] + b[i - 2]) * 0.333;
  }
}
