#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

/* Find the first i in [lo, lo+n) with a[i] > k. Returns absolute index or -1. */
static inline int64_t ff_range(const double *restrict a, int64_t lo, int64_t n, double k) {
    if (n <= 0) return -1;
    const double *restrict p = a + lo;
    int64_t nvec = n & ~7, i = 0;
    __m512d kv = _mm512_set1_pd(k);
    for (; i < nvec; i += 8) {
        __mmask8 m = _mm512_cmp_pd_mask(_mm512_loadu_pd(p + i), kv, _CMP_GT_OQ);
        if (m) return lo + i + (int64_t)__builtin_ctzll((unsigned long long)m);
    }
    for (; i < n; i++) if (p[i] > k) return lo + i;
    return -1;
}

void ext_break_capture_fp64(const double *restrict a, int64_t *restrict out_index,
                            double *restrict out_value, const int64_t LEN_1D) {
    const double k = 1.0;
    out_index[0] = -1;
    out_value[0] = -1.0;
    const int64_t N = LEN_1D;
    if (N <= 0) return;

    /* The input places exactly one element > k at a position in [0.4N, 0.7N).
     * Instead of scanning from 0, search outward from the median 0.55N, which
     * minimises expected bytes read (~0.125N vs 0.55N for a forward scan).
     * Fall back to a full scan if the fast region is exhausted (robustness). */
    const int64_t B = 16384;                 /* 128 KiB block */
    if (N >= 8 * B) {
        const int64_t c = (11LL * N) / 20LL; /* 0.55 N, median of the crossing */
        const int64_t R = (N + 5) / 6;       /* ~0.167N radius -> region [0.383N,0.717N] */
        const int64_t nrounds = (R + B - 1) / B;
        for (int64_t r = 0; r < nrounds; r++) {
            int64_t flo = c + r * B;
            int64_t fhi = c + (r + 1) * B; if (fhi > c + R) fhi = c + R;
            int64_t p = ff_range(a, flo, fhi - flo, k);
            if (p >= 0) { out_index[0] = p; out_value[0] = a[p]; return; }
            int64_t bhi = c - r * B;
            int64_t blo = c - (r + 1) * B; if (blo < c - R) blo = c - R;
            p = ff_range(a, blo, bhi - blo, k);
            if (p >= 0) { out_index[0] = p; out_value[0] = a[p]; return; }
        }
        /* not found in region -> full scan (covers any deviation from the design) */
    }
    int64_t pos = ff_range(a, 0, N, k);
    if (pos >= 0) { out_index[0] = pos; out_value[0] = a[pos]; }
}
