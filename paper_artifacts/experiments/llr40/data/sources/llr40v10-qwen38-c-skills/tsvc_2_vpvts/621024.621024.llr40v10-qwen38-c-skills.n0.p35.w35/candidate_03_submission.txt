#include <stdint.h>
#include <omp.h>
#include <immintrin.h>
void tsvc_2_vpvts_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D, const int64_t S) {
  const double sd = (double)S;
  const int64_t HALF = LEN_1D / 2;
  const int64_t nfull = HALF / 64;
  #pragma omp parallel for schedule(static)
  for (int64_t c = 0; c < nfull; ++c) {
    const int64_t j = c * 64;
    const int64_t k = j + HALF;
    __m512d s = _mm512_set1_pd(sd);
    for (int64_t off = 0; off < 64; off += 8) {
      __m512d a0 = _mm512_loadu_pd(a + j + off), b0 = _mm512_loadu_pd(b + j + off);
      __m512d a1 = _mm512_loadu_pd(a + k + off), b1 = _mm512_loadu_pd(b + k + off);
      _mm512_storeu_pd(a + j + off, _mm512_add_pd(a0, _mm512_mul_pd(b0, s)));
      _mm512_storeu_pd(a + k + off, _mm512_add_pd(a1, _mm512_mul_pd(b1, s)));
    }
  }
  for (int64_t i = nfull * 64; i < HALF; ++i) a[i] += b[i] * sd;
  for (int64_t i = HALF + nfull * 64; i < LEN_1D; ++i) a[i] += b[i] * sd;
}
