
#include <stdint.h>
#include <omp.h>
#include <immintrin.h>
void tsvc_2_vpvts_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D, const int64_t S) {
  const double sd = (double)S;
  #pragma omp parallel for schedule(static)
  for (int64_t i = 0; i < LEN_1D; i += 32) {
    int64_t end = LEN_1D < i + 32 ? LEN_1D : i + 32;
    int64_t j = i;
    for (; j + 32 <= end; j += 32) {
      __m512d s = _mm512_set1_pd(sd);
      __m512d a0 = _mm512_loadu_pd(a + j + 0), b0 = _mm512_loadu_pd(b + j + 0);
      __m512d a1 = _mm512_loadu_pd(a + j + 8), b1 = _mm512_loadu_pd(b + j + 8);
      __m512d a2 = _mm512_loadu_pd(a + j + 16), b2 = _mm512_loadu_pd(b + j + 16);
      __m512d a3 = _mm512_loadu_pd(a + j + 24), b3 = _mm512_loadu_pd(b + j + 24);
      _mm512_storeu_pd(a + j + 0, _mm512_add_pd(a0, _mm512_mul_pd(b0, s)));
      _mm512_storeu_pd(a + j + 8, _mm512_add_pd(a1, _mm512_mul_pd(b1, s)));
      _mm512_storeu_pd(a + j + 16, _mm512_add_pd(a2, _mm512_mul_pd(b2, s)));
      _mm512_storeu_pd(a + j + 24, _mm512_add_pd(a3, _mm512_mul_pd(b3, s)));
    }
    for (; j + 8 <= end; j += 8) {
      __m512d s = _mm512_set1_pd(sd);
      _mm512_storeu_pd(a + j, _mm512_add_pd(_mm512_loadu_pd(a + j), _mm512_mul_pd(_mm512_loadu_pd(b + j), s)));
    }
    for (; j < end; ++j) a[j] += b[j] * sd;
  }
}
