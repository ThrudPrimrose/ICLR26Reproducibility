subroutine compact_threshold_pack_fp64(out_count, packed, src, weight, n, workspace, workspace_size) bind(C, name="compact_threshold_pack_fp64")
  use iso_c_binding, only: c_double, c_int64_t, c_int8_t
  implicit none
  integer(c_int64_t), value, intent(in) :: n, workspace_size
  integer(c_int64_t), intent(out) :: out_count(1)
  real(c_double), intent(out) :: packed(n)
  real(c_double), intent(in) :: src(n), weight(n)
  integer(c_int8_t), intent(inout) :: workspace(workspace_size)

  integer(c_int64_t), parameter :: BS = 65536
  integer :: b, j
  integer(c_int64_t) :: nblocks, lo, hi, idx, c, total
  integer(c_int64_t), allocatable :: cnt(:), off(:)
  integer(c_int8_t), allocatable :: mask(:)

  nblocks = (n + BS - 1) / BS
  allocate(cnt(0:nblocks-1), off(0:nblocks))
  cnt = 0
  off = 0

  if (workspace_size >= n) then
    !$omp parallel do schedule(static) private(b, lo, hi, idx, j, c)
    do b = 0, int(nblocks - 1)
      lo = int(b, c_int64_t) * BS + 1
      hi = min(n, lo + BS - 1)
      c = 0
      !$omp simd reduction(+:c)
      do j = 1, int(hi - lo + 1)
        idx = lo + int(j, c_int64_t) - 1
        if (src(idx) > 0.0d0) then
          workspace(idx) = 1
          c = c + 1
        else
          workspace(idx) = 0
        end if
      end do
      !$omp end simd
      cnt(b) = c
    end do
    !$omp end parallel do

    total = 0
    do b = 0, int(nblocks - 1)
      off(b) = total
      total = total + cnt(b)
    end do
    off(nblocks) = total

    !$omp parallel do schedule(dynamic, 2) private(b, lo, hi, idx, j, c)
    do b = 0, int(nblocks - 1)
      lo = int(b, c_int64_t) * BS + 1
      hi = min(n, lo + BS - 1)
      c = off(b)
      do j = 1, int(hi - lo + 1)
        idx = lo + int(j, c_int64_t) - 1
        if (workspace(idx) /= 0) then
          c = c + 1
          packed(c) = src(idx) * weight(idx)
        end if
      end do
    end do
    !$omp end parallel do
  else
    allocate(mask(n))
    mask = 0
    !$omp parallel do schedule(static) private(b, lo, hi, idx, j, c)
    do b = 0, int(nblocks - 1)
      lo = int(b, c_int64_t) * BS + 1
      hi = min(n, lo + BS - 1)
      c = 0
      !$omp simd reduction(+:c)
      do j = 1, int(hi - lo + 1)
        idx = lo + int(j, c_int64_t) - 1
        if (src(idx) > 0.0d0) then
          mask(idx) = 1
          c = c + 1
        else
          mask(idx) = 0
        end if
      end do
      !$omp end simd
      cnt(b) = c
    end do
    !$omp end parallel do

    total = 0
    do b = 0, int(nblocks - 1)
      off(b) = total
      total = total + cnt(b)
    end do
    off(nblocks) = total

    !$omp parallel do schedule(dynamic, 2) private(b, lo, hi, idx, j, c)
    do b = 0, int(nblocks - 1)
      lo = int(b, c_int64_t) * BS + 1
      hi = min(n, lo + BS - 1)
      c = off(b)
      do j = 1, int(hi - lo + 1)
        idx = lo + int(j, c_int64_t) - 1
        if (mask(idx) /= 0) then
          c = c + 1
          packed(c) = src(idx) * weight(idx)
        end if
      end do
    end do
    !$omp end parallel do
    deallocate(mask)
  end if

  out_count(1) = total
  deallocate(cnt, off)
end subroutine compact_threshold_pack_fp64
