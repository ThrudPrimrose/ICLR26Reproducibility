subroutine argmax_with_index_fp64(a, out_index, out_value, LEN_1D) bind(C)
  use iso_c_binding
  use omp_lib
  implicit none
  integer(c_int64_t), value, intent(in) :: LEN_1D
  real(c_double), intent(in) :: a(LEN_1D)
  integer(c_int64_t), intent(out) :: out_index
  real(c_double), intent(out) :: out_value
  real(c_double) :: x, xi
  integer(c_int64_t) :: i, idx, ii, t, nt, lo, hi
  real(c_double), allocatable :: local_val(:)
  integer(c_int64_t), allocatable :: local_idx(:)

  nt = omp_get_max_threads()
  allocate(local_val(nt))
  allocate(local_idx(nt))

  !$omp parallel private(t, lo, hi, xi, ii, i)
  t = omp_get_thread_num()
  lo = (LEN_1D * t) / nt + 1
  hi = (LEN_1D * (t + 1)) / nt
  if (lo <= hi) then
     xi = a(lo)
     ii = lo
     do i = lo + 1, hi
        if (a(i) > xi) then
           xi = a(i)
           ii = i
        end if
     end do
     local_val(t + 1) = xi
     local_idx(t + 1) = ii
  else
     local_val(t + 1) = -huge(x)
     local_idx(t + 1) = LEN_1D + 1
  end if
  !$omp end parallel

  x = a(1)
  idx = 1
  do t = 1, nt
     if (local_val(t) > x .or. (local_val(t) == x .and. local_idx(t) < idx)) then
        x = local_val(t)
        idx = local_idx(t)
     end if
  end do

  out_value = x
  out_index = idx
end subroutine argmax_with_index_fp64
