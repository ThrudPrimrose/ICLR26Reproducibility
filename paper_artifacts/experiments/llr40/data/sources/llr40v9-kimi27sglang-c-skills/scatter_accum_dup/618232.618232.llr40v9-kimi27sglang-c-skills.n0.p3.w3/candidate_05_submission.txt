#include <stdint.h>
#include <stdlib.h>
#include <stdbool.h>
#include <string.h>
#include <math.h>
#include <complex.h>
#include <omp.h>

void scatter_accum_dup_fp64(double *restrict bins, const int32_t *restrict ip, const double *restrict src, const int64_t LEN_1D) {
    if (LEN_1D < 4096) {
        for (int64_t i = 0; i < LEN_1D; ++i) {
            const int64_t idx = (int64_t)ip[i];
            bins[idx] = bins[idx] + src[i];
        }
        return;
    }
    #pragma omp parallel for schedule(guided, 3072)
    for (int64_t i = 0; i < LEN_1D; ++i) {
        const int64_t idx = (int64_t)ip[i];
        if (i + 12 < LEN_1D) { __builtin_prefetch(&bins[(int64_t)ip[i + 12]], 1, 3); }
        if (i + 24 < LEN_1D) { __builtin_prefetch(&bins[(int64_t)ip[i + 24]], 1, 3); }

        #pragma omp atomic update relaxed
        bins[idx] += src[i];
    }
}
