#include <stdint.h>
#include <stdlib.h>
#include <omp.h>
#ifdef __AVX512F__
#include <immintrin.h>
#endif

/* Stream compaction: packed[n++] = src[i]*weight[i] for src[i] > 0.
 * Two-phase with per-thread blocks so the scatter pass re-reads src (and
 * prefetched weight) from L2.
 */

#ifndef BLOCK_ELEMS
#define BLOCK_ELEMS 32768LL           /* 256 KB of src */
#endif
#ifndef CHUNK_MULT
#define CHUNK_MULT 4
#endif
#define SMALL_LIMIT (1LL << 15)       /* below: serial single pass */
#define USE_NT 1

#ifdef __AVX512F__

/* group loop body for the scatter phase; off is the packed cursor */
static inline void group_store(const double *s, const double *w, double *p,
                               int64_t i, int64_t off, int64_t *offp)
{
    __m512d v = _mm512_loadu_pd(s + i);
    const __m512d zero = _mm512_setzero_pd();
    __mmask8 m = _mm512_cmp_pd_mask(v, zero, _CMP_GT_OQ);
    if (m == 0) return;
    if (m == 0xff) {
        _mm512_storeu_pd(p + off, _mm512_mul_pd(v, _mm512_loadu_pd(w + i)));
        *offp = off + 8;
        return;
    }
    __m512d pr = _mm512_mul_pd(v, _mm512_loadu_pd(w + i));
    double buf[8];
    _mm512_storeu_pd(buf, pr);
    int c = 0;
    if (m & 1) { p[off + c] = buf[0]; c++; }
    if (m & 2) { p[off + c] = buf[1]; c++; }
    if (m & 4) { p[off + c] = buf[2]; c++; }
    if (m & 8) { p[off + c] = buf[3]; c++; }
    if (m & 16) { p[off + c] = buf[4]; c++; }
    if (m & 32) { p[off + c] = buf[5]; c++; }
    if (m & 64) { p[off + c] = buf[6]; c++; }
    if (m & 128) { p[off + c] = buf[7]; c++; }
    *offp = off + c;
}

static inline int64_t group_count(const double *s, const double *w, int64_t i)
{
    __builtin_prefetch(w + i + 128, 0, 1);
    __m512d v = _mm512_loadu_pd(s + i);
    const __m512d zero = _mm512_setzero_pd();
    return (int64_t)__builtin_popcountll((unsigned long long)_mm512_cmp_pd_mask(v, zero, _CMP_GT_OQ));
}

static int64_t scan_block(const double *s, const double *w, int64_t len)
{
    int64_t c = 0;
    int64_t i = 0;
    const int64_t nvec = len & ~7LL;
    for (; i < nvec; i += 8) c += group_count(s, w, i);
    for (; i < len; i++) c += (s[i] > 0.0);
    return c;
}

/* scatter pass */
static void pack_block(const double *s, const double *w, double *p,
                       int64_t len, int64_t off)
{
    int64_t i = 0;
    const int64_t nvec = len & ~7LL;
    const __m512d zero = _mm512_setzero_pd();
    const int nt_store = ((uintptr_t)p & 63) == 0;
    for (; i < nvec; i += 8) {
        __m512d v = _mm512_loadu_pd(s + i);
        __mmask8 m = _mm512_cmp_pd_mask(v, zero, _CMP_GT_OQ);
        if (m == 0) continue;
        if (m == 0xff) {
            __m512d pr = _mm512_mul_pd(v, _mm512_loadu_pd(w + i));
            if (nt_store && (off & 7) == 0)
                _mm512_stream_pd(p + off, pr);
            else
                _mm512_storeu_pd(p + off, pr);
            off += 8;
            continue;
        }
        double buf[8];
        _mm512_storeu_pd(buf, _mm512_mul_pd(v, _mm512_loadu_pd(w + i)));
        if (m & 1) p[off++] = buf[0];
        if (m & 2) p[off++] = buf[1];
        if (m & 4) p[off++] = buf[2];
        if (m & 8) p[off++] = buf[3];
        if (m & 16) p[off++] = buf[4];
        if (m & 32) p[off++] = buf[5];
        if (m & 64) p[off++] = buf[6];
        if (m & 128) p[off++] = buf[7];
    }
    for (; i < len; i++)
        if (s[i] > 0.0) p[off++] = s[i] * w[i];
    if (nt_store) _mm_sfence();
}

static void pack_serial(const double *s, const double *w, double *p,
                        int64_t N, int64_t *out_count)
{
    int64_t off = 0;
    int64_t i = 0;
    const int64_t nvec = N & ~7LL;
    for (; i < nvec; i += 8)
        group_store(s, w, p, i, off, &off);
    for (; i < N; i++)
        if (s[i] > 0.0) p[off++] = s[i] * w[i];
    *out_count = off;
}

void compact_threshold_pack_fp64(int64_t *restrict out_count,
                                 double *restrict packed,
                                 const double *restrict src,
                                 const double *restrict weight,
                                 const int64_t LEN_1D)
{
    const int64_t N = LEN_1D;
    if (N <= 0) { *out_count = 0; return; }

    if (N < SMALL_LIMIT) {
        pack_serial(src, weight, packed, N, out_count);
        return;
    }

    int nt = omp_get_max_threads();
    if (nt < 1) nt = 1;
    if (nt > 24) nt = 24;

    const int64_t nblocks = (N + BLOCK_ELEMS - 1) / BLOCK_ELEMS;
    if (nblocks <= 4 && nt > nblocks) nt = (int)nblocks;
    int64_t *bcnt = (int64_t *)malloc(nblocks * sizeof(int64_t));
    int64_t next_c = 0, next_p = 0;
    const int64_t BATCH = 2;   /* blocks grabbed per atomic ticket */

    /* level 1: count blocks (dynamic, self-balancing) */
    #pragma omp parallel num_threads(nt)
    {
        for (;;) {
            const int64_t b = __atomic_fetch_add(&next_c, BATCH, __ATOMIC_RELAXED);
            if (b >= nblocks) break;
            int64_t b1 = b + BATCH;
            if (b1 > nblocks) b1 = nblocks;
            for (int64_t k = b; k < b1; k++) {
                int64_t s0 = k * BLOCK_ELEMS;
                int64_t len = N - s0;
                if (len > BLOCK_ELEMS) len = BLOCK_ELEMS;
                bcnt[k] = scan_block(src + s0, weight + s0, len);
            }
        }
    }
    /* prefix over block counts -> base offsets */
    {
        int64_t gbase = 0;
        for (int64_t b = 0; b < nblocks; b++) {
            int64_t c = bcnt[b];
            bcnt[b] = gbase;
            gbase += c;
        }
        *out_count = gbase;
    }
    /* level 2: pack blocks (dynamic, self-balancing) */
    #pragma omp parallel num_threads(nt)
    {
        for (;;) {
            const int64_t b = __atomic_fetch_add(&next_p, BATCH, __ATOMIC_RELAXED);
            if (b >= nblocks) break;
            int64_t b1 = b + BATCH;
            if (b1 > nblocks) b1 = nblocks;
            for (int64_t k = b; k < b1; k++) {
                int64_t s0 = k * BLOCK_ELEMS;
                int64_t len = N - s0;
                if (len > BLOCK_ELEMS) len = BLOCK_ELEMS;
                pack_block(src + s0, weight + s0, packed, len, bcnt[k]);
            }
        }
    }
    free(bcnt);
}

#else
void compact_threshold_pack_fp64(int64_t *restrict out_count,
                                 double *restrict packed,
                                 const double *restrict src,
                                 const double *restrict weight,
                                 const int64_t LEN_1D)
{
    const int64_t N = LEN_1D;
    int64_t n = 0;
    for (int64_t i = 0; i < N; i++) n += (src[i] > 0.0);
    int64_t off = 0;
    for (int64_t i = 0; i < N; i++)
        if (src[i] > 0.0) packed[off++] = src[i] * weight[i];
    *out_count = n;
}
