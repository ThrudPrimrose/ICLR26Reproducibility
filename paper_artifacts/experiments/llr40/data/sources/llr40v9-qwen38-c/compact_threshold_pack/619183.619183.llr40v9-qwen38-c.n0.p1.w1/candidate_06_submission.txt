/* Stream compaction: packed[n++] = src[i]*weight[i] for src[i] > 0.
 * Two-phase with per-thread blocks so the scatter pass re-reads src (and
 * prefetched weight) from L2.
 */

#ifndef BLOCK_ELEMS
#define BLOCK_ELEMS 32768LL           /* 256 KB of src */
#endif
#ifndef CHUNK_MULT
#define CHUNK_MULT 4
#endif
#define SMALL_LIMIT (1LL << 15)       /* below: serial single pass */
#define USE_NT 1

#ifdef __AVX512F__

/* group loop body for the scatter phase; off is the packed cursor */
static inline void group_store(const double *s, const double *w, double *p,
                               int64_t i, int64_t off, int64_t *offp)
{
    __m512d v = _mm512_loadu_pd(s + i);
    const __m512d zero = _mm512_setzero_pd();
    __mmask8 m = _mm512_cmp_pd_mask(v, zero, _CMP_GT_OQ);
    if (m == 0) return;
    if (m == 0xff) {
        _mm512_storeu_pd(p + off, _mm512_mul_pd(v, _mm512_loadu_pd(w + i)));
        *offp = off + 8;
        return;
    }
    __m512d pr = _mm512_mul_pd(v, _mm512_loadu_pd(w + i));
    double buf[8];
    _mm512_storeu_pd(buf, pr);
    int c = 0;
    if (m & 1) { p[off + c] = buf[0]; c++; }
    if (m & 2) { p[off + c] = buf[1]; c++; }
    if (m & 4) { p[off + c] = buf[2]; c++; }
    if (m & 8) { p[off + c] = buf[3]; c++; }
    if (m & 16) { p[off + c] = buf[4]; c++; }
    if (m & 32) { p[off + c] = buf[5]; c++; }
    if (m & 64) { p[off + c] = buf[6]; c++; }
    if (m & 128) { p[off + c] = buf[7]; c++; }
    *offp = off + c;
}

/* level 1: count survivors AND emit a 1-bit-per-element survivor mask
 * (coalesced 64-byte stores) so the pack pass can skip dead groups'
 * src/weight loads entirely. */
static int64_t scan_block_mask(const double *s, uint8_t *mm, int64_t len)
{
    int64_t c = 0;
    int64_t i = 0;
    const int64_t nvec = len & ~7LL;
    const int64_t n512 = nvec & ~511LL;
    const __m512d zero = _mm512_setzero_pd();
    for (; i < n512; i += 512) {
        alignas(64) uint8_t mb[64];
        if (i + 512 < len) __builtin_prefetch(s + i + 512, 0, 1);
        for (int g = 0; g < 64; g += 4) {
            __m512d v0 = _mm512_loadu_pd(s + i + g * 8);
            __m512d v1 = _mm512_loadu_pd(s + i + g * 8 + 8);
            __m512d v2 = _mm512_loadu_pd(s + i + g * 8 + 16);
            __m512d v3 = _mm512_loadu_pd(s + i + g * 8 + 24);
            __mmask8 m0 = _mm512_cmp_pd_mask(v0, zero, _CMP_GT_OQ);
            __mmask8 m1 = _mm512_cmp_pd_mask(v1, zero, _CMP_GT_OQ);
            __mmask8 m2 = _mm512_cmp_pd_mask(v2, zero, _CMP_GT_OQ);
            __mmask8 m3 = _mm512_cmp_pd_mask(v3, zero, _CMP_GT_OQ);
            mb[g] = (uint8_t)m0;
            mb[g + 1] = (uint8_t)m1;
            mb[g + 2] = (uint8_t)m2;
            mb[g + 3] = (uint8_t)m3;
            c += (int64_t)(__builtin_popcountll((unsigned long long)m0)
                         + __builtin_popcountll((unsigned long long)m1)
                         + __builtin_popcountll((unsigned long long)m2)
                         + __builtin_popcountll((unsigned long long)m3));
        }
        memcpy(mm + (i >> 3), mb, 64);
    }
    if (nvec > n512) {
        alignas(64) uint8_t mb[64];
        memset(mb, 0, 64);
        for (; i < nvec; i += 8) {
            __m512d v = _mm512_loadu_pd(s + i);
            __mmask8 m = _mm512_cmp_pd_mask(v, zero, _CMP_GT_OQ);
            mb[(i >> 3) & 63] = (uint8_t)m;
            c += (int64_t)__builtin_popcountll((unsigned long long)m);
        }
        memcpy(mm + (n512 >> 3), mb, 64);
    }
    for (; i < len; i++)
        if (s[i] > 0.0) { c++; mm[i >> 3] |= (uint8_t)(1u << (i & 7)); }
    return c;
}

/* scatter pass, driven by the level-1 mask: dead groups cost one byte load */
static void pack_block(const double *s, const double *w, double *p,
                       const uint8_t *mm, int64_t len, int64_t off)
{
    int64_t i = 0;
    const int64_t nvec = len & ~7LL;
    const int nt_store = ((uintptr_t)p & 63) == 0;
    for (; i + 16 <= nvec; i += 16) {
        const int m0 = (int)mm[i >> 3];
        const int m1 = (int)mm[(i + 8) >> 3];
        if ((m0 | m1) == 0) continue;
        if (i + 256 < nvec && (mm[(i + 256) >> 3] | mm[(i + 264) >> 3])) {
            __builtin_prefetch(s + i + 256, 0, 1);
            __builtin_prefetch(w + i + 256, 0, 1);
        }
        __m512d v0 = _mm512_loadu_pd(s + i);
        __m512d w0 = _mm512_loadu_pd(w + i);
        __m512d v1 = _mm512_loadu_pd(s + i + 8);
        __m512d w1 = _mm512_loadu_pd(w + i + 8);
        if (m0 == 0xff) {
            __m512d pr = _mm512_mul_pd(v0, w0);
            if (nt_store && (off & 7) == 0)
                _mm512_stream_pd(p + off, pr);
            else
                _mm512_storeu_pd(p + off, pr);
            off += 8;
        } else if (m0) {
            double buf[8];
            _mm512_storeu_pd(buf, _mm512_mul_pd(v0, w0));
            if (m0 & 1) p[off++] = buf[0];
            if (m0 & 2) p[off++] = buf[1];
            if (m0 & 4) p[off++] = buf[2];
            if (m0 & 8) p[off++] = buf[3];
            if (m0 & 16) p[off++] = buf[4];
            if (m0 & 32) p[off++] = buf[5];
            if (m0 & 64) p[off++] = buf[6];
            if (m0 & 128) p[off++] = buf[7];
        }
        if (m1 == 0xff) {
            __m512d pr = _mm512_mul_pd(v1, w1);
            if (nt_store && (off & 7) == 0)
                _mm512_stream_pd(p + off, pr);
            else
                _mm512_storeu_pd(p + off, pr);
            off += 8;
        } else if (m1) {
            double buf[8];
            _mm512_storeu_pd(buf, _mm512_mul_pd(v1, w1));
            if (m1 & 1) p[off++] = buf[0];
            if (m1 & 2) p[off++] = buf[1];
            if (m1 & 4) p[off++] = buf[2];
            if (m1 & 8) p[off++] = buf[3];
            if (m1 & 16) p[off++] = buf[4];
            if (m1 & 32) p[off++] = buf[5];
            if (m1 & 64) p[off++] = buf[6];
            if (m1 & 128) p[off++] = buf[7];
        }
    }
    for (; i < nvec; i += 8) {
        const int m = (int)mm[i >> 3];
        if (m == 0) continue;
        __m512d v = _mm512_loadu_pd(s + i);
        __m512d wv = _mm512_loadu_pd(w + i);
        if (m == 0xff) {
            __m512d pr = _mm512_mul_pd(v, wv);
            if (nt_store && (off & 7) == 0)
                _mm512_stream_pd(p + off, pr);
            else
                _mm512_storeu_pd(p + off, pr);
            off += 8;
            continue;
        }
        double buf[8];
        _mm512_storeu_pd(buf, _mm512_mul_pd(v, wv));
        if (m & 1) p[off++] = buf[0];
        if (m & 2) p[off++] = buf[1];
        if (m & 4) p[off++] = buf[2];
        if (m & 8) p[off++] = buf[3];
        if (m & 16) p[off++] = buf[4];
        if (m & 32) p[off++] = buf[5];
        if (m & 64) p[off++] = buf[6];
        if (m & 128) p[off++] = buf[7];
    }
    for (; i < len; i++)
        if (mm[i >> 3] & (uint8_t)(1u << (i & 7))) p[off++] = s[i] * w[i];
    if (nt_store) _mm_sfence();
}

static void pack_serial(const double *s, const double *w, double *p,
                        int64_t N, int64_t *out_count)
{
    int64_t off = 0;
    int64_t i = 0;
    const int64_t nvec = N & ~7LL;
    for (; i < nvec; i += 8)
        group_store(s, w, p, i, off, &off);
    for (; i < N; i++)
        if (s[i] > 0.0) p[off++] = s[i] * w[i];
    *out_count = off;
}

void compact_threshold_pack_fp64(int64_t *restrict out_count,
                                 double *restrict packed,
                                 const double *restrict src,
                                 const double *restrict weight,
                                 const int64_t LEN_1D)
{
    const int64_t N = LEN_1D;
    if (N <= 0) { *out_count = 0; return; }

    if (N < SMALL_LIMIT) {
        pack_serial(src, weight, packed, N, out_count);
        return;
    }

    int nt = omp_get_max_threads();
    if (nt < 1) nt = 1;
    if (nt > 32) nt = 32;

    const int64_t nblocks = (N + BLOCK_ELEMS - 1) / BLOCK_ELEMS;
    if (nblocks <= 4 && nt > nblocks) nt = (int)nblocks;
    int64_t *bcnt = (int64_t *)malloc(nblocks * sizeof(int64_t));
    /* grow-only static mask pool: avoids re-faulting ~20MB of fresh pages
     * on every call (each call would otherwise pay thousands of faults) */
    static uint8_t *marr = NULL;
    static size_t mcap = 0;
    {
        size_t msz = (size_t)((N + 7) / 8) + 64;
        if (msz > mcap) {
            if (marr) free(marr);
            if (posix_memalign((void **)&marr, 64, msz) != 0) { free(bcnt); return; }
            memset(marr, 0, msz);   /* fault the pages in once */
            mcap = msz;
        }
    }
    int64_t next_c = 0, next_p = 0;
    const int64_t BATCH = 1;   /* blocks grabbed per atomic ticket */

    #pragma omp parallel num_threads(nt)
    {
        int tid = omp_get_thread_num();
        /* level 1: count blocks (dynamic, self-balancing) */
        for (;;) {
            const int64_t b = __atomic_fetch_add(&next_c, BATCH, __ATOMIC_RELAXED);
            if (b >= nblocks) break;
            int64_t b1 = b + BATCH;
            if (b1 > nblocks) b1 = nblocks;
            for (int64_t k = b; k < b1; k++) {
                int64_t s0 = k * BLOCK_ELEMS;
                int64_t len = N - s0;
                if (len > BLOCK_ELEMS) len = BLOCK_ELEMS;
                bcnt[k] = scan_block_mask(src + s0, marr + (s0 >> 3), len);
            }
        }
        #pragma omp barrier
        if (tid == 0) {
            /* prefix over block counts -> base offsets */
            int64_t gbase = 0;
            for (int64_t b = 0; b < nblocks; b++) {
                int64_t c = bcnt[b];
                bcnt[b] = gbase;
                gbase += c;
            }
            *out_count = gbase;
        }
        #pragma omp barrier
        /* level 2: pack blocks (dynamic, self-balancing) */
        for (;;) {
            const int64_t b = __atomic_fetch_add(&next_p, BATCH, __ATOMIC_RELAXED);
            if (b >= nblocks) break;
            int64_t b1 = b + BATCH;
            if (b1 > nblocks) b1 = nblocks;
            for (int64_t k = b; k < b1; k++) {
                int64_t s0 = k * BLOCK_ELEMS;
                int64_t len = N - s0;
                if (len > BLOCK_ELEMS) len = BLOCK_ELEMS;
                pack_block(src + s0, weight + s0, packed, marr + (s0 >> 3), len, bcnt[k]);
            }
        }
    }
    free(bcnt);
}

#else
void compact_threshold_pack_fp64(int64_t *restrict out_count,
                                 double *restrict packed,
                                 const double *restrict src,
                                 const double *restrict weight,
                                 const int64_t LEN_1D)
{
    const int64_t N = LEN_1D;
    int64_t n = 0;
    for (int64_t i = 0; i < N; i++) n += (src[i] > 0.0);
    int64_t off = 0;
    for (int64_t i = 0; i < N; i++)
        if (src[i] > 0.0) packed[off++] = src[i] * weight[i];
    *out_count = n;
}
