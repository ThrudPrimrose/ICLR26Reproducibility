subroutine tsvc_2_s235_fp64(a, aa, b, bb, c, len2d) bind(C)
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: len2d
  real(c_double), intent(inout) :: a(len2d)
  real(c_double), intent(inout) :: aa(len2d, len2d)
  real(c_double), intent(in) :: b(len2d)
  real(c_double), intent(in) :: bb(len2d, len2d)
  real(c_double), intent(in) :: c(len2d)

  integer(c_int64_t) :: i, j, ib, i0, i1, nblocks
  real(c_double), allocatable :: running(:)
  integer, parameter :: K = 64

  allocate(running(len2d))
  nblocks = (len2d + K - 1) / K

  !$omp parallel do schedule(static)
  do ib = 1, nblocks
    i0 = (ib - 1) * K + 1
    i1 = min(ib * K, len2d)
    do i = i0, i1
      a(i) = a(i) + b(i)*c(i)
      running(i) = aa(i, 1)
    end do
    do j = 2, len2d
      do i = i0, i1
        running(i) = running(i) + bb(i, j)*a(i)
        aa(i, j) = running(i)
      end do
    end do
  end do
  !$omp end parallel do

  deallocate(running)
end subroutine tsvc_2_s235_fp64
