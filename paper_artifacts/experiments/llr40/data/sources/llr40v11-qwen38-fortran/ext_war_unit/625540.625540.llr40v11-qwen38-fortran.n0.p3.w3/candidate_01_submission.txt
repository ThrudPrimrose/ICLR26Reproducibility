module war_mod
  use iso_c_binding
  implicit none
  real(c_double), allocatable :: tmp(:)
  integer(c_int64_t) :: tmp_size = 0_8
end module war_mod

subroutine ext_war_unit_fp64(a, b, len1d) bind(C, name='ext_war_unit_fp64')
  use war_mod
  use iso_c_binding
  implicit none
  real(c_double), intent(inout) :: a(*)
  real(c_double), intent(in)    :: b(*)
  integer(c_int64_t), intent(in), value :: len1d

  integer(c_int64_t) :: n, i

  n = len1d
  if (n <= 1_8) return
  if (n > tmp_size) then
    if (allocated(tmp)) deallocate(tmp)
    allocate(tmp(n))
    tmp_size = n
  end if
  !$omp parallel
  !$omp do schedule(static)
  do i = 1_8, n
    tmp(i) = a(i)
  end do
  !$omp end do
  !$omp do schedule(static)
  do i = 1_8, n - 1_8
    a(i) = tmp(i + 1_8) + b(i)
  end do
  !$omp end do
  !$omp end parallel
end subroutine
