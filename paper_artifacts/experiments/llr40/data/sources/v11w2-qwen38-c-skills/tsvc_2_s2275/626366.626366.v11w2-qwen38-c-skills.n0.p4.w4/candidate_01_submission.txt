#include <stdint.h>
#include <omp.h>

void tsvc_2_s2275_fp64(double *restrict a, double *restrict aa, const double *restrict b, const double *restrict bb,
                       const double *restrict c, const double *restrict cc, const double *restrict d,
                       const int64_t LEN_2D) {
  /* Reference semantics:
     for i in 0..N-1:
       for j in 0..N-1: aa[j*N+i] += bb[j*N+i]*cc[j*N+i]   (idx = j*N+i)
       a[i] = b[i] + c[i]*d[i]
     The (i,j) nest sweeps idx = j*N+i over ALL of [0, N*N) exactly once,
     so the matrix update is a plain unit-stride elementwise op, fully
     independent of the small vector update. */
  const int64_t M = LEN_2D * LEN_2D;
  #pragma omp parallel for simd schedule(static)
  for (int64_t k = 0; k < M; ++k) {
    aa[k] = aa[k] + bb[k] * cc[k];
  }
  #pragma omp parallel for simd schedule(static)
  for (int64_t i = 0; i < LEN_2D; ++i) {
    a[i] = b[i] + c[i] * d[i];
  }
}
