#include <stdint.h>
#include <omp.h>

void tsvc_2_s231_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
    // Offload to GPU/device. Parallelize across the independent columns (i).
    // Map the full 2D arrays to/from the device.
    #pragma omp target teams distribute parallel for schedule(static) \
        map(tofrom: aa[0:LEN_2D*LEN_2D]) \
        map(to: bb[0:LEN_2D*LEN_2D])
    for (int64_t i = 0; i < LEN_2D; ++i) {
        // Sequential scan down the column j for this column i.
        for (int64_t j = 1; j < LEN_2D; ++j) {
            aa[j * LEN_2D + i] = aa[(j - 1) * LEN_2D + i] + bb[j * LEN_2D + i];
        }
    }
}
