#include <stdint.h>
#include <stdlib.h>
#include <omp.h>

void tsvc_2_s323_fp64(double *restrict a, double *restrict b,
                       const double *restrict c, const double *restrict d,
                       const double *restrict e, const int64_t LEN_1D) {
#if defined(__clang__) && defined(_OPENMP)
    // GPU version: compute a and b using a parallel prefix sum
    //   inc[i] = c[i] * (d[i] + e[i])
    //   b[i]   = b[0] + sum_{k=1..i} inc[k]
    //   a[i]   = b[i] - c[i] * e[i]
    // Allocate temporary storage for the increments.
    double *inc = (double *)malloc((size_t)LEN_1D * sizeof(double));
    if (!inc) {
        // Allocation failure – fall back to serial.
        for (int64_t i = 1; i < LEN_1D; ++i) {
            a[i] = b[i - 1] + c[i] * d[i];
            b[i] = a[i] + c[i] * e[i];
        }
        return;
    }
    // Choose a safe upper bound for the number of threads that may appear on the device.
    const int MAX_THREADS = 1024;
    double *block_sum = (double *)malloc(MAX_THREADS * sizeof(double));
    if (!block_sum) {
        free(inc);
        for (int64_t i = 1; i < LEN_1D; ++i) {
            a[i] = b[i - 1] + c[i] * d[i];
            b[i] = a[i] + c[i] * e[i];
        }
        return;
    }

    #pragma omp target map(to: c[0:LEN_1D], d[0:LEN_1D], e[0:LEN_1D]) \
                     map(tofrom: a[0:LEN_1D], b[0:LEN_1D]) \
                     map(tofrom: inc[0:LEN_1D], block_sum[0:MAX_THREADS])
    {
        // Step 1: compute the per‑iteration increments.
        #pragma omp parallel for
        for (int64_t i = 1; i < LEN_1D; ++i) {
            inc[i] = c[i] * (d[i] + e[i]);
        }
        inc[0] = 0.0;

        // Step 2: parallel prefix sum of inc (Hillis‑Steele style with per‑thread chunks).
        #pragma omp parallel
        {
            int tid = omp_get_thread_num();
            int nt = omp_get_num_threads();
            int64_t start = (LEN_1D * tid) / nt;
            int64_t end   = (LEN_1D * (tid + 1)) / nt;
            // Compute local prefix and total for this chunk.
            double cur = 0.0;
            for (int64_t i = start; i < end; ++i) {
                cur += inc[i];
                inc[i] = cur; // local prefix (still missing offset of previous chunks)
            }
            block_sum[tid] = cur; // store chunk total.
            #pragma omp barrier
            // Compute exclusive offsets for each chunk (single thread).
            #pragma omp single
            {
                double offset = 0.0;
                for (int i = 0; i < nt; ++i) {
                    double sum_i = block_sum[i];
                    block_sum[i] = offset;    // overwrite with exclusive offset
                    offset += sum_i;
                }
            }
            #pragma omp barrier
            // Add the offset to each element of the chunk to obtain the global prefix.
            double add = block_sum[tid];
            for (int64_t i = start; i < end; ++i) {
                inc[i] += add;
            }
        }

        // Step 3: compute final a and b using the prefix sums.
        double b0 = b[0];
        #pragma omp parallel for
        for (int64_t i = 1; i < LEN_1D; ++i) {
            double bi = b0 + inc[i];
            b[i] = bi;
            a[i] = bi - c[i] * e[i];
        }
    }

    free(inc);
    free(block_sum);

    // Verify execution on the device; if not, fall back to serial.
    int on_device = 0;
    #pragma omp target map(from: on_device)
    on_device = !omp_is_initial_device();
    if (!on_device) {
        for (int64_t i = 1; i < LEN_1D; ++i) {
            a[i] = b[i - 1] + c[i] * d[i];
            b[i] = a[i] + c[i] * e[i];
        }
    }
#else
    // Serial fallback (compilers without the GPU implementation).
    for (int64_t i = 1; i < LEN_1D; ++i) {
        a[i] = b[i - 1] + c[i] * d[i];
        b[i] = a[i] + c[i] * e[i];
    }
#endif
}
