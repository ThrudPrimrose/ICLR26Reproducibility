#include <hip/hip_runtime.h>
#include <cstdint>

extern "C" __global__ void tsvc_2_s1244_fp64_compute_a(double * __restrict__ a,
                                                     const double * __restrict__ b,
                                                     const double * __restrict__ c,
                                                     int64_t n) {
    int64_t i = static_cast<int64_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    if (i < n) {
        a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
    }
}

extern "C" __global__ void tsvc_2_s1244_fp64_compute_d(const double * __restrict__ a_new,
                                                     const double * __restrict__ a_old,
                                                     double * __restrict__ d,
                                                     int64_t n) {
    int64_t i = static_cast<int64_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    if (i < n) {
        d[i] = a_new[i] + a_old[i + 1];
    }
}
