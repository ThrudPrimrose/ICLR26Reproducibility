/* Device kernel for TSVC s3110 HIP implementation */
#include <hip/hip_runtime.h>
#include <cstddef>
#include <cstdint>
#include <limits>

constexpr int BLOCK_SIZE = 256;

__global__ void tsvc_2_s3110_kernel(const double* __restrict__ aa,
                                    const int64_t LEN_2D,
                                    double* __restrict__ partial_max,
                                    int64_t* __restrict__ partial_idx) {
    const int64_t total = LEN_2D * LEN_2D;
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int stride = gridDim.x * blockDim.x;
    double local_max = -std::numeric_limits<double>::infinity();
    int64_t local_idx = -1;
    for (int64_t idx = tid; idx < total; idx += stride) {
        double v = aa[idx];
        if (v > local_max) {
            local_max = v;
            local_idx = idx;
        }
    }
    // Block reduction.
    __shared__ double smax[BLOCK_SIZE];
    __shared__ int64_t sidx[BLOCK_SIZE];
    int lane = threadIdx.x;
    smax[lane] = local_max;
    sidx[lane] = local_idx;
    __syncthreads();
    for (int offset = BLOCK_SIZE / 2; offset > 0; offset >>= 1) {
        if (lane < offset) {
            if (smax[lane + offset] > smax[lane]) {
                smax[lane] = smax[lane + offset];
                sidx[lane] = sidx[lane + offset];
            }
        }
        __syncthreads();
    }
    if (lane == 0) {
        partial_max[blockIdx.x] = smax[0];
        partial_idx[blockIdx.x] = sidx[0];
    }
}
