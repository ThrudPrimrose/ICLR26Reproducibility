#include <stdint.h>
#include <omp.h>

void tsvc_2_s2275_fp64(double *restrict a, double *restrict aa, const double *restrict b, const double *restrict bb,
                       const double *restrict c, const double *restrict cc, const double *restrict d,
                       const int64_t LEN_2D) {
  const int64_t N2 = LEN_2D * LEN_2D;
  #pragma omp target data map(from: a[0:LEN_2D]) \
                          map(to: b[0:LEN_2D], bb[0:N2], c[0:LEN_2D], cc[0:N2], d[0:LEN_2D]) \
                          map(tofrom: aa[0:N2])
  {
    #pragma omp target
    {
      #pragma omp teams distribute parallel for simd
      for (int64_t idx = 0; idx < N2; ++idx)
        aa[idx] = aa[idx] + bb[idx] * cc[idx];
    }
    #pragma omp target
    {
      #pragma omp teams distribute parallel for simd
      for (int64_t i = 0; i < LEN_2D; ++i)
        a[i] = b[i] + c[i] * d[i];
    }
  }
}
