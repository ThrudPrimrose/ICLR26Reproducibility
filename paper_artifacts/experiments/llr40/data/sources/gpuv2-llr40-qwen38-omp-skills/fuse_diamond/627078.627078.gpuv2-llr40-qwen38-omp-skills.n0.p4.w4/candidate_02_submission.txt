#include <stdint.h>
#include <omp.h>
#include <immintrin.h>

void fuse_diamond_fp64(const double *restrict a, double *restrict out, const int64_t LEN_1D) {
  double tag = 0.0;
  #pragma omp target map(from: tag)
  tag = 1.0;

  #pragma omp parallel for schedule(static)
  for (int64_t base = 0; base < LEN_1D; base += 8) {
    const int64_t n = LEN_1D - base;
    const double *pa = a + base;
    double *po = out + base;
    if (n >= 8 && ((uintptr_t)po & 63) == 0) {
      __m512d x = _mm512_loadu_pd(pa);
      __m512d t = _mm512_mul_pd(x, x);
      __m512d o = _mm512_mul_pd(_mm512_add_pd(t, _mm512_set1_pd(1.0)),
                                _mm512_sub_pd(t, _mm512_set1_pd(1.0)));
      _mm512_stream_pd(po, o);
    } else {
      for (int64_t i = 0; i < n; ++i) {
        double t = pa[i] * pa[i];
        po[i] = (t + 1.0) * (t - 1.0);
      }
    }
  }
}
