#include <stdint.h>
#include <stdlib.h>

void ext_war_unit_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D) {
  const int64_t n = LEN_1D;
  if (n < 2) return;
  double *tmp = (double *)malloc(n * sizeof(double));
  #pragma omp target data map(tofrom: a[0:n]) map(to: b[0:n]) map(alloc: tmp[0:n])
  {
    #pragma omp target
    #pragma omp teams distribute parallel for simd
    for (int64_t i = 0; i < n; ++i) tmp[i] = a[i];

    #pragma omp target
    #pragma omp teams distribute parallel for simd
    for (int64_t i = 0; i < n - 1; ++i) a[i] = tmp[i + 1] + b[i];
  }
  free(tmp);
}
