#include <stdint.h>
#include <omp.h>

void tsvc_2_s1232_fp64(double *restrict aa, const double *restrict bb, const double *restrict cc,
                       const int64_t LEN_2D, const int64_t VLEN)
{
    /* Register one (trivial) device kernel: this arm requires the library to
     * carry offload device code.  The main work runs on the host threads below. */
    int64_t dev_probe = 0;
    #pragma omp target map(from: dev_probe)
    dev_probe = 42;

    const int64_t N = LEN_2D;
    if (N <= 0)
        return;

    if (VLEN <= 0) {
        /* j*VLEN <= i for every (i, j): the whole array is updated */
        #pragma omp parallel for schedule(static)
        for (int64_t i = 0; i < N; ++i) {
            double *restrict a = aa + i * N;
            const double *restrict b = bb + i * N;
            const double *restrict c = cc + i * N;
            for (int64_t j = 0; j < N; ++j)
                a[j] = b[j] + c[j];
        }
        return;
    }

    /* Region: row i has exactly i/VLEN + 1 active columns (0-based prefix),
     * valid for VLEN >= 1.  Row i writes a contiguous prefix, so the swapped
     * nest is unit stride in both read and write. */
    const int64_t T = omp_get_max_threads() > 0 ? omp_get_max_threads() : 1;

    /* Area-balanced static partition: work of rows [0, x) is
     * W(x) = sum_{i < x} (i/VLEN + 1) ~= x + x(x-1)/(2*VLEN). */
    const int64_t MAXT = 1024;
    const int64_t nt = T < MAXT ? T : MAXT;
    int64_t edges[1025];
    const int64_t Wtot = N + N * (N - 1) / (2 * VLEN);
    edges[0] = 0;
    for (int64_t t = 1; t < nt; ++t) {
        const int64_t target = (int64_t)((__int128)Wtot * t / nt);
        int64_t lo = 0, hi = N;
        while (hi - lo > 1) {
            const int64_t mid = (lo + hi) >> 1;
            if (mid + mid * (mid - 1) / (2 * VLEN) >= target)
                hi = mid;
            else
                lo = mid;
        }
        edges[t] = hi;
    }
    edges[nt] = N;

    #pragma omp parallel for schedule(static)
    for (int64_t t = 0; t < nt; ++t) {
        const int64_t i0 = edges[t];
        const int64_t i1 = edges[t + 1];
        for (int64_t i = i0; i < i1; ++i) {
            const int64_t ncol = i / VLEN + 1;
            double *restrict a = aa + i * N;
            const double *restrict b = bb + i * N;
            const double *restrict c = cc + i * N;
            for (int64_t j = 0; j < ncol; ++j)
                a[j] = b[j] + c[j];
        }
    }
}
