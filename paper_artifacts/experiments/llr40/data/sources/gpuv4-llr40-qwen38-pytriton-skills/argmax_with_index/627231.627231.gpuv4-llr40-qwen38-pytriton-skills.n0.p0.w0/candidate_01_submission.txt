import numpy as np
import os
from concurrent.futures import ThreadPoolExecutor

_EXEC = ThreadPoolExecutor(max_workers=32)
_MIN_AFF = max(1, len(os.sched_getaffinity(0)) if hasattr(os, "sched_getaffinity") else os.cpu_count() or 1)
_K = min(32, max(1, _MIN_AFF))

def argmax_with_index(a, out_value, out_index, LEN_1D):
    n = a.shape[0]
    if LEN_1D < n:
        n = LEN_1D
    if n <= 1:
        out_value[0] = a[0]
        out_index[0] = 0
        return None
    K = _K if _K < n else 1
    if K == 1:
        idx = np.argmax(a[:n])
    else:
        base = n // K
        lo = 0
        futs = []
        for k in range(K):
            hi = n if k == K - 1 else lo + base
            futs.append((lo, _EXEC.submit(np.argmax, a[lo:hi])))
            lo = hi
        best = None
        best_idx = -1
        for lo, f in futs:
            i = int(lo + f.result())
            v = a[i]
            if best is None or v > best:
                best = v
                best_idx = i
        idx = best_idx
    if np.isnan(best if K != 1 else a[idx]):
        f0 = a[0]
        if np.isnan(f0):
            out_value[0] = f0
            out_index[0] = 0
            return None
        b = np.where(np.isnan(a[:n]), -np.inf, a[:n])
        idx = int(np.argmax(b))
        out_value[0] = b[idx]
    else:
        out_value[0] = a[idx]
    out_index[0] = idx
    return None
