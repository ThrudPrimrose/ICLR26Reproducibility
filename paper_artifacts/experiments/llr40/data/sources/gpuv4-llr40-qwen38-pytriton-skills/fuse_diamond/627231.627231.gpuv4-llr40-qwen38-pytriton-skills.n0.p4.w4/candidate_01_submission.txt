import numpy as np
from numba import njit, prange

@njit(parallel=True, nogil=True)
def _fd(out, a, n):
    for i in prange(n):
        t = a[i] * a[i]
        out[i] = (t + 1.0) * (t - 1.0)

def fuse_diamond(out, a, LEN_1D):
    _fd(out, a, LEN_1D)
