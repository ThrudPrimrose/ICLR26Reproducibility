import numpy as np
import numba as nb


@nb.njit(parallel=True)
def _fused(a, b, src, cond, LEN_2D, K):
    for i in nb.prange(LEN_2D):
        if cond[i] > 0.0:
            for j in range(LEN_2D):
                a[i, j] = src[i, j] * 2.0
        if K > 0:
            for j in range(LEN_2D):
                b[i, j] = src[i, j] + 1.0


def fuse_move_ifs(a, b, src, cond, LEN_2D, K):
    _fused(a, b, src, cond, LEN_2D, K)
    return None


def _warm():
    m = 16
    aa = np.zeros((m, m)); bb = np.zeros((m, m)); ss = np.ones((m, m))
    cc = np.array([1.0, -1.0] * (m // 2))
    _fused(aa, bb, ss, cc, m, 1)
    _fused(aa, bb, ss, cc, m, -1)


_warm()
