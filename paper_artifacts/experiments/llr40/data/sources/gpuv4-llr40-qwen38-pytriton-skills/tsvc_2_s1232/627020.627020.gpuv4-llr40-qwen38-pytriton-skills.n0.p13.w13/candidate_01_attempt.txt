import numpy as np
import numba
from numba import prange


@numba.njit(parallel=True, fastmath=True)
def _k(aa, bb, cc, N, VLEN):
    # Region updated: aa[i, j] = bb[i, j] + cc[i, j] for i >= j*VLEN.
    # Row-major arrays: for a fixed row i, the updated columns j are
    # 0 .. i//VLEN  (a contiguous prefix).  Parallelize over rows so the
    # inner loop streams contiguous memory.
    for i in prange(N):
        jend = i // VLEN + 1
        for j in range(jend):
            aa[i, j] = bb[i, j] + cc[i, j]


def s1232(aa, bb, cc, LEN_2D, VLEN):
    _k(aa, bb, cc, LEN_2D, VLEN)
    return None


# --- Warm up at import time (runs once, before the clock starts) --------
# This forces the numba JIT compilation AND the parallel thread-pool init
# to happen now, so no timed call ever pays for it.  The compiled object is
# keyed only on argument types (float64 C-contiguous 2-D, int64), not on
# shape, so a small dummy call produces the exact object reused at runtime.
def _warm():
    n = 128
    v = 8
    aa = np.zeros((n, n))
    bb = np.ones((n, n))
    cc = np.ones((n, n))
    _k(aa, bb, cc, n, v)


_warm()
