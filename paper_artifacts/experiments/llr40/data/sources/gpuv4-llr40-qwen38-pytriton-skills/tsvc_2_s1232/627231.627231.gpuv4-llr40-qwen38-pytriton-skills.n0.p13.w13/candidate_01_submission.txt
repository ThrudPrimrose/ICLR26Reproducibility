"""TSVC tsvc_2 s1232 -- in-place: aa[i, j] = bb[i, j] + cc[i, j] for j*VLEN <= i < LEN_2D.

Row-oriented: row i needs only the contiguous prefix of columns 0..i//VLEN, so the
staircase is processed in VLEN-row blocks of equal width (one cache line per row,
not one per element). The hot loop lives in a prebuilt C extension loaded at import
time (outside the timed region); a numba fallback compiles at import if the .so
cannot be loaded on the grading node.
"""
import os
import importlib.util

_SO_NAME = "libtsvc2_s1232.so"


def _load_ext():
    cands = []
    try:
        cands.append(os.path.join(os.path.dirname(os.path.abspath(__file__)), _SO_NAME))
    except Exception:
        pass
    for sd in (os.environ.get("HPCAGENT_BENCH_SHARED_DIR"), "/shared"):
        if sd:
            cands.append(os.path.join(sd, "agent-13", _SO_NAME))
    cands.append(_SO_NAME)
    for path in cands:
        try:
            spec = importlib.util.spec_from_file_location("_tsvc2s1232", path)
            if spec is None or spec.loader is None:
                continue
            mod = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(mod)
            fn = mod.s1232
            # cheap sanity check on a 4x4 case
            import numpy as _np
            _a = _np.zeros((4, 4))
            fn(_a, _a, _a, 4, 2)
            return fn
        except Exception:
            continue
    return None


def _numba_impl():
    import numpy as np
    import numba

    @numba.njit(fastmath=True)
    def _core(aa, bb, cc, L, V):
        if V <= 0:
            for i in range(L):
                for j in range(L):
                    aa[i, j] = bb[i, j] + cc[i, j]
            return
        Q = (L - 1) // V
        for q in range(Q + 1):
            w = q + 1
            base = q * V
            rows = V
            if base + rows > L:
                rows = L - base
            for r in range(rows):
                i = base + r
                arow = aa[i]
                brow = bb[i]
                crow = cc[i]
                for j in range(w):
                    arow[j] = brow[j] + crow[j]

    _z = np.zeros((8, 8))
    _core(_z, _z, _z, 8, 8)

    def s1232(aa, bb, cc, LEN_2D, VLEN):
        _core(aa, bb, cc, int(LEN_2D), int(VLEN))
        return None

    return s1232


def _numpy_impl():
    import numpy as np

    def s1232(aa, bb, cc, LEN_2D, VLEN):
        if VLEN <= 0:
            np.add(bb, cc, out=aa)
            return None
        for j in range((LEN_2D - 1) // VLEN + 1):
            s = j * VLEN
            np.add(bb[s:, j], cc[s:, j], out=aa[s:, j])
        return None

    return s1232


s1232 = _load_ext()
if s1232 is None:
    try:
        s1232 = _numba_impl()
    except Exception:
        s1232 = _numpy_impl()
