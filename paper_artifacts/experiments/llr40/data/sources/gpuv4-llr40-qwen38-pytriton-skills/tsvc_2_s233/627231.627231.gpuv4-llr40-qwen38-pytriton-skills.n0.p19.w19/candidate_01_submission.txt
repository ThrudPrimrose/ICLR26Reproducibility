import numpy as np


def s233(aa, bb, cc, LEN_2D):
    # Column-wise scan of aa:  aa[j,i] = aa[7,i] + sum_{k=8..j} cc[k,i]
    # Row-wise    scan of bb:  bb[j,i] = bb[j,7] + sum_{k=8..i} cc[j,k]
    # Both depend only on cc and a single base row/col of the original array,
    # so each is one cumulative-sum pass (no cross-row/col serial dependence).
    aa[8:, 8:] = np.cumsum(cc[8:, 8:], axis=0) + aa[7, 8:][None, :]
    bb[8:, 8:] = np.cumsum(cc[8:, 8:], axis=1) + bb[8:, 7][:, None]
    return None
