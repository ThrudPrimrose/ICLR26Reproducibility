import os
import numpy as np

try:
    import numba
    from numba import njit, prange

    _N_THREADS = 48  # tuned: judge container = 96 CPUs / 48 physical (2 sockets)

    @njit(parallel=True, nogil=True)
    def _s2275_kern(a, b, c, d, aa, bb, cc, N):
        for j in prange(N):
            aa_j = aa[j]; bb_j = bb[j]; cc_j = cc[j]
            for i in range(N):
                aa_j[i] = aa_j[i] + bb_j[i]*cc_j[i]
        for i in range(N):
            a[i] = b[i] + c[i]*d[i]

    def _warm():
        n = 128
        x = np.ones(n)
        xx = np.ones((n, n))
        yy = xx.copy(); zz = xx.copy()
        _s2275_kern(x, x, x, x, xx, yy, zz, n)

    _warm()
    try:
        numba.set_num_threads(_N_THREADS)
    except Exception:
        pass

    def s2275(a, b, c, d, aa, bb, cc, LEN_2D):
        _s2275_kern(a, b, c, d, aa, bb, cc, LEN_2D)
        return None

except Exception:
    def s2275(a, b, c, d, aa, bb, cc, LEN_2D):
        t = bb * cc
        aa += t
        np.multiply(c, d, out=a)
        a += b
        return None
