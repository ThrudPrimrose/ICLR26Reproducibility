# C/OpenMP parallel chunked min via a prebuilt .so; warmed at import time.
import ctypes
import os

try:
    _T = 24
except Exception:
    _T = 24
os.environ["OMP_NUM_THREADS"] = str(_T)
os.environ["OMP_DYNAMIC"] = "false"

_LIB = "/shared/agent-29/libmin316.so"
_lib = ctypes.CDLL(_LIB)
for _n in ("min_omp", "min_vec"):
    getattr(_lib, _n).argtypes = [ctypes.c_void_p, ctypes.c_int64, ctypes.c_void_p]


def s316(a, result, LEN_1D):
    n = a.shape[0]
    if n < 4096:
        _lib.min_vec(a.ctypes.data, n, result.ctypes.data)
    else:
        _lib.min_omp(a.ctypes.data, n, result.ctypes.data)
    return None
