import numpy as np

def s311(a, sum_out, LEN_1D):
    """Sum the elements of a into sum_out[0]. In-place version.
    Parameters
    ----------
    a : np.ndarray
        Input 1D array of length LEN_1D.
    sum_out : np.ndarray
        Output 1D array, at least length 1. The result is written to sum_out[0].
    LEN_1D : int
        Number of valid elements in ``a`` to sum.
    """
    sum_out[0] = np.sum(a[:LEN_1D])
