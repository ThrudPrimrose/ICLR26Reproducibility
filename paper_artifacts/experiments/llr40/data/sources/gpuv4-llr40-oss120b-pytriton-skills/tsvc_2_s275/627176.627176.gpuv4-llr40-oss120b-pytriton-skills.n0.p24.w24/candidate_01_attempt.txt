"""Optimized implementation of TSVC kernel s275 using a hybrid Numba strategy.

The reference implementation loops over columns and, for each column where
``aa[0, i] > 0``, updates the remaining rows with a recurrence:

    aa[j, i] = aa[j-1, i] + bb[j, i] * cc[j, i]

Each column is independent, so the outer loop can be parallelised.  However, for
very small problem sizes the overhead of a parallel region can outweigh the
benefit.  This implementation therefore compiles **both** a sequential and a
parallel version with Numba and chooses between them at runtime based on the
matrix dimension ``LEN_2D``.

The function follows the in‑place ABI expected by the benchmark: it mutates the
``aa`` array and returns ``None``.
"""

import numpy as np
import numba

# Sequential version – good for small matrices
@numba.njit(fastmath=True)
def _s275_seq(aa, bb, cc, LEN_2D):
    mask = aa[0, :] > 0.0
    for i in range(LEN_2D):
        if mask[i]:
            for j in range(1, LEN_2D):
                aa[j, i] = aa[j - 1, i] + bb[j, i] * cc[j, i]

# Parallel version – beneficial for larger matrices
@numba.njit(parallel=True, fastmath=True)
def _s275_par(aa, bb, cc, LEN_2D):
    mask = aa[0, :] > 0.0
    for i in numba.prange(LEN_2D):
        if mask[i]:
            for j in range(1, LEN_2D):
                aa[j, i] = aa[j - 1, i] + bb[j, i] * cc[j, i]

def s275(aa, bb, cc, LEN_2D):
    # Heuristic threshold – empirical value that balances parallel overhead
    # against the work per thread.  Values around 256–512 work well on the
    # reference hardware; adjust if needed.
    if LEN_2D <= 512:
        _s275_seq(aa, bb, cc, LEN_2D)
    else:
        _s275_par(aa, bb, cc, LEN_2D)
    return None
