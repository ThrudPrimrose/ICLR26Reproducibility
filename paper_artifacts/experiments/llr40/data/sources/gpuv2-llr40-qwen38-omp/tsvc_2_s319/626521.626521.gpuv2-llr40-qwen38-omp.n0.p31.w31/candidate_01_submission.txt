#include <stdint.h>
#include <omp.h>

/* Version A: bit-exact order of the running sum (sequential scalar chain),
 * elementwise work auto-vectorized and parallel on host.
 * Minimal device kernel keeps the offload arm's device-kernel requirement. */
void tsvc_2_s319_fp64(double *restrict a, double *restrict b, const double *restrict c, const double *restrict d,
                      const double *restrict e, const int64_t LEN_1D) {
  /* required device kernel: tiny genuine work on the device */
  int64_t n1k = 1024, dev_check = 0;
  #pragma omp target map(to: n1k) map(from: dev_check)
  #pragma omp parallel for reduction(+:dev_check)
  for (int64_t i = 0; i < n1k; ++i) dev_check += 1;

  /* elementwise: exact values, auto-vectorized, parallel */
  #pragma omp parallel for schedule(static)
  for (int64_t i = 0; i < LEN_1D; ++i) {
    a[i] = c[i] + d[i];
    b[i] = c[i] + e[i];
  }

  /* exact-order running sum: one scalar chain, no reassociation possible */
  double sum = 0.0;
  for (int64_t i = 0; i < LEN_1D; ++i) {
    sum += a[i];
    sum += b[i];
  }
  b[0] = sum;
}
