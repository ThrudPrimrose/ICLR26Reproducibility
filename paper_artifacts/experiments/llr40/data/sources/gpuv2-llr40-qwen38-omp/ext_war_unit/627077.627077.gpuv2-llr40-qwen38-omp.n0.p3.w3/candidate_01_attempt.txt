#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <omp.h>

int setenv(const char *name, const char *value, int overwrite); /* not in _POSIX_C_SOURCE=199309L */

static void gpu_head(double *a, const double *b, int64_t k) {
    double tmp[16384];
    #pragma omp target map(tofrom: a[0:k + 1]) map(to: b[0:k]) map(from: tmp[0:k])
    {
        #pragma omp parallel for
        for (int64_t i = 0; i < k; ++i) tmp[i] = a[i + 1] + b[i];
        #pragma omp parallel for
        for (int64_t i = 0; i < k; ++i) a[i] = tmp[i];
    }
}

static void cpu_body(double *a, const double *b, int64_t k, int64_t n, int threads) {
    omp_set_num_threads(threads);
    #pragma omp parallel
    {
        int nt = omp_get_num_threads();
        static double S[1024];
        static int64_t F[1024];
        int64_t m = n - k;
        #pragma omp for schedule(static, 1)
        for (int c = 0; c < nt; ++c) {
            int64_t lo = k + (m / nt) * c;
            int64_t hi = (c + 1 == nt) ? n : k + (m / nt) * (c + 1);
            if (hi > lo) {
                double s = a[lo + 1];
                for (int64_t i = lo + 1; i < hi; ++i) a[i] = a[i + 1] + b[i];
                S[c] = s;
                F[c] = lo;
            } else {
                F[c] = -1;
            }
        }
        #pragma omp barrier
        #pragma omp for schedule(static, 1)
        for (int c = 0; c < nt; ++c)
            if (F[c] >= 0) a[F[c]] = S[c] + b[F[c]];
    }
}

static double timed_cpu_body(double *sa, double *sb, int64_t n, int threads) {
    double t0 = omp_get_wtime();
    cpu_body(sa, sb, 0, n, threads);
    double t1 = omp_get_wtime();
    printf("DIAG cpu threads=%d t=%.3f ms (bw %.2f GB/s)\n", threads, (t1 - t0) * 1000.0,
           3.0 * (double)n * 8.0 / (t1 - t0) / 1e9);
    fflush(stdout);
    return t1 - t0;
}

void ext_war_unit_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D) {
    if (LEN_1D < 2) return;
    int64_t n = LEN_1D - 1;
    static int inited = 0;
    if (!inited) {
        inited = 1;
        setenv("OMP_PLACES", "cores", 1);
        setenv("OMP_PROC_BIND", "spread", 1);
        setenv("OMP_WAIT_POLICY", "active", 1);
        printf("DIAG nproc=%ld OMP_NUM_THREADS=[%s]\n", (long)sysconf(_SC_NPROCESSORS_ONLN),
               getenv("OMP_NUM_THREADS") ? getenv("OMP_NUM_THREADS") : "(unset)");
        fflush(stdout);
    }
    int64_t k = (n > 16384) ? 16384 : n;
    gpu_head(a, b, k);
    cpu_body(a, b, k, n, 48);

    static int swept = 0;
    if (!swept) {
        swept = 1;
        double *sa = (double *)malloc((size_t)(n + 1) * sizeof(double));
        double *sb = (double *)malloc((size_t)n * sizeof(double));
        for (int64_t i = 0; i <= n; ++i) sa[i] = 1.0;
        for (int64_t i = 0; i < n; ++i) sb[i] = 2.0;
        timed_cpu_body(sa, sb, n, 32);
        timed_cpu_body(sa, sb, n, 40);
        timed_cpu_body(sa, sb, n, 48);
        timed_cpu_body(sa, sb, n, 64);
        double *ga = (double *)malloc(3 * 1024 * sizeof(double));
        double *gb = (double *)malloc(2 * 1024 * sizeof(double));
        double t5 = omp_get_wtime();
        gpu_head(ga, gb, 1024);
        printf("DIAG gpu_head steady1 t=%.3f ms\n", (omp_get_wtime() - t5) * 1000.0);
        double t6 = omp_get_wtime();
        gpu_head(ga, gb, 1024);
        printf("DIAG gpu_head steady2 t=%.3f ms\n", (omp_get_wtime() - t6) * 1000.0);
        fflush(stdout);
        free(ga);
        free(gb);
        free(sa);
        free(sb);
    }
}
