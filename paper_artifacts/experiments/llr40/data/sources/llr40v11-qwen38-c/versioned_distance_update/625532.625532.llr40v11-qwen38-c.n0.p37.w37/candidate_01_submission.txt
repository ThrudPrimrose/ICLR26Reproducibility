#include <stdint.h>

/* ABI: (a, b, c, K, LEN_1D)  -- K BEFORE LEN_1D.
 * a[i] = 0.75*a[i-K] + b[i]*c[i], i in [K, LEN_1D).  Only a is output.
 *
 * The judge pins the kernel to a single core, so this is a single-core
 * optimization. There are K independent chains (residue classes mod K);
 * a modern OoO core keeps up to K iterations in flight, hiding the FMA
 * latency automatically for K >= 4. Only K == 1 (a true serial recurrence)
 * needs a manual register to avoid a store->load round trip each step. */

void versioned_distance_update_fp64(double *restrict a, const double *restrict b,
                                    const double *restrict c, const int64_t K,
                                    const int64_t LEN_1D) {
    if (LEN_1D <= K) return;
    if (K == 1) {
        double x = a[0];
        for (int64_t i = 1; i < LEN_1D; ++i) {
            x = 0.75 * x + b[i] * c[i];
            a[i] = x;
        }
        return;
    }
    for (int64_t i = K; i < LEN_1D; ++i)
        a[i] = 0.75 * a[i - K] + b[i] * c[i];
}
