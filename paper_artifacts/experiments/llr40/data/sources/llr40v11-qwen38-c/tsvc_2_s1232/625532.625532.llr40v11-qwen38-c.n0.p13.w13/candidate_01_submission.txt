#include <stdint.h>

/* TSVC s1232: aa[i,j] = bb[i,j] + cc[i,j] for i >= j*VLEN, matrices N x N row-major.
 * Rewritten as: for each row i, columns j = 0..min(N-1, floor(i/VLEN)) -- unit stride. */

void tsvc_2_s1232_fp64(double *restrict aa, const double *restrict bb, const double *restrict cc,
                       const int64_t LEN_2D, const int64_t VLEN) {
  const int64_t N = LEN_2D;

#pragma omp parallel for schedule(static)
  for (int64_t i = 0; i < N; ++i) {
    int64_t jmax = (VLEN > 0) ? (i / VLEN + 1) : N;
    if (jmax > N)
      jmax = N;
    double *a = aa + i * N;
    const double *b = bb + i * N;
    const double *c = cc + i * N;
    for (int64_t j = 0; j < jmax; ++j)
      a[j] = b[j] + c[j];
  }
}
