#include <stdint.h>
#include <omp.h>

/* Segmented dot product over ragged CSR-style data.
 *
 * ABI (v2): out first, then the input arrays, then the size.
 *
 * Parallel strategy: split the element range [0, total) into nt contiguous
 * chunks; thread k owns every segment whose first element (row_ptr[s]) lies in
 * its chunk -- which is exactly one segment-count per element, so the heavy
 * lognormal-length tail can never unbalance the static split.  The inner
 * reduction keeps the reference associativity, so each out[s] is bit-identical
 * to the serial oracle. */

static inline int64_t lb_ge(const int64_t *rp, int64_t nseg, int64_t x) {
  /* first index i in [0, nseg+1) with rp[i] >= x */
  int64_t lo = 0, hi = nseg + 1;
  while (lo < hi) {
    int64_t mid = (lo + hi) >> 1;
    if (rp[mid] >= x) hi = mid;
    else lo = mid + 1;
  }
  return lo;
}

void segment_reduce_ragged_fp64(double *restrict out,
                                const int64_t *restrict row_ptr,
                                const double *restrict val,
                                const double *restrict w,
                                const int64_t NSEG) {
  if (NSEG <= 0) return;
  const int64_t total = row_ptr[NSEG];

  if (total < 4194304) {
    for (int64_t s = 0; s < NSEG; ++s) {
      double acc = 0.0;
      for (int64_t e = row_ptr[s]; e < row_ptr[s + 1]; ++e)
        acc += val[e] * w[e];
      out[s] = acc;
    }
    return;
  }

#pragma omp parallel
  {
    const int nt = omp_get_num_threads();
    const int tid = omp_get_thread_num();
    const int64_t base = total / nt, rem = total - base * nt;
    const int64_t lo = (int64_t)tid * base + (tid < rem ? tid : rem);
    const int64_t hi = lo + base + (tid < rem ? 1 : 0);
    /* the last thread also takes segments starting exactly at total (empty tail) */
    const int64_t hi_eff = hi + (tid == nt - 1 ? 1 : 0);

    const int64_t s_from = lb_ge(row_ptr, NSEG, lo);
    int64_t s_last = lb_ge(row_ptr, NSEG, hi_eff) - 1;
    if (s_last >= NSEG) s_last = NSEG - 1;

    for (int64_t s = s_from; s <= s_last; ++s) {
      double acc = 0.0;
      for (int64_t e = row_ptr[s]; e < row_ptr[s + 1]; ++e)
        acc += val[e] * w[e];
      out[s] = acc;
    }
  }
}
