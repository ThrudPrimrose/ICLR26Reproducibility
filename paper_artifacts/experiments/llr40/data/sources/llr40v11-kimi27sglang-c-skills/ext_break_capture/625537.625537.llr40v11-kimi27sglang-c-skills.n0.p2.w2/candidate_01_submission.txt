#include <stdint.h>
#include <limits.h>
#include <immintrin.h>
#include <omp.h>

static inline void find_first_serial(const double *restrict a, int64_t *restrict out_index,
                                     double *restrict out_value, int64_t LEN_1D, double k) {
    const __m256d vk = _mm256_set1_pd(k);
    int64_t i = 0;
    const int64_t n16 = LEN_1D & ~15;
    for (; i < n16; i += 16) {
        __m256d v0 = _mm256_loadu_pd(a + i + 0);
        __m256d v1 = _mm256_loadu_pd(a + i + 4);
        __m256d v2 = _mm256_loadu_pd(a + i + 8);
        __m256d v3 = _mm256_loadu_pd(a + i + 12);
        __m256d c0 = _mm256_cmp_pd(v0, vk, _CMP_GT_OQ);
        __m256d c1 = _mm256_cmp_pd(v1, vk, _CMP_GT_OQ);
        __m256d c2 = _mm256_cmp_pd(v2, vk, _CMP_GT_OQ);
        __m256d c3 = _mm256_cmp_pd(v3, vk, _CMP_GT_OQ);
        __m256d m01 = _mm256_or_pd(c0, c1);
        __m256d m23 = _mm256_or_pd(c2, c3);
        __m256d mall = _mm256_or_pd(m01, m23);
        if (_mm256_movemask_pd(mall)) {
            int j;
            int m;
            if ((m = _mm256_movemask_pd(c0))) {
                j = __builtin_ctz(m);
            } else if ((m = _mm256_movemask_pd(c1))) {
                j = 4 + __builtin_ctz(m);
            } else if ((m = _mm256_movemask_pd(c2))) {
                j = 8 + __builtin_ctz(m);
            } else {
                m = _mm256_movemask_pd(c3);
                j = 12 + __builtin_ctz(m);
            }
            out_index[0] = i + j;
            out_value[0] = a[i + j];
            return;
        }
    }
    const int64_t n4 = LEN_1D & ~3;
    for (; i < n4; i += 4) {
        __m256d v = _mm256_loadu_pd(a + i);
        int m = _mm256_movemask_pd(_mm256_cmp_pd(v, vk, _CMP_GT_OQ));
        if (m) {
            int j = __builtin_ctz(m);
            out_index[0] = i + j;
            out_value[0] = a[i + j];
            return;
        }
    }
    for (; i < LEN_1D; ++i) {
        if (a[i] > k) {
            out_index[0] = i;
            out_value[0] = a[i];
            return;
        }
    }
}

void ext_break_capture_fp64(const double *restrict a, int64_t *restrict out_index,
                            double *restrict out_value, const int64_t LEN_1D) {
    const double k = 1.0;
    out_index[0] = -1;
    out_value[0] = -1.0;

    if (LEN_1D <= 0) return;

    int nthreads = omp_get_max_threads();
    /* For small arrays or a single thread, keep the fast serial scan. */
    if (nthreads <= 1 || LEN_1D < (1LL << 18)) {
        find_first_serial(a, out_index, out_value, LEN_1D, k);
        return;
    }

    int64_t best = INT64_MAX;
    #pragma omp parallel for reduction(min:best) schedule(static)
    for (int64_t i = 0; i < LEN_1D; ++i) {
        if (a[i] > k) {
            if (i < best) best = i;
        }
    }

    if (best < LEN_1D) {
        out_index[0] = best;
        out_value[0] = a[best];
    }
}
