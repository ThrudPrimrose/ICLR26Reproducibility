#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

void tsvc_2_vag_fp64(double *restrict a, const double *restrict b, const int32_t *restrict ip, const int64_t LEN_1D) {
  const int nt = omp_get_max_threads();
  const int64_t chunk = (LEN_1D + nt - 1) / nt;

#pragma omp parallel for schedule(static)
  for (int t = 0; t < nt; ++t) {
    int64_t i = t * chunk;
    const int64_t end = (i + chunk < LEN_1D) ? i + chunk : LEN_1D;

    while (i < end && (((uintptr_t)(a + i)) & 63) != 0) {
      a[i] = b[ip[i]];
      ++i;
    }

    const int64_t vec_end = end - 15;
    for (; i < vec_end; i += 16) {
      __m256i idx0 = _mm256_loadu_si256((const __m256i*)(ip + i));
      __m256i idx1 = _mm256_loadu_si256((const __m256i*)(ip + i + 8));
      __m512d val0 = _mm512_i32gather_pd(idx0, (const void*)b, 8);
      __m512d val1 = _mm512_i32gather_pd(idx1, (const void*)b, 8);
      _mm512_stream_pd(a + i, val0);
      _mm512_stream_pd(a + i + 8, val1);
    }

    for (; i < end; ++i) {
      a[i] = b[ip[i]];
    }
  }
}
