#include <stdint.h>
#include <omp.h>

void fuse_move_ifs_fp64(double *restrict a, double *restrict b,
                        const double *restrict cond, const double *restrict src,
                        const int64_t K, const int64_t LEN_2D) {
    #pragma omp parallel
    {
        if (K > 0) {
            #pragma omp for
            for (int64_t i = 0; i < LEN_2D; ++i) {
                bool ca = cond[i] > 0.0;
                #pragma omp simd
                for (int64_t j = 0; j < LEN_2D; ++j) {
                    double s = src[i * LEN_2D + j];
                    b[i * LEN_2D + j] = s + 1.0;
                    if (ca) {
                        a[i * LEN_2D + j] = s * 2.0;
                    }
                }
            }
        } else {
            #pragma omp for
            for (int64_t i = 0; i < LEN_2D; ++i) {
                if (cond[i] > 0.0) {
                    #pragma omp simd
                    for (int64_t j = 0; j < LEN_2D; ++j) {
                        a[i * LEN_2D + j] = src[i * LEN_2D + j] * 2.0;
                    }
                }
            }
        }
    }
}
