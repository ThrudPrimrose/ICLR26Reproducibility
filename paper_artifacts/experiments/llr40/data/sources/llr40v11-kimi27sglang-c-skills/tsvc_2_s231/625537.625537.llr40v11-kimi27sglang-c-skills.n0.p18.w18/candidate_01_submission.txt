#include <stdint.h>
#include <omp.h>

#ifndef BLOCK
#define BLOCK 128
#endif

void tsvc_2_s231_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
    const int64_t n = LEN_2D;
    #pragma omp parallel for schedule(static,1)
    for (int64_t ii = 0; ii < n; ii += BLOCK) {
        const int64_t iend = (ii + BLOCK < n) ? ii + BLOCK : n;
        for (int64_t j = 1; j < n; ++j) {
            #pragma omp simd
            for (int64_t i = ii; i < iend; ++i) {
                aa[j * n + i] = aa[(j - 1) * n + i] + bb[j * n + i];
            }
        }
    }
}
