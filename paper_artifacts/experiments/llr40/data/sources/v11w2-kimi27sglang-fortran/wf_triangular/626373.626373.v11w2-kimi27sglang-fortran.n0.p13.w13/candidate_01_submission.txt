subroutine wf_triangular_fp64(a, LEN_2D) bind(c,name='wf_triangular_fp64')
  use iso_c_binding, only: c_double, c_int64_t
  implicit none
  real(c_double), intent(inout) :: a(0:*)
  integer(c_int64_t), value :: LEN_2D
  integer(c_int64_t) :: n, nm1, d, i0, i1, i, j

  n = LEN_2D
  if (n <= 1_c_int64_t) return
  nm1 = n - 1_c_int64_t

  !$omp parallel default(none) shared(a,n,nm1) private(d,i0,i1,i,j)
  do d = 2_c_int64_t, 2_c_int64_t*nm1
    i0 = max(1_c_int64_t, d - nm1)
    i1 = min(nm1, d/2_c_int64_t)
    !$omp do schedule(static)
    do i = i0, i1
      j = d - i
      a(i*n + j) = a(i*n + j) + a((i-1)*n + j) + a(i*n + (j-1))
    end do
    !$omp end do
  end do
  !$omp end parallel
end subroutine wf_triangular_fp64
