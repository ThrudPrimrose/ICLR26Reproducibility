subroutine tsvc_2_s3112_fp64(a, b, LEN_1D) bind(c, name='tsvc_2_s3112_fp64')
  use, intrinsic :: iso_c_binding
  use omp_lib
  implicit none
  real(C_DOUBLE), dimension(*), intent(in) :: a
  real(C_DOUBLE), dimension(*), intent(inout) :: b
  integer(C_INT64_T), value :: LEN_1D
  integer(C_INT64_T) :: i, j, tid, nthreads, start, end
  real(C_DOUBLE) :: sum, offset, total
  real(C_DOUBLE), allocatable :: offsets(:)
  integer(C_INT64_T) :: chunk_size
  integer, parameter :: VL = 8
  real(C_DOUBLE) :: v(VL)
  
  nthreads = omp_get_max_threads()
  chunk_size = ((LEN_1D + nthreads - 1_C_INT64_T) / nthreads + VL - 1_C_INT64_T) / VL * VL
  allocate(offsets(nthreads))
  
  !$omp parallel private(tid, start, end, i, j, sum, v)
  tid = omp_get_thread_num()
  start = tid * chunk_size + 1_C_INT64_T
  end = min(start + chunk_size - 1_C_INT64_T, LEN_1D)
  
  if (start <= end) then
    sum = 0.0_C_DOUBLE
    
    do i = start, end - VL + 1_C_INT64_T, VL
      do concurrent (j = 1:VL)
        v(j) = a(i + j - 1)
      end do
      do j = 2, VL
        v(j) = v(j) + v(j - 1)
      end do
      do concurrent (j = 1:VL)
        b(i + j - 1) = v(j) + sum
      end do
      sum = sum + v(VL)
    end do
    
    do i = i, end
      sum = sum + a(i)
      b(i) = sum
    end do
    
    offsets(tid + 1) = sum
  else
    offsets(tid + 1) = 0.0_C_DOUBLE
  end if
  !$omp barrier
  
  !$omp single
  total = 0.0_C_DOUBLE
  do i = 1, nthreads
    offsets(i) = offsets(i) + total
    total = offsets(i)
  end do
  !$omp end single
  
  if (start <= end .and. tid > 0) then
    offset = offsets(tid)
    !$omp simd simdlen(VL)
    do i = start, end
      b(i) = b(i) + offset
    end do
  end if
  !$omp end parallel
  
  deallocate(offsets)
end subroutine tsvc_2_s3112_fp64
