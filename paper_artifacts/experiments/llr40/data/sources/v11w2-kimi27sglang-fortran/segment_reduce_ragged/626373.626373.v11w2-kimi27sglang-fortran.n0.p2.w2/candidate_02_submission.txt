module segment_reduce_ragged_mod
  use iso_c_binding
  implicit none
contains
  subroutine segment_reduce_ragged_fp64(out, row_ptr, val, w, NSEG, workspace, workspace_bytes) bind(C, name="segment_reduce_ragged_fp64")
    integer(c_int64_t), value, intent(in) :: NSEG
    real(c_double), intent(out) :: out(NSEG)
    integer(c_int64_t), intent(in) :: row_ptr(NSEG + 1)
    real(c_double), intent(in) :: val(NSEG * 24)
    real(c_double), intent(in) :: w(NSEG * 24)
    type(c_ptr), value, intent(in) :: workspace
    integer(c_int64_t), value, intent(in) :: workspace_bytes
    integer(c_int64_t) :: s_l0, e_l1
    real(c_double) :: acc
    !$omp parallel do schedule(guided) private(s_l0, e_l1, acc)
    do s_l0 = 0_c_int64_t, NSEG - 1_c_int64_t
      acc = 0.0_c_double
      !$omp simd reduction(+:acc)
      do e_l1 = row_ptr(s_l0 + 1_c_int64_t), row_ptr(s_l0 + 2_c_int64_t) - 1_c_int64_t
        acc = acc + val(e_l1 + 1_c_int64_t) * w(e_l1 + 1_c_int64_t)
      end do
      out(s_l0 + 1_c_int64_t) = acc
    end do
  end subroutine segment_reduce_ragged_fp64
end module segment_reduce_ragged_mod
