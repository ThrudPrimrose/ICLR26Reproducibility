/* Offload-optimized version of tsvc_2_s255_fp64.
   Computes a circular 3‑point moving average of b into a.
   Uses an OpenMP target region with a combined teams distribute parallel for
   loop to perform the computation entirely on the device.
*/
#include <stdint.h>
#ifdef _OPENMP
#include <omp.h>
#endif

void tsvc_2_s255_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D) {
    if (LEN_1D <= 0) {
        return;
    }
    /* For tiny problem sizes fall back to the reference algorithm to avoid edge‑case indexing.
       This also handles LEN_1D == 1 or 2 safely. */
    if (LEN_1D <= 2) {
        double x = b[LEN_1D - 1];
        double y = b[LEN_1D - 2];
        for (int64_t i = 0; i < LEN_1D; ++i) {
            a[i] = (b[i] + x + y) * 0.333;
            y = x;
            x = b[i];
        }
        return;
    }

    /* Offload the main computation to the GPU. */
    #pragma omp target map(to: b[0:LEN_1D]) map(from: a[0:LEN_1D])
    #pragma omp teams distribute parallel for schedule(static)
    for (int64_t i = 0; i < LEN_1D; ++i) {
        int64_t im1 = i - 1;
        int64_t im2 = i - 2;
        if (im1 < 0) im1 += LEN_1D;
        if (im2 < 0) im2 += LEN_1D;
        a[i] = (b[i] + b[im1] + b[im2]) * 0.333;
    }
}
