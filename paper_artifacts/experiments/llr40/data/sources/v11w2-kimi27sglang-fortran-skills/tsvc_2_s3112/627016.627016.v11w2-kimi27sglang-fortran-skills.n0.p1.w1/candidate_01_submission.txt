module tsvc_2_s3112_mod
  use iso_c_binding, only: c_int64_t, c_double
  implicit none
  integer, save :: saved_nt = -1
  real(c_double), allocatable, save :: part(:)
contains
  subroutine tsvc_2_s3112_fp64(a, b, n) bind(C, name="tsvc_2_s3112_fp64")
    use omp_lib, only: omp_get_max_threads, omp_get_thread_num
    implicit none
    integer(c_int64_t), value, intent(in) :: n
    real(c_double), intent(in) :: a(n)
    real(c_double), intent(out) :: b(n)
    integer(c_int64_t) :: i, lo, hi
    integer :: nt, t, j
    real(c_double) :: s, run

    if (n <= 0) return

    nt = omp_get_max_threads()
    if (nt /= saved_nt) then
       if (allocated(part)) deallocate(part)
       allocate(part(0:nt))
       saved_nt = nt
    end if

    if (n < 4096 .or. nt <= 1) then
       s = 0.0d0
       do i = 1, n
          s = s + a(i)
          b(i) = s
       end do
       return
    end if

    part(0:nt) = 0.0d0

    !$omp parallel private(t, lo, hi, run, i)
    t = omp_get_thread_num()
    lo = (n * int(t, kind=c_int64_t)) / int(nt, kind=c_int64_t) + 1
    hi = (n * int(t + 1, kind=c_int64_t)) / int(nt, kind=c_int64_t)
    run = 0.0d0
    do i = lo, hi
       run = run + a(i)
    end do
    part(t + 1) = run
    !$omp barrier
    !$omp single
    do j = 1, nt
       part(j) = part(j) + part(j - 1)
    end do
    !$omp end single
    run = part(t)
    do i = lo, hi
       run = run + a(i)
       b(i) = run
    end do
    !$omp end parallel
  end subroutine tsvc_2_s3112_fp64
end module tsvc_2_s3112_mod
