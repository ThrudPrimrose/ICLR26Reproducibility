subroutine tsvc_2_vpvts(a, b, LEN_1D, S) bind(C, name='tsvc_2_vpvts_fp64')
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: LEN_1D, S
  real(c_double), intent(inout) :: a(LEN_1D)
  real(c_double), intent(in) :: b(LEN_1D)

  a = a + b * real(S, c_double)
end subroutine tsvc_2_vpvts
