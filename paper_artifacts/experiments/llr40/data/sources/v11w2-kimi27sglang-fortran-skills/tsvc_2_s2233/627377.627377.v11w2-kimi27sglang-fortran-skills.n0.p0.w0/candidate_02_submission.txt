subroutine tsvc_2_s2233_fp64(aa, bb, cc, LEN_2D) bind(C)
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: LEN_2D
  real(c_double), intent(inout) :: aa(LEN_2D, LEN_2D)
  real(c_double), intent(inout) :: bb(LEN_2D, LEN_2D)
  real(c_double), intent(in) :: cc(LEN_2D, LEN_2D)
  integer(c_int64_t) :: i, j, n, B, nt, it, jt, ib, ie, jb, je

  n = LEN_2D
  B = 256

  nt = (n - 8 + B - 1) / B

  do jt = 1, nt
    jb = (jt - 1) * B + 9
    je = min(jb + B - 1, n)
    !$omp parallel do schedule(static) private(it, ib, ie, i, j)
    do it = 1, nt
      ib = (it - 1) * B + 9
      ie = min(ib + B - 1, n)
      do j = jb, je
        !$omp simd
        do i = ib, ie
          aa(i, j) = aa(i, j - 1) + cc(i, j)
        end do
        !$omp end simd
      end do
    end do
    !$omp end parallel do
  end do

  do it = 1, nt
    ib = (it - 1) * B + 9
    ie = min(ib + B - 1, n)
    !$omp parallel do schedule(static) private(jt, jb, je, i, j)
    do jt = 1, nt
      jb = (jt - 1) * B + 9
      je = min(jb + B - 1, n)
      do i = ib, ie
        !$omp simd
        do j = jb, je
          bb(j, i) = bb(j, i - 1) + cc(j, i)
        end do
        !$omp end simd
      end do
    end do
    !$omp end parallel do
  end do

end subroutine tsvc_2_s2233_fp64
