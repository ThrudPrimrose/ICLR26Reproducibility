subroutine tsvc_2_s2710_fp64(a, b, c, d, e, x, len_1d) bind(C, name='tsvc_2_s2710_fp64')
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: len_1d
  real(c_double), intent(inout) :: a(len_1d), b(len_1d), c(len_1d)
  real(c_double), intent(in) :: d(len_1d), e(len_1d), x(len_1d)
  integer(c_int64_t) :: i
  real(c_double) :: m, ai, bi, di, ei

  if (len_1d > 10) then
     if (x(1) > 0.0d0) then
        !$omp parallel do simd schedule(static)
        do i = 1, len_1d
           ai = a(i); bi = b(i); di = d(i); ei = e(i)
           m = merge(1.0d0, 0.0d0, ai > bi)
           c(i) = di*di + m*c(i) + (1.0d0 - m)*ai
           b(i) = m*bi + (1.0d0 - m)*(ai + ei*ei)
           a(i) = ai + m*bi*di
        end do
     else
        !$omp parallel do simd schedule(static)
        do i = 1, len_1d
           ai = a(i); bi = b(i); di = d(i); ei = e(i)
           m = merge(1.0d0, 0.0d0, ai > bi)
           c(i) = c(i) + m*di*di + (1.0d0 - m)*ei*ei
           b(i) = m*bi + (1.0d0 - m)*(ai + ei*ei)
           a(i) = ai + m*bi*di
        end do
     end if
  else
     if (x(1) > 0.0d0) then
        !$omp parallel do simd schedule(static)
        do i = 1, len_1d
           ai = a(i); bi = b(i); di = d(i); ei = e(i)
           m = merge(1.0d0, 0.0d0, ai > bi)
           c(i) = m*(di*ei + 1.0d0) + (1.0d0 - m)*(ai + di*di)
           b(i) = m*bi + (1.0d0 - m)*(ai + ei*ei)
           a(i) = ai + m*bi*di
        end do
     else
        !$omp parallel do simd schedule(static)
        do i = 1, len_1d
           ai = a(i); bi = b(i); di = d(i); ei = e(i)
           m = merge(1.0d0, 0.0d0, ai > bi)
           c(i) = m*(di*ei + 1.0d0) + (1.0d0 - m)*(c(i) + ei*ei)
           b(i) = m*bi + (1.0d0 - m)*(ai + ei*ei)
           a(i) = ai + m*bi*di
        end do
     end if
  end if
end subroutine tsvc_2_s2710_fp64
