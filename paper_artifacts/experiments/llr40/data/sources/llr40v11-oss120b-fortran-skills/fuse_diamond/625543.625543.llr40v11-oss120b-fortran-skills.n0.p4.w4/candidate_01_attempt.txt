subroutine fuse_diamond_fp64(a, out, len_1d) bind(C, name="fuse_diamond_fp64")
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: len_1d
  real(c_double), intent(in) :: a(*)
  real(c_double), intent(inout) :: out(*)
  integer(c_int64_t) :: i
  do i = 1, len_1d
    out(i) = 0.0d0
  end do
end subroutine fuse_diamond_fp64
