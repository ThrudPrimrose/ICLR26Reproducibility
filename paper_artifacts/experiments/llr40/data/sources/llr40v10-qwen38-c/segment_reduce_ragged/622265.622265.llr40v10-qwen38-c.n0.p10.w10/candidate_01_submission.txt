#include <stdint.h>
#include <omp.h>

static void srr_impl(double *restrict out, const int64_t *restrict row_ptr,
                     const double *restrict val, const double *restrict w,
                     const int64_t NSEG)
{
    if (NSEG <= 0) return;
    const int64_t total = row_ptr[NSEG];
    int nt = (int)omp_get_max_threads();
    if (nt < 1) nt = 1;
    if ((int64_t)nt > NSEG) nt = (int)NSEG;
    if ((int64_t)nt > total) nt = (int)total;
    if (nt <= 1) {
        for (int64_t s = 0; s < NSEG; ++s) {
            const int64_t rs = row_ptr[s];
            int64_t n = row_ptr[s+1] - rs;
            const double *pv = val + rs;
            const double *pw = w + rs;
            double acc = 0.0;
            for (int64_t e = 0; e < n; ++e) acc += pv[e] * pw[e];
            out[s] = acc;
        }
        return;
    }
    #pragma omp parallel num_threads(nt)
    {
        const int64_t tid = omp_get_thread_num();
        const int64_t lo = (total * tid) / nt;
        const int64_t hi = (total * (tid + 1)) / nt;
        int64_t a = 0, b = NSEG;
        while (a < b) { int64_t mid = (a + b) >> 1; if (row_ptr[mid+1] <= lo) a = mid+1; else b = mid; }
        int64_t s_lo = a;
        a = s_lo; b = NSEG;
        while (a < b) { int64_t mid = (a + b) >> 1; if (row_ptr[mid+1] <= hi) a = mid+1; else b = mid; }
        int64_t s_hi = a;
        for (int64_t s = s_lo; s < s_hi; ++s) {
            const int64_t rs = row_ptr[s];
            int64_t n = row_ptr[s+1] - rs;
            const double *pv = val + rs;
            const double *pw = w + rs;
            double acc = 0.0;
            for (int64_t e = 0; e < n; ++e) acc += pv[e] * pw[e];
            out[s] = acc;
        }
    }
}

void segment_reduce_ragged_fp64(double *restrict out, const int64_t *restrict row_ptr,
                                const double *restrict val, const double *restrict w,
                                const int64_t NSEG)
{ srr_impl(out, row_ptr, val, w, NSEG); }
