#include <stdint.h>
#include <omp.h>

void fuse_move_ifs_fp64(double *restrict a, double *restrict b, const double *restrict cond, const double *restrict src,
                        const int64_t K, const int64_t LEN_2D) {
  const int do_b = (K > 0);
  #pragma omp parallel for schedule(static)
  for (int64_t i = 0; i < LEN_2D; ++i) {
    const double *restrict si = src + i * LEN_2D;
    if (cond[i] > 0.0) {
      double *restrict ai = a + i * LEN_2D;
      if (do_b) {
        double *restrict bi = b + i * LEN_2D;
        for (int64_t j = 0; j < LEN_2D; ++j) {
          ai[j] = si[j] * 2.0;
          bi[j] = si[j] + 1.0;
        }
      } else {
        for (int64_t j = 0; j < LEN_2D; ++j) ai[j] = si[j] * 2.0;
      }
    } else if (do_b) {
      double *restrict bi = b + i * LEN_2D;
      for (int64_t j = 0; j < LEN_2D; ++j) bi[j] = si[j] + 1.0;
    }
  }
}
