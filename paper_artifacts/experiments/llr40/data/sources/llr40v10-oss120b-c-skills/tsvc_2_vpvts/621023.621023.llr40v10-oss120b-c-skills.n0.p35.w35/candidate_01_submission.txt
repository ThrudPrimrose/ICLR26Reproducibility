/* Optimized implementation of the tsvc_2 vpvts kernel.
   Computes a[i] += b[i] * S for i in [0, LEN_1D).
   Uses a runtime threshold to avoid OpenMP overhead for small arrays.
*/

#include <stdint.h>
#include <omp.h>

void tsvc_2_vpvts_fp64(double *restrict a, const double *restrict b,
                       const int64_t LEN_1D, const int64_t S) {
    const double Sd = (double)S;
    // Use OpenMP parallelism only for sufficiently large workloads.
    if (LEN_1D >= 1024) {
        #pragma omp parallel for schedule(static)
        for (int64_t i = 0; i < LEN_1D; ++i) {
            a[i] += b[i] * Sd;
        }
    } else {
        for (int64_t i = 0; i < LEN_1D; ++i) {
            a[i] += b[i] * Sd;
        }
    }
}
