/* Optimized version of tsvc_2_s2710 kernel with OpenMP parallelization and
 * loop-invariant condition hoisting. */

#include <stdint.h>
#include <stdbool.h>
#include <omp.h>

void tsvc_2_s2710_fp64(double *restrict a, double *restrict b, double *restrict c,
                       const double *restrict d, const double *restrict e,
                       const double *restrict x, const int64_t LEN_1D) {
    // Loop-invariant conditions hoisted out of the loop
    const bool len_gt_10 = (LEN_1D > 10);
    const bool x_gt0 = (x[0] > 0.0);

    #pragma omp parallel for simd if(LEN_1D > 1024)
    for (int64_t i = 0; i < LEN_1D; ++i) {
        if (a[i] > b[i]) {
            a[i] += b[i] * d[i];
            if (len_gt_10) {
                c[i] += d[i] * d[i];
            } else {
                c[i] = d[i] * e[i] + 1.0;
            }
        } else {
            b[i] = a[i] + e[i] * e[i];
            if (x_gt0) {
                c[i] = a[i] + d[i] * d[i];
            } else {
                c[i] += e[i] * e[i];
            }
        }
    }
}
