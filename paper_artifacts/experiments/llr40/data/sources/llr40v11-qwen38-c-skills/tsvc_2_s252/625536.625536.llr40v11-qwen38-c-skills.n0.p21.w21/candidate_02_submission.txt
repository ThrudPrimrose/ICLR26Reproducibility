#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

/*
 * Reference:
 *   t = 0.0
 *   for i: s = b[i]*c[i]; a[i] = s + t; t = s
 * Carry is one step: a[0] = b[0]*c[0]; a[i] = b[i]*c[i] + b[i-1]*c[i-1].
 * Bit-identical to reference. 64B-aligned non-temporal stores stream a
 * (bypass cache, no write-allocate); reads use unaligned 128B windows.
 */
void tsvc_2_s252_fp64(double *restrict a,
                      const double *restrict b,
                      const double *restrict c,
                      const int64_t LEN_1D) {
    if (LEN_1D <= 0) return;
    if (LEN_1D == 1) { a[0] = b[0] * c[0]; return; }
    a[0] = b[0] * c[0];
    int64_t i = 1;
    while (((uintptr_t)(a + i) & 63) && i < LEN_1D) {
        a[i] = b[i] * c[i] + b[i - 1] * c[i - 1];
        ++i;
    }
    const int64_t nvec = (LEN_1D - i) / 8;
    #pragma omp parallel for schedule(static)
    for (int64_t v = 0; v < nvec; ++v) {
        const int64_t s = i + v * 8;
        __m512d r = _mm512_fmadd_pd(_mm512_loadu_pd(b + s), _mm512_loadu_pd(c + s),
                                    _mm512_mul_pd(_mm512_loadu_pd(b + s - 1),
                                                  _mm512_loadu_pd(c + s - 1)));
        _mm512_stream_pd(a + s, r);
    }
    for (int64_t t = i + nvec * 8; t < LEN_1D; ++t) {
        a[t] = b[t] * c[t] + b[t - 1] * c[t - 1];
    }
}
