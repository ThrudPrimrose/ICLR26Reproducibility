subroutine segment_reduce_ragged_fp64(out, row_ptr, val, w, NSEG, workspace, wssize) bind(C, name="segment_reduce_ragged_fp64")
    use, intrinsic :: iso_c_binding
    implicit none
    integer(c_int64_t), value, intent(in) :: NSEG
    real(c_double), intent(inout) :: out(NSEG)
    integer(c_int64_t), intent(in) :: row_ptr(NSEG + 1)
    real(c_double), intent(in) :: val(NSEG * 24)
    real(c_double), intent(in) :: w(NSEG * 24)
    type(c_ptr), intent(in) :: workspace
    integer(c_int64_t), value, intent(in) :: wssize
    integer(c_int64_t) :: s, e
    real(c_double) :: acc
    do s = 1, NSEG
        acc = 0.0_c_double
        do e = row_ptr(s) + 1, row_ptr(s + 1)
            acc = acc + val(e) * w(e)
        end do
        out(s) = acc
    end do
end subroutine segment_reduce_ragged_fp64
