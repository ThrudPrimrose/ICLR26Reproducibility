subroutine tsvc_2_vpvts_fp64(a, b, len_1d, s) bind(C, name='tsvc_2_vpvts_fp64')
  use iso_c_binding
  implicit none
  real(kind=c_double), intent(inout) :: a(*)
  real(kind=c_double), intent(in) :: b(*)
  integer(kind=c_int64_t), value :: len_1d
  integer(kind=c_int64_t), value :: s
  integer(kind=8) :: i
  real(kind=c_double) :: sd

  sd = dble(s)
  !$omp parallel do schedule(static)
  do i = 1_8, len_1d
     a(i) = a(i) + b(i) * sd
  end do
  !$omp end parallel do
end subroutine
