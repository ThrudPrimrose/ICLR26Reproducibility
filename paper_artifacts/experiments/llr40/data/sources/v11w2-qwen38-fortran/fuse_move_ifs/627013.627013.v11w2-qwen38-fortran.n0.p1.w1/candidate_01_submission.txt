subroutine fuse_move_ifs_fp64(a, b, cond, src, K, LEN_2D) bind(C, name="fuse_move_ifs_fp64")
  use iso_c_binding
  implicit none
  integer(c_int64_t), intent(in), value :: K
  integer(c_int64_t), intent(in), value :: LEN_2D
  real(c_double), intent(inout) :: a(0:LEN_2D*LEN_2D-1)
  real(c_double), intent(inout) :: b(0:LEN_2D*LEN_2D-1)
  real(c_double), intent(in) :: cond(0:LEN_2D-1)
  real(c_double), intent(in) :: src(0:LEN_2D*LEN_2D-1)

  integer(c_int64_t) :: n, i, j

  n = LEN_2D

  !$omp parallel do schedule(static) default(none) shared(a,cond,src,n)
  do i = 0, n - 1
     if (cond(i) > 0.0d0) then
        do j = 0, n - 1
           a(i * n + j) = src(i * n + j) * 2.0d0
        end do
     end if
  end do

  if (K > 0) then
     !$omp parallel do schedule(static) default(none) shared(b,src,n)
     do i = 0, n - 1
        do j = 0, n - 1
           b(i * n + j) = src(i * n + j) + 1.0d0
        end do
     end do
  end if
end subroutine fuse_move_ifs_fp64
