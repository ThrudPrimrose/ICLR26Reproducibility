subroutine tsvc_2_vpvts_fp64(a, b, len_1d, s) bind(C)
  use iso_c_binding
  use omp_lib
  implicit none
  integer(c_int64_t), value, intent(in) :: len_1d, s
  real(c_double), intent(inout) :: a(len_1d)
  real(c_double), intent(in) :: b(len_1d)
  real(c_double) :: s_d
  integer(c_int64_t) :: i, j, base, nchunks
  integer :: nthr

  s_d = real(s, c_double)
  nthr = omp_get_max_threads()
  nchunks = len_1d / 32
  !$omp parallel do schedule(static)
  do i = 1, nchunks
    base = 32 * (i - 1)
    do j = 1, 32
      a(base + j) = a(base + j) + b(base + j) * s_d
    end do
  end do
  do i = 32 * nchunks + 1, len_1d
    a(i) = a(i) + b(i) * s_d
  end do
end subroutine tsvc_2_vpvts_fp64
