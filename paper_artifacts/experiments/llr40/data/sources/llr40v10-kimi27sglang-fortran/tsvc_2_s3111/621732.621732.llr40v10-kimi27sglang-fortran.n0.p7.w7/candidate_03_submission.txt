subroutine tsvc_2_s3111_fp64(a, b, LEN_1D) bind(c, name='tsvc_2_s3111_fp64')
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: LEN_1D
  real(c_double), intent(in) :: a(LEN_1D)
  real(c_double), intent(out) :: b(2)
  real(c_double) :: s0, s1, s2, s3, s4, s5, s6, s7, x
  integer(c_int64_t) :: i, nrem

  s0 = 0.0_c_double
  s1 = 0.0_c_double
  s2 = 0.0_c_double
  s3 = 0.0_c_double
  s4 = 0.0_c_double
  s5 = 0.0_c_double
  s6 = 0.0_c_double
  s7 = 0.0_c_double

  nrem = iand(LEN_1D, 7_c_int64_t)
  do i = 1, LEN_1D - nrem, 8
     x = a(i)
     s0 = s0 + max(x, 0.0_c_double)
     x = a(i+1)
     s1 = s1 + max(x, 0.0_c_double)
     x = a(i+2)
     s2 = s2 + max(x, 0.0_c_double)
     x = a(i+3)
     s3 = s3 + max(x, 0.0_c_double)
     x = a(i+4)
     s4 = s4 + max(x, 0.0_c_double)
     x = a(i+5)
     s5 = s5 + max(x, 0.0_c_double)
     x = a(i+6)
     s6 = s6 + max(x, 0.0_c_double)
     x = a(i+7)
     s7 = s7 + max(x, 0.0_c_double)
  end do

  ! remainder
  do i = LEN_1D - nrem + 1, LEN_1D
     x = a(i)
     s0 = s0 + max(x, 0.0_c_double)
  end do

  b(1) = ((s0 + s4) + (s1 + s5)) + ((s2 + s6) + (s3 + s7))
end subroutine tsvc_2_s3111_fp64
