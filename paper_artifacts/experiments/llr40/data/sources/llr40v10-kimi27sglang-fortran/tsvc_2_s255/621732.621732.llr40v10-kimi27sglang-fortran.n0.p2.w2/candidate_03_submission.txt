subroutine tsvc_2_s255_fp64(a, b, LEN_1D) bind(C,name="tsvc_2_s255_fp64")
  use iso_c_binding, only: c_double, c_int64_t
  implicit none
  real(c_double), intent(inout) :: a(*)
  real(c_double), intent(in)    :: b(*)
  integer(c_int64_t), value, intent(in) :: LEN_1D
  integer(c_int64_t) :: i, n
  real(c_double) :: x, y

  n = LEN_1D
  if (n <= 0) return
  if (n == 1) then
    a(1) = (b(1) + b(1) + b(1)) * 0.333_c_double
    return
  end if

  x = b(n)
  y = b(n - 1)
  a(1) = (b(1) + x + y) * 0.333_c_double
  y = x
  x = b(1)
  a(2) = (b(2) + x + y) * 0.333_c_double

  if (n >= 3) then
    !$omp parallel do simd simdlen(4) if(n > 2048) schedule(static)
    do i = 3, n
      a(i) = (b(i) + b(i - 1) + b(i - 2)) * 0.333_c_double
    end do
    !$omp end parallel do simd
  end if
end subroutine tsvc_2_s255_fp64
