module tsvc_2_s255_mod
  use iso_c_binding, only: c_double, c_int64_t
  implicit none
contains
  subroutine tsvc_2_s255_fp64(a, b, LEN_1D) bind(C,name="tsvc_2_s255_fp64")
    real(c_double), intent(inout) :: a(*)
    real(c_double), intent(in)    :: b(*)
    integer(c_int64_t), value, intent(in) :: LEN_1D
    integer(c_int64_t) :: i, n
    real(c_double) :: x, y

    n = LEN_1D
    if (n <= 0) return

    ! Match NumPy wrap-around semantics for the degenerate n==1 case.
    if (n == 1) then
      x = b(1)
      a(1) = (b(1) + x + x) * 0.333_c_double
      return
    end if

    ! First two iterations carry the wrap-around values from the end of b.
    x = b(n)
    y = b(n - 1)
    do i = 1, 2
      a(i) = (b(i) + x + y) * 0.333_c_double
      y = x
      x = b(i)
    end do

    ! The remaining iterations are a plain 3-point stencil and vectorize.
    if (n >= 3) then
      !$omp simd simdlen(4)
      do i = 3, n
        a(i) = (b(i) + b(i - 1) + b(i - 2)) * 0.333_c_double
      end do
      !$omp end simd
    end if
  end subroutine tsvc_2_s255_fp64
end module tsvc_2_s255_mod
