subroutine tsvc_2_s318_fp64(a, result, LEN_1D, inc) bind(C, name="tsvc_2_s318_fp64")
  use iso_c_binding
  implicit none
  integer(c_int64_t), value :: LEN_1D, inc
  real(c_double), intent(in) :: a(0:LEN_1D*inc-1)
  real(c_double), intent(inout) :: result(0:0)
  integer(c_int64_t) :: i, index
  real(c_double) :: maxv, v

  index = 0_c_int64_t
  maxv = abs(a(0_c_int64_t))

  do i = 1_c_int64_t, LEN_1D - 1_c_int64_t
    v = abs(a(i * inc))
    if (v > maxv) then
      index = i
      maxv = v
    end if
  end do

  result(0_c_int64_t) = maxv + real(index, c_double)
end subroutine tsvc_2_s318_fp64
