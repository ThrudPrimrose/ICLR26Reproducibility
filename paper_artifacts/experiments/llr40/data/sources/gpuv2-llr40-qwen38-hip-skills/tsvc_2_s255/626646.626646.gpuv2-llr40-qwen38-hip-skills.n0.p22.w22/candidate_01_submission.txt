/* Device half: kernel + launcher.
 *
 * The reference recurrence
 *     x = b[N-1]; y = b[N-2];
 *     for i in 0..N: a[i] = (b[i] + x + y) * 0.333; y = x; x = b[i];
 * degenerates: after iteration i, x == b[i] always, so
 *     a[i] = (b[i] + b[i-1 mod N] + b[i-2 mod N]) * 0.333
 * with wrap-around reproduced exactly, including for i = 0, 1. Each output
 * element is an independent function of three input elements -> embarrassingly
 * parallel, bitwise identical to the sequential reference (same op order).
 */
#include <hip/hip_runtime.h>
#include <stdint.h>
#include <cstdio>
#include <cstdlib>

#define HIP_CHECK(x)                                                                \
    do {                                                                            \
        hipError_t e_ = (x);                                                        \
        if (e_ != hipSuccess) {                                                     \
            std::fprintf(stderr, "HIP error %s: %s (line %d)\n", #x,               \
                         hipGetErrorString(e_), __LINE__);                          \
            std::abort();                                                           \
        }                                                                           \
    } while (0)

__global__ void s255_kernel(double *__restrict__ a, const double *__restrict__ b, int64_t n) {
    const int64_t stride = (int64_t)gridDim.x * (int64_t)blockDim.x;
    for (int64_t i = (int64_t)blockIdx.x * (int64_t)blockDim.x + threadIdx.x; i < n; i += stride) {
        const int64_t i1 = (i == 0) ? (n - 1) : (i - 1);
        const int64_t i2 = (i < 2) ? (n - 2 + i) : (i - 2);
        a[i] = (b[i] + b[i1] + b[i2]) * 0.333;
    }
}

void tsvc_2_s255_launch(double *a, const double *b, int64_t n) {
    if (n <= 0) return;
    constexpr int kBlock = 256;
    int64_t want = (n + kBlock - 1) / kBlock;
    const int grid = (want > 32768) ? 32768 : (int)want;
    s255_kernel<<<dim3(grid), dim3(kBlock), 0, nullptr>>>(a, b, n);
    HIP_CHECK(hipGetLastError());
    HIP_CHECK(hipDeviceSynchronize());
}
