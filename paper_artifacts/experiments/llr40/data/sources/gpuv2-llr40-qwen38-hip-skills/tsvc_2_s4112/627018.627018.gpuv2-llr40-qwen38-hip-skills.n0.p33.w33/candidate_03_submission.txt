#include <hip/hip_runtime.h>
#include <cstdint>

__global__ void tsvc2_gather(double* __restrict__ a, const double* __restrict__ b,
                             const int32_t* __restrict__ ip, const int64_t n) {
  int64_t i = (int64_t)blockIdx.x * blockDim.x + threadIdx.x;
  if (i < n) a[i] += b[ip[i]] * 2.0;
}

extern "C" void tsvc_2_s4112_launcher(double* a, const double* b, const int32_t* ip, int64_t n) {
  if (n <= 0) return;
  const int block = 256;
  int64_t grid = (n + block - 1) / block;
  tsvc2_gather<<<grid, block>>>(a, b, ip, n);
  hipError_t e = hipGetLastError();
  if (e != hipSuccess) { fprintf(stderr, "launch: %s\n", hipGetErrorString(e)); abort(); }
  e = hipDeviceSynchronize();
  if (e != hipSuccess) { fprintf(stderr, "sync: %s\n", hipGetErrorString(e)); abort(); }
}
