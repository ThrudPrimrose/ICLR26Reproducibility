// Device half: kernel + launcher.
#include <hip/hip_runtime.h>
#include <cstdint>

__global__ void tsvc_2_s252_kernel(double* __restrict__ a,
                                   const double* __restrict__ b,
                                   const double* __restrict__ c,
                                   const long long n)
{
    const long long i = (long long)blockIdx.x * (long long)blockDim.x + (long long)threadIdx.x;
    if (i < n) {
        const double s = b[i] * c[i];
        const double t = (i > 0) ? b[i - 1] * c[i - 1] : 0.0;
        a[i] = s + t;
    }
}

extern "C" void tsvc_2_s252_launch(double* a, const double* b, const double* c, const int64_t n)
{
    if (n <= 0) return;
    const int threads = 256;
    const long long blocks = (n + (long long)threads - 1) / (long long)threads;
    tsvc_2_s252_kernel<<<(unsigned)blocks, threads>>>(a, b, c, n);
}
