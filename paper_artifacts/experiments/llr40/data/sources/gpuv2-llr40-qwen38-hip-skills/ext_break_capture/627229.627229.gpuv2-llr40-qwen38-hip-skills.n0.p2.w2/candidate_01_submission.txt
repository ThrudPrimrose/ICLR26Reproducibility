#include <hip/hip_runtime.h>
#include <cstdint>
#include <climits>

#define EBC_BLOCK 256u

__global__ void __launch_bounds__(EBC_BLOCK) ebc_kernel(
    const double* __restrict__ a, const int64_t n, uint64_t* __restrict__ g_val,
    uint32_t* __restrict__ g_done, int64_t* __restrict__ out_index, double* __restrict__ out_value,
    const unsigned warp_size) {
  const int64_t stride = (int64_t)gridDim.x * (int64_t)EBC_BLOCK;
  const int64_t base = (int64_t)blockIdx.x * (int64_t)EBC_BLOCK + threadIdx.x;

  /* per-thread first index i with a[i] > 1.0 (4-way unroll for MLP) */
  int64_t idx = -1;
  int64_t i = base;
  while (i + 3 * stride < n) {
    const double x0 = a[i];
    const double x1 = a[i + stride];
    const double x2 = a[i + 2 * stride];
    const double x3 = a[i + 3 * stride];
    if (x0 > 1.0) { idx = i; break; }
    if (x1 > 1.0) { idx = i + stride; break; }
    if (x2 > 1.0) { idx = i + 2 * stride; break; }
    if (x3 > 1.0) { idx = i + 3 * stride; break; }
    i += 4 * stride;
  }
  for (; i < n; i += stride) {
    if (a[i] > 1.0) { idx = i; break; }
  }

  /* transform: found -> UINT64_MAX - idx (max over found == smallest idx); not found -> 0 */
  const uint64_t u = (idx >= 0) ? (UINT64_MAX - (uint64_t)idx) : 0ULL;
  unsigned hi = (unsigned)(u >> 32), lo = (unsigned)u;
  for (unsigned off = warp_size / 2; off > 0; off >>= 1) {
    const unsigned h2 = __shfl_down(hi, (int)off, (int)warp_size);
    const unsigned l2 = __shfl_down(lo, (int)off, (int)warp_size);
    if (h2 > hi || (h2 == hi && l2 > lo)) { hi = h2; lo = l2; }
  }
  const int wid = (int)(threadIdx.x / warp_size);
  const int nwarp = (int)(EBC_BLOCK / warp_size);
  __shared__ unsigned s_hi[8], s_lo[8];
  if ((int)(threadIdx.x % warp_size) == 0) { s_hi[wid] = hi; s_lo[wid] = lo; }
  __syncthreads();
  if (threadIdx.x == 0) {
    unsigned H = s_hi[0], L = s_lo[0];
    for (int w = 1; w < nwarp; ++w) {
      const unsigned h2 = s_hi[w], l2 = s_lo[w];
      if (h2 > H || (h2 == H && l2 > L)) { H = h2; L = l2; }
    }
    const uint64_t blockv = ((uint64_t)H << 32) | (uint64_t)L;
    if (blockv != 0ULL) (void)__hip_atomic_fetch_max(g_val, blockv, __ATOMIC_SEQ_CST, __HIP_MEMORY_SCOPE_AGENT);
    const unsigned old = __hip_atomic_fetch_add(g_done, 1u, __ATOMIC_SEQ_CST, __HIP_MEMORY_SCOPE_AGENT);
    if (old == (unsigned)gridDim.x - 1u) {
      /* all other blocks' atomicMax is sequenced before their atomicAdd (seq_cst total order) */
      const uint64_t v = __hip_atomic_load(g_val, __ATOMIC_SEQ_CST, __HIP_MEMORY_SCOPE_AGENT);
      if (v == 0ULL) {
        out_index[0] = -1;
        out_value[0] = -1.0;
      } else {
        const int64_t f = (int64_t)(UINT64_MAX - v);
        out_index[0] = f;
        out_value[0] = a[f];
      }
    }
  }
}

void ebc_device_launch(const double* a, int64_t n, int64_t* out_index, double* out_value,
                       uint64_t* g_val, uint32_t* g_done, unsigned warp_size, unsigned max_grid,
                       hipStream_t stream) {
  int64_t need = (n + (int64_t)EBC_BLOCK - 1) / (int64_t)EBC_BLOCK;
  unsigned grid = (need < 1) ? 1u : (unsigned)need;
  if (grid > max_grid) grid = max_grid;
  ebc_kernel<<<grid, EBC_BLOCK, 0, stream>>>(a, n, g_val, g_done, out_index, out_value, warp_size);
}
