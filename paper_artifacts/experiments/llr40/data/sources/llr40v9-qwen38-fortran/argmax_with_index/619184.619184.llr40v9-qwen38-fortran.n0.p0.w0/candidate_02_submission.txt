subroutine argmax_with_index_fp64(a, out_index, out_value, len_1d) bind(c, name='argmax_with_index_fp64')
  use, intrinsic :: iso_c_binding
  implicit none
  real(c_double), intent(in), dimension(*) :: a
  integer(c_int64_t), intent(out) :: out_index
  real(c_double), intent(out) :: out_value
  integer(c_int64_t), value, intent(in) :: len_1d
  integer, external :: omp_get_max_threads
  real(c_double) :: x, gmax=-huge(1.0d0), lmax(512)
  integer(c_int64_t) :: i, idx, gidx, lidx(512)
  integer(c_int64_t) :: n, nt, t, start, fin, chunk
  logical :: has(512)
  if (len_1d <= 0) then
    out_value = 0.0d0
    out_index = 0
    return
  end if
  n = len_1d
  nt = omp_get_max_threads()
  if (nt > 512) nt = 512
  if (nt < 1) nt = 1
  chunk = (n + nt - 1) / nt
  !$omp parallel do schedule(static) private(t,start,fin,x,idx)
  do t = 0, nt-1
    start = t * chunk
    fin = min(n, start + chunk)
    if (start < fin) then
      has(t+1) = .true.
      x = a(start+1)
      idx = start+1
      !$novec
      do i = start+2, fin
        if (a(i) > x) then; x = a(i); idx = i; end if
      end do
      lmax(t+1) = x
      lidx(t+1) = idx
    else
      has(t+1) = .false.
    end if
  end do
  !$omp end parallel do
  gidx = 0
  do t = 0, nt-1
    if (has(t+1)) then
      if (gidx == 0) then
        gmax = lmax(t+1); gidx = lidx(t+1)
      else if (lmax(t+1) > gmax) then
        gmax = lmax(t+1); gidx = lidx(t+1)
      end if
    end if
  end do
  out_value = gmax
  out_index = gidx
end subroutine
