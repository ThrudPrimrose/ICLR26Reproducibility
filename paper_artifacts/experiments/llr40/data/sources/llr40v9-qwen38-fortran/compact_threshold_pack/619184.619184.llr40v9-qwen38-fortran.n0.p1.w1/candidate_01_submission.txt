! Stream compaction (parallel, two-pass block scan).
! Pack src[i]*weight[i] for every src[i] > 0, preserve source order, publish count.
! Canonical C-ABI (c-abi-v2) entry point; trailing workspace pair is ignored.
subroutine compact_threshold_pack_fp64(out_count, packed, src, weight, LEN_1D, workspace, workspace_size) bind(C, &
&name="compact_threshold_pack_fp64")
    use, intrinsic :: iso_c_binding
    implicit none
    integer(c_int64_t), value, intent(in) :: LEN_1D
    integer(c_int64_t), intent(inout) :: out_count(1)
    real(c_double), intent(inout) :: packed(LEN_1D)
    real(c_double), intent(in) :: src(LEN_1D)
    real(c_double), intent(in) :: weight(LEN_1D)
    type(c_ptr), value :: workspace
    integer(c_int64_t), value, intent(in) :: workspace_size

    integer(c_int64_t) :: N, nb, b, i, bs, lo, hi, tot, cnt
    integer(c_int64_t), allocatable :: bcnt(:), bbase(:)

    N = LEN_1D

    ! ---- small-N serial path (no OpenMP / heap overhead) ----
    if (N < int(65536, c_int64_t)) then
        tot = 0
        do i = 0, N - 1
            if (src(i + 1) > 0.0d0) then
                packed(tot + 1) = src(i + 1) * weight(i + 1)
                tot = tot + 1
            end if
        end do
        out_count(1) = tot
        return
    end if

    ! ---- large-N two-pass block scan ----
    bs = 2048
    nb = (N + bs - 1) / bs

    allocate(bcnt(nb)); allocate(bbase(nb))

    ! Pass 1: per-block survivor counts.
    bcnt = 0
    !$omp parallel do schedule(static)
    do b = 0, nb - 1
        lo = b * bs
        hi = min((b + 1) * bs, N)
        cnt = 0
        do i = lo, hi - 1
            cnt = cnt + merge(1, 0, src(i + 1) > 0.0d0)
        end do
        bcnt(b + 1) = cnt
    end do
    !$omp end parallel do

    ! Exclusive scan of block counts -> block base offsets (small, serial).
    tot = 0
    do b = 1, nb
        bbase(b) = tot
        tot = tot + bcnt(b)
    end do
    out_count(1) = tot

    ! Pass 2: scatter survivors into packed at base+running.
    !$omp parallel do schedule(static)
    do b = 1, nb
        lo = (b - 1) * bs
        hi = min(b * bs, N)
        cnt = bbase(b)
        do i = lo, hi - 1
            if (src(i + 1) > 0.0d0) then
                packed(cnt + 1) = src(i + 1) * weight(i + 1)
                cnt = cnt + 1
            end if
        end do
    end do
    !$omp end parallel do

    deallocate(bcnt); deallocate(bbase)
end subroutine compact_threshold_pack_fp64
