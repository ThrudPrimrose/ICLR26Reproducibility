"""TSVC tsvc_2 s255 optimized.

a[i] = (b[i] + b[(i-1) % n] + b[(i-2) % n]) * 0.333   (bit-exact reassociation:
the serial recurrence only carries b[i-1] and b[i-2] forward, no dependence on a).
"""
import numpy as np
from numba import njit, prange


@njit(parallel=False, fastmath=False, cache=False)
def _ser(a, b, n):
    x = b[n - 1]
    y = b[n - 2]
    for i in range(n):
        a[i] = (b[i] + x + y) * 0.333
        y = x
        x = b[i]


@njit(parallel=True, fastmath=False, cache=False)
def _par(a, b, n):
    for i in prange(n):
        if i == 0:
            x = b[n - 1]
            y = b[n - 2]
        elif i == 1:
            x = b[0]
            y = b[n - 1]
        else:
            x = b[i - 1]
            y = b[i - 2]
        a[i] = (b[i] + x + y) * 0.333


_PAR_MIN = 1 << 18


def s255(a, b, LEN_1D):
    n = LEN_1D
    if n >= _PAR_MIN:
        _par(a, b, n)
    else:
        _ser(a, b, n)
    return None


# Warm up / compile at import time so the timed call is pure execution.
_n = 4
_b = np.zeros(_n)
_a = np.zeros(_n)
_ser(_a, _b, _n)
_par(_a, _b, _n)
# force the parallel specialization path with an actual multi-thread run
_b2 = np.zeros(_PAR_MIN + 3)
_a2 = np.zeros(_PAR_MIN + 3)
_par(_a2, _b2, _PAR_MIN + 3)
del _a, _b, _a2, _b2
