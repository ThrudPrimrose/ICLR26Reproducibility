"""fused single-pass numba implementation of TSVC fuse_diamond, warm at import."""
import numpy as np
from numba import njit

@njit(nogil=True, cache=False)
def _fuse(a, out, n):
    for i in range(n):
        t = a[i] * a[i]
        out[i] = (t + 1.0) * (t - 1.0)

# warm the JIT at import time so the first timed call is native code
_w = np.ones(8192, dtype=np.float64)
_ww = np.empty(8192, dtype=np.float64)
_fuse(_w, _ww, 8192)
del _w, _ww

def fuse_diamond(out, a, LEN_1D):
    n = int(LEN_1D)
    if a.ndim == 1 and a.flags.c_contiguous and out.flags.c_contiguous:
        _fuse(a, out, n)
    else:
        ac = np.ascontiguousarray(a)
        oc = np.empty_like(ac)
        _fuse(ac, oc, n)
        out[...] = oc
