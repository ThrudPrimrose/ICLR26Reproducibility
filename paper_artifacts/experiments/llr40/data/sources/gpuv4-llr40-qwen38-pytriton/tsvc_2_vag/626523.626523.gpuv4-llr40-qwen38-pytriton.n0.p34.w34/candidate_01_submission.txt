import os
os.environ.setdefault("NUMBA_THREADING_LAYER", "omp")
import numpy as np
from numba import njit, prange

@njit(parallel=True, nogil=True, fastmath=True)
def _vag_par(a, b, ip, n):
    # 4-wide body for memory-level parallelism
    nb4 = n >> 2
    for j in prange(nb4):
        i = j * 4
        a[i] = b[ip[i]]
        a[i+1] = b[ip[i+1]]
        a[i+2] = b[ip[i+2]]
        a[i+3] = b[ip[i+3]]
    for i in range(nb4 << 2, n):
        a[i] = b[ip[i]]

def _warm():
    n = 16
    af = np.zeros(n, dtype=np.float64)
    b32 = np.zeros(n, dtype=np.float32)
    idxs = [np.zeros(n, dtype=d) for d in
            (np.int8, np.int16, np.int32, np.int64, np.uint8, np.uint16, np.uint32)]
    for idx in idxs:
        _vag_par(af, af, idx, n)
    for idx in (np.zeros(n, dtype=np.int32), np.zeros(n, dtype=np.int64)):
        _vag_par(b32, b32, idx, n)

_warm()

def vag(a, b, ip, LEN_1D):
    _vag_par(a, b, ip, LEN_1D)
    return None
