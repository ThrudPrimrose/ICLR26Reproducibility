"""Optimized TSVC tsvc_2 s2275 (python arm).

    aa[j,i] = aa[j,i] + bb[j,i]*cc[j,i]   for all j,i in [0,n)
    a[i]    = b[i] + c[i]*d[i]            for all i in [0,n)

Both are pure elementwise ops on the flat memory. Implemented as a single
Numba kernel with flat 1D indexing (numba's 2D fancy-index codegen is
pathologically slow on this target) and a prange over the flat index so all
cores stream the arrays. The kernel is compiled at import time (untimed) so
the first measured call hits the cached dispatcher.

In-place ABI: outputs written into the handed buffers, returns None.
"""
import numpy as np
from numba import njit, prange


@njit(parallel=True)
def _kern(aa, bb, cc, a, b, c, d, M):
    for idx in prange(M):
        aa[idx] = aa[idx] + bb[idx] * cc[idx]
    for i in range(a.shape[0]):
        a[i] = b[i] + c[i] * d[i]


def s2275(a, b, c, d, aa, bb, cc, LEN_2D):
    n = int(LEN_2D)
    if n <= 0:
        return None
    if (aa.flags.c_contiguous and bb.flags.c_contiguous and cc.flags.c_contiguous
            and a.flags.c_contiguous and b.flags.c_contiguous
            and c.flags.c_contiguous and d.flags.c_contiguous):
        _kern(aa.ravel(), bb.ravel(), cc.ravel(), a, b, c, d, n * n)
    else:
        # generic fallback: arbitrary strides (rare; correctness first)
        t = np.ascontiguousarray(aa)
        t += np.ascontiguousarray(bb) * np.ascontiguousarray(cc)
        aa[...] = t
        a[...] = np.asarray(b) + np.asarray(c) * np.asarray(d)
    return None


# ---------------- import-time warmup (untimed) ----------------
try:
    _w1 = np.random.rand(64)
    _w2 = np.random.rand(64)
    _w3 = np.random.rand(64)
    _kern(_w1.copy(), _w1, _w2, _w1, _w1, _w2, _w3, 4096)
    del _w1, _w2, _w3
except Exception:
    # if warmup fails the first real call will just compile (still correct)
    pass
