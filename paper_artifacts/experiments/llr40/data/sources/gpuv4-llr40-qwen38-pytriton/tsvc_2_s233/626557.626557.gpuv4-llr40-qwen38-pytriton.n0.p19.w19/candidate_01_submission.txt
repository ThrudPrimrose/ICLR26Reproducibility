"""TSVC tsvc_2 s233 -- optimized Python submission.

Semantics (reference):
    for i in 8..n-1:
        for j in 8..n-1:  aa[j,i] = aa[j-1,i] + cc[j,i]   (scan down column i of aa)
        for j in 8..n-1:  bb[j,i] = bb[j,i-1] + cc[j,i]   (scan across row j of bb)

The two updates are independent of each other; each is a set of
embarrassingly-parallel prefix scans (one per column for aa, one per
row for bb).  We run both as Numba parallel regions over the scan
index, keeping the reference's exact sequential addition order inside
every scan so the result is bit-for-bit identical to the reference.
"""
import numpy as np
import numba as nb

# ---------------------------------------------------------------------------
# kernels: one parallel region per scan family, bit-exact FP order
# ---------------------------------------------------------------------------

@nb.njit(parallel=True, boundscheck=False, fastmath=False)
def _aa_kernel(aa, cc, n):
    # independent scans down each column i of aa, seeded by row 7
    for i in nb.prange(8, n):
        v = aa[7, i]
        for j in range(8, n):
            v += cc[j, i]
            aa[j, i] = v


@nb.njit(parallel=True, boundscheck=False, fastmath=False)
def _bb_kernel(bb, cc, n):
    # independent scans across each row j of bb, seeded by column 7
    for j in nb.prange(8, n):
        v = bb[j, 7]
        for i in range(8, n):
            v += cc[j, i]
            bb[j, i] = v


# warm the JIT + thread pool at import time (not charged to the timed call)
_w = np.zeros((32, 32))
_aa_kernel(_w, _w, 32)
_bb_kernel(_w, _w, 32)


def s233(aa, bb, cc, LEN_2D):
    n = int(LEN_2D)
    if n <= 8:
        return None
    if aa.flags.c_contiguous and bb.flags.c_contiguous and cc.flags.c_contiguous:
        _aa_kernel(aa, cc, n)
        _bb_kernel(bb, cc, n)
        return None
    # non-contiguous fallback: compute on contiguous copies, then copy back
    aac = np.ascontiguousarray(aa)
    bbc = np.ascontiguousarray(bb)
    ccc = np.ascontiguousarray(cc)
    _aa_kernel(aac, ccc, n)
    _bb_kernel(bbc, ccc, n)
    aa[...] = aac
    bb[...] = bbc
    return None
