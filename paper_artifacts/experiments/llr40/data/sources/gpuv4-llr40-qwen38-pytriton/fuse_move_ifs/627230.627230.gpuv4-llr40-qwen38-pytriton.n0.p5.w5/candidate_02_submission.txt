import numpy as np
import numba as nb
from numba import prange

@nb.njit(parallel=True, fastmath=True)
def _impl(a, b, src, cond, LEN_2D, K):
    for i in prange(LEN_2D):
        if cond[i] > 0.0:
            for j in range(LEN_2D):
                a[i, j] = src[i, j] * 2.0
        if K > 0:
            for j in range(LEN_2D):
                b[i, j] = src[i, j] + 1.0

def fuse_move_ifs(a, b, src, cond, LEN_2D, K):
    _impl(a, b, src, cond, int(LEN_2D), int(K))
    return None

# warm up at import time (before the clock starts): compiles the jit + warms thread pool
_wa = np.zeros((8, 8), np.float64)
_wb = np.zeros((8, 8), np.float64)
_ws = np.ones((8, 8), np.float64)
_wc = np.zeros(8, np.float64)
for _k in (1, -1):
    _impl(_wa, _wb, _ws, _wc, 8, _k)
