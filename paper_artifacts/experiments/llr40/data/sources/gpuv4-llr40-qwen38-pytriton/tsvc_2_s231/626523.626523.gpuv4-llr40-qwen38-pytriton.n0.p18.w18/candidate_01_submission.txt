import os
import numpy as np
import numba

# Match the usable core count (affinity) so prange uses them all.
try:
    _nt = len(os.sched_getaffinity(0))
except Exception:
    _nt = os.cpu_count() or 1
_nt = max(1, _nt)
try:
    numba.set_num_threads(_nt)
except Exception:
    pass


@numba.njit(parallel=True)
def _chunks(aa, bb, N, W):
    nch = (N + W - 1) // W
    for c in numba.prange(nch):
        i0 = c * W
        i1 = i0 + W
        if i1 > N:
            i1 = N
        for j in range(1, N):
            for i in range(i0, i1):
                aa[j, i] = aa[j - 1, i] + bb[j, i]


# Warm the JIT at import so no compile lands in the timed section.
_d = np.zeros((8, 8))
_chunks(_d.copy(), _d, 8, 4)


def s231(aa, bb, LEN_2D):
    N = int(LEN_2D)
    # ~20 column-strips is the measured sweet spot (good contiguity + thread use).
    W = (N + 19) // 20
    if W < 8:
        W = 8
    _chunks(aa, bb, N, W)
    return None
