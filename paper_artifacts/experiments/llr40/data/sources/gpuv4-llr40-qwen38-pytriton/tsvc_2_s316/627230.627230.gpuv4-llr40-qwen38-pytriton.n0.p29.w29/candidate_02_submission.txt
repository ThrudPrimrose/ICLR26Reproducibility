# TSVC s316: result[0] = min(a[0:LEN_1D])  (fp64 array minimum)
# Parallel numba reduction, warmed at import time so no compile is timed.
import numpy as np
import numba
from numba import prange


@numba.njit(parallel=True, fastmath=False)
def _par_min(a, n):
    m = np.inf
    for i in prange(n):
        m = min(m, a[i])
    return m


def _warm():
    # Compile the exact signatures before the clock starts.
    for dt in (np.float64, np.float32):
        w = np.ones(1 << 20, dtype=dt)
        r = _par_min(w, w.shape[0])
        if r != 1.0:
            raise RuntimeError("warmup failed")


_warm()


def s316(a, result, LEN_1D):
    n = LEN_1D
    if a.dtype in (np.float64, np.float32) and a.ndim == 1 and a.flags.c_contiguous:
        result[0] = _par_min(a, n)
    else:
        result[0] = np.min(a[:n])
    return None
