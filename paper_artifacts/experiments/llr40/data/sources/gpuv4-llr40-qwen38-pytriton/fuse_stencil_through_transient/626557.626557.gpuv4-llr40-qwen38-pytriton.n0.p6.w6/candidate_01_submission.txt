"""TSVC fuse_stencil_through_transient -- fused single-pass stencil.

out[i] = (a[i-1] + a[i] + a[i+1]) * (a[i] + a[i+1] + a[i+2])  for 1 <= i < n-2

Fused through the transient: one pass, 4 reads of a + 1 write of out.
Parallel over all elements; operations in the exact reference order
(bit-identical results). Numba-compiled; warmed at import time so the
timed call is pure execution.
"""
import numpy as np
from numba import njit, prange


@njit(parallel=True, fastmath=False, cache=False)
def _fused(out, a, n):
    for i in prange(1, n - 2):
        out[i] = (a[i - 1] + a[i] + a[i + 1]) * (a[i] + a[i + 1] + a[i + 2])


def fuse_stencil_through_transient(out, a, LEN_1D):
    n = int(LEN_1D)
    if n > 3:
        _fused(out, a, n)
    return None


# Warm up / compile at import time (untimed) so the first timed call is fast.
def _warm():
    m = 1 << 16
    _x = np.zeros(m)
    _a = np.random.rand(m)
    _fused(_x, _a, m)
    _fused(_x, _a, 4)  # small-size path
    del _x, _a


_warm()
