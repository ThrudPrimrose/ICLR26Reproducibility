# TSVC tsvc_2 s152:  b = d*e ;  a += b*c   (in-place)
# Strategy: OpenMP C extension (auto-vectorized, compiled at import) with
# a warmed numba parallel fallback.
import ctypes
import os
import subprocess
import tempfile

import numpy as np
import numba
from numba import njit, prange

_C = r'''
#include <stdint.h>
void s152_onepass(double *a, double *b, const double *c, const double *d, const double *e, int64_t n, int64_t chunk) {
#pragma omp parallel for schedule(dynamic, chunk)
  for (int64_t i = 0; i < n; ++i) {
    double t = d[i] * e[i];
    b[i] = t;
    a[i] += t * c[i];
  }
}
void s152_twopass(double *a, double *b, const double *c, const double *d, const double *e, int64_t n, int64_t chunk) {
#pragma omp parallel for schedule(dynamic, chunk)
  for (int64_t i = 0; i < n; ++i) b[i] = d[i] * e[i];
#pragma omp parallel for schedule(dynamic, chunk)
  for (int64_t i = 0; i < n; ++i) a[i] += b[i] * c[i];
}
'''

def _build():
    d = tempfile.mkdtemp(prefix="s152c_")
    src = os.path.join(d, "k.c")
    so = os.path.join(d, "k.so")
    with open(src, "w") as f:
        f.write(_C)
    r = subprocess.run(["gcc", "-O3", "-march=native", "-ffp-contract=off",
                        "-fopenmp", "-shared", "-fPIC", "-o", so, src],
                       capture_output=True, text=True, timeout=180)
    if r.returncode != 0:
        raise RuntimeError(r.stderr[-2000:])
    lib = ctypes.CDLL(so)
    fns = {}
    for name in ("s152_onepass", "s152_twopass"):
        fn = getattr(lib, name)
        fn.argtypes = [ctypes.c_void_p] * 5 + [ctypes.c_int64, ctypes.c_int64]
        fn.restype = None
        fns[name] = fn
    return fns

try:
    _C_FNS = _build()
    _USE_C = True
except Exception:
    _C_FNS = None
    _USE_C = False

@njit(parallel=True)
def _fused_par(a, b, c, d, e, n):
    for i in prange(n):
        b[i] = d[i] * e[i]
    for i in prange(n):
        a[i] += b[i] * c[i]

@njit()
def _fused_ser(a, b, c, d, e, n):
    for i in range(n):
        b[i] = d[i] * e[i]
    for i in range(n):
        a[i] += b[i] * c[i]

_TINY = 65536
_MODE = "s152_onepass"
_CHUNK = 262144

def s152(a, b, c, d, e, LEN_1D):
    n = int(LEN_1D)
    if n == 0:
        return None
    if _USE_C and n > _TINY:
        _C_FNS[_MODE](a.ctypes.data, b.ctypes.data, c.ctypes.data,
                      d.ctypes.data, e.ctypes.data,
                      ctypes.c_int64(n), ctypes.c_int64(_CHUNK))
    elif n <= _TINY:
        _fused_ser(a, b, c, d, e, n)
    else:
        _fused_par(a, b, c, d, e, n)
    return None

def _warm():
    n = 1 << 20
    a = np.zeros(n); b = np.zeros(n); c = np.zeros(n)
    d = np.zeros(n); e = np.zeros(n)
    _fused_ser(a, b, c, d, e, n)
    _fused_par(a, b, c, d, e, n)
    if _USE_C:
        for name in _C_FNS:
            _C_FNS[name](a.ctypes.data, b.ctypes.data, c.ctypes.data,
                         d.ctypes.data, e.ctypes.data,
                         ctypes.c_int64(n), ctypes.c_int64(_CHUNK))

_warm()
