"""TSVC tsvc_2 s252, optimized.

Reference semantics:
    t = 0.0
    for i in range(LEN_1D):
        s = b[i] * c[i]
        a[i] = s + t
        t = s

Since t is always the *previous* product, the loop unrolls to a pure
elementwise shifted op with no cross-iteration dependence:

    a[i] = b[i]*c[i] + (b[i-1]*c[i-1] if i > 0 else 0.0)

The kernel computes exactly that, bit-for-bit (same two multiplies and one
add per element, in the reference's order).  It is a single memory pass
(read b, c; write a = 24 bytes/element, the information-theoretic minimum)
on numba's OpenMP pool.

Threads: the container is affined to a small set of CPUs, but more
in-flight loads beat fewer at the memory wall -- measured read bandwidth
scales from ~10 GB/s (1 thread) to ~38 GB/s (8+ threads) on this node, so
we oversubscribe deliberately.
"""
import os

import numpy as np

_phys = (len(os.sched_getaffinity(0)) if hasattr(os, "sched_getaffinity")
         else (os.cpu_count() or 1))
_NTHREADS = 8 if _phys >= 3 else max(1, _phys)

os.environ["NUMBA_NUM_THREADS"] = str(_NTHREADS)
os.environ.setdefault("OMP_NUM_THREADS", str(_NTHREADS))
os.environ.setdefault("GOMP_NUM_THREADS", str(_NTHREADS))

import numba  # noqa: E402


@numba.njit(parallel=True, nogil=True)
def _s252_core(a, b, c, n):
    if n <= 0:
        return
    a[0] = b[0] * c[0]
    for i in numba.prange(1, n):
        a[i] = b[i] * c[i] + b[i - 1] * c[i - 1]


def s252(a, b, c, LEN_1D):
    """In-place: write outputs into `a` (C ABI, return None)."""
    n = int(LEN_1D)
    if n <= 0:
        return None
    if (a.dtype != np.float64 or b.dtype != np.float64 or c.dtype != np.float64
            or not (a.flags.c_contiguous and b.flags.c_contiguous
                    and c.flags.c_contiguous)):
        # Fallback for any non-canonical input: exact reference semantics.
        a64 = np.ascontiguousarray(a, dtype=np.float64)[:n]
        b64 = np.ascontiguousarray(b, dtype=np.float64)[:n]
        c64 = np.ascontiguousarray(c, dtype=np.float64)[:n]
        _s252_core(a64, b64, c64, n)
        a[:n] = a64
        return None
    if n < a.shape[0]:
        a = a[:n]
        b = b[:n]
        c = c[:n]
    _s252_core(a, b, c, n)
    return None


# Warm the JIT before the timer starts: compile the exact signature used at
# run time (1-D C-contiguous float64) on a tiny input.  Shape changes at run
# time do not recompile; only dtype/layout changes would.
_w = np.arange(256, dtype=np.float64)
_s252_core(_w, _w + 1.0, _w + 2.0, 256)
del _w
