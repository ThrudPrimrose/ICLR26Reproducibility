import numpy as np
from numba import njit, prange


@njit(parallel=True, fastmath=False, cache=False)
def _s2710(a, b, c, d, e, x, LEN_1D):
    big = LEN_1D > 10
    pos = x[0] > 0.0
    for i in prange(LEN_1D):
        if a[i] > b[i]:
            a[i] = a[i] + b[i] * d[i]
            if big:
                c[i] = c[i] + d[i] * d[i]
            else:
                c[i] = d[i] * e[i] + 1.0
        else:
            b[i] = a[i] + e[i] * e[i]
            if pos:
                c[i] = a[i] + d[i] * d[i]
            else:
                c[i] = c[i] + e[i] * e[i]


def s2710(a, b, c, d, e, x, LEN_1D):
    _s2710(a, b, c, d, e, x, LEN_1D)


# Import-time warmup: the JIT compile happens before the timed section.
_w = np.zeros(256)
_wx = np.zeros(1)
_s2710(_w.copy(), _w.copy(), _w.copy(), _w, _w.copy(), _wx, 256)
