subroutine ext_break_capture_fp64(a, out_index, out_value, LEN_1D, workspace, workspace_size) bind(C)
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: LEN_1D
  integer(c_int8_t), intent(inout) :: workspace(*)
  integer(c_int64_t), value, intent(in) :: workspace_size
  real(c_double), intent(in) :: a(LEN_1D)
  integer(c_int64_t), intent(out) :: out_index(1)
  real(c_double), intent(out) :: out_value(1)

  real(c_double), parameter :: K = 1.0d0
  integer(c_int64_t) :: i, first

  out_index(1) = 0
  out_value(1) = -1.0d0

  if (LEN_1D <= 0) return

  first = LEN_1D + 1
  !$omp parallel do simd reduction(min:first) schedule(static)
  do i = 1, LEN_1D
    if (a(i) > K) first = min(first, i)
  end do

  if (first <= LEN_1D) then
    out_index(1) = first
    out_value(1) = a(first)
  end if
end subroutine ext_break_capture_fp64
