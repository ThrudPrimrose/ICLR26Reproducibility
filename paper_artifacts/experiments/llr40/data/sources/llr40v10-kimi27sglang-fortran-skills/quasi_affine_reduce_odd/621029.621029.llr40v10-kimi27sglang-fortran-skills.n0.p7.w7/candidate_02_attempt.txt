subroutine quasi_affine_reduce_odd_fp64(a, out, LEN_1D, workspace, workspace_size) bind(C)
  use iso_c_binding
  implicit none
  integer(c_int64_t), value, intent(in) :: LEN_1D
  integer(c_int64_t), value, intent(in) :: workspace_size
  real(c_double), intent(in) :: a(LEN_1D)
  real(c_double), intent(out) :: out
  integer(c_int8_t), intent(inout) :: workspace(workspace_size)

  real(c_double) :: acc0, acc1, acc2, acc3, acc4, acc5, acc6, acc7
  integer(c_int64_t) :: i

  acc0 = 0.0d0; acc1 = 0.0d0; acc2 = 0.0d0; acc3 = 0.0d0
  acc4 = 0.0d0; acc5 = 0.0d0; acc6 = 0.0d0; acc7 = 0.0d0

  !$omp parallel do reduction(+:acc0,acc1,acc2,acc3,acc4,acc5,acc6,acc7) schedule(static)
  do i = 2, LEN_1D - 14, 16
    acc0 = acc0 + a(i)
    acc1 = acc1 + a(i + 2)
    acc2 = acc2 + a(i + 4)
    acc3 = acc3 + a(i + 6)
    acc4 = acc4 + a(i + 8)
    acc5 = acc5 + a(i + 10)
    acc6 = acc6 + a(i + 12)
    acc7 = acc7 + a(i + 14)
  end do
  !$omp end parallel do

  ! remainder
  do i = ((LEN_1D - 2) / 16) * 16 + 2, LEN_1D, 2
    acc0 = acc0 + a(i)
  end do

  out = acc0 + acc1 + acc2 + acc3 + acc4 + acc5 + acc6 + acc7
end subroutine quasi_affine_reduce_odd_fp64
