module tsvc_2_s115_mod
  use iso_c_binding, only: c_double, c_int64_t
  implicit none
contains
  subroutine tsvc_2_s115_fp64(a, aa, n) bind(c, name="tsvc_2_s115_fp64")
    integer(c_int64_t), value :: n
    real(c_double), intent(inout) :: a(n)
    real(c_double), intent(in) :: aa(n*n)
    integer(c_int64_t) :: i, j
    real(c_double) :: aj
    do j = 1, n - 1
      aj = a(j)
      !$omp simd
      do i = j + 1, n
        a(i) = a(i) - aa((j - 1) * n + i) * aj
      end do
    end do
  end subroutine tsvc_2_s115_fp64
end module tsvc_2_s115_mod
