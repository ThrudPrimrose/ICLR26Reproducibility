#include <stdint.h>
#include <omp.h>

void tsvc_2_s231_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
  #pragma omp parallel
  {
    const int64_t nthreads = omp_get_num_threads();
    const int64_t tid = omp_get_thread_num();
    const int64_t i_start = (LEN_2D * tid) / nthreads;
    const int64_t i_end = (LEN_2D * (tid + 1)) / nthreads;

    for (int64_t j = 1; j < LEN_2D; ++j) {
      #pragma omp simd
      for (int64_t i = i_start; i < i_end; ++i) {
        aa[j * LEN_2D + i] = aa[(j - 1) * LEN_2D + i] + bb[j * LEN_2D + i];
      }
    }
  }
}
