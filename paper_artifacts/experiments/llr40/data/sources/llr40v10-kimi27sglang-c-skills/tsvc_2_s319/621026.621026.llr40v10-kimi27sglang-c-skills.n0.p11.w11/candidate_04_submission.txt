#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

void tsvc_2_s319_fp64(double *restrict a, double *restrict b, const double *restrict c,
                      const double *restrict d, const double *restrict e,
                      const int64_t LEN_1D) {
  double sum = 0.0;

#pragma omp parallel
  {
    const int nt = omp_get_num_threads();
    const int tid = omp_get_thread_num();
    const int64_t chunk = (LEN_1D + nt - 1) / nt;
    int64_t lo = tid * chunk;
    int64_t hi = (lo + chunk < LEN_1D) ? lo + chunk : LEN_1D;

    __m512d vsum = _mm512_setzero_pd();
    double partial = 0.0;

    while (lo < hi && (((uintptr_t)&a[lo]) & 63) != 0) {
      const double ci = c[lo];
      const double ai = ci + d[lo];
      const double bi = ci + e[lo];
      a[lo] = ai;
      b[lo] = bi;
      partial += ai + bi;
      ++lo;
    }

    const int64_t vec_end = lo + ((hi - lo) & ~7);
    for (int64_t i = lo; i < vec_end; i += 8) {
      const __m512d vc = _mm512_loadu_pd(&c[i]);
      const __m512d vd = _mm512_loadu_pd(&d[i]);
      const __m512d ve = _mm512_loadu_pd(&e[i]);
      const __m512d va = _mm512_add_pd(vc, vd);
      const __m512d vb = _mm512_add_pd(vc, ve);
      _mm512_stream_pd(&a[i], va);
      _mm512_storeu_pd(&b[i], vb);
      vsum = _mm512_add_pd(vsum, _mm512_add_pd(va, vb));
    }

    for (int64_t i = vec_end; i < hi; ++i) {
      const double ci = c[i];
      const double ai = ci + d[i];
      const double bi = ci + e[i];
      a[i] = ai;
      b[i] = bi;
      partial += ai + bi;
    }

    const double thread_sum = partial + _mm512_reduce_add_pd(vsum);
#pragma omp atomic
    sum += thread_sum;
  }

  b[0] = sum;
}
