#include <stdint.h>
#include <stdlib.h>
#include <omp.h>

void tsvc_2_s275_fp64(double *restrict aa, const double *restrict bb, const double *restrict cc, const int64_t LEN_2D) {
    const int64_t n = LEN_2D;

    int64_t *restrict act = malloc((size_t)n * sizeof(int64_t));
    double *restrict init = malloc((size_t)n * sizeof(double));
    int64_t na = 0;
    for (int64_t i = 0; i < n; i++) {
        if (aa[i] > 0.0) {
            act[na] = i;
            init[na] = aa[i];
            na++;
        }
    }

    if (na == 0) {
        free(act);
        free(init);
        return;
    }

    if (na * 20 < n) {
        #pragma omp parallel for schedule(static)
        for (int64_t k = 0; k < na; k++) {
            const int64_t i = act[k];
            double prev = init[k];
            for (int64_t j = 1; j < n; j++) {
                const int64_t idx = j * n + i;
                prev = prev + bb[idx] * cc[idx];
                aa[idx] = prev;
            }
        }
    } else {
        #pragma omp parallel
        {
            const int nt = omp_get_num_threads();
            const int tid = omp_get_thread_num();
            const int64_t a0 = (na * tid) / nt;
            const int64_t a1 = (na * (tid + 1)) / nt;
            const int64_t cnt = a1 - a0;
            if (cnt > 0) {
                double prev[cnt];
                for (int64_t k = 0; k < cnt; k++) {
                    prev[k] = init[a0 + k];
                }
                for (int64_t j = 1; j < n; j++) {
                    const int64_t row = j * n;
                    const double *restrict brow = bb + row;
                    const double *restrict crow = cc + row;
                    double *restrict arow = aa + row;
                    for (int64_t k = 0; k < cnt; k++) {
                        const int64_t i = act[a0 + k];
                        const double nv = prev[k] + brow[i] * crow[i];
                        arow[i] = nv;
                        prev[k] = nv;
                    }
                }
            }
        }
    }

    free(act);
    free(init);
}
