#include <stdint.h>
#include <math.h>
#include <float.h>
#include <omp.h>

typedef struct {
    double maxv;
    int64_t idx;
} maxloc_t;

#pragma omp declare reduction(maxloc : maxloc_t : \
    omp_out = (omp_in.maxv > omp_out.maxv || (omp_in.maxv == omp_out.maxv && omp_in.idx < omp_out.idx)) ? omp_in : omp_out) \
    initializer(omp_priv = {-DBL_MAX, INT64_MAX})

void tsvc_2_s3110_fp64(const double *restrict aa, double *restrict bb, const int64_t LEN_2D) {
    if (LEN_2D <= 0) {
        bb[0] = 0.0;
        return;
    }

    maxloc_t res = {-DBL_MAX, INT64_MAX};

    #pragma omp parallel for collapse(2) reduction(maxloc:res) schedule(static)
    for (int64_t i = 0; i < LEN_2D; ++i) {
        for (int64_t j = 0; j < LEN_2D; ++j) {
            const int64_t k = i * LEN_2D + j;
            const double v = aa[k];
            maxloc_t cur = {v, k};
            if (cur.maxv > res.maxv || (cur.maxv == res.maxv && cur.idx < res.idx)) {
                res = cur;
            }
        }
    }

    const int64_t xindex = res.idx / LEN_2D;
    const int64_t yindex = res.idx % LEN_2D;
    bb[0] = res.maxv + (double)xindex + (double)yindex;
}
