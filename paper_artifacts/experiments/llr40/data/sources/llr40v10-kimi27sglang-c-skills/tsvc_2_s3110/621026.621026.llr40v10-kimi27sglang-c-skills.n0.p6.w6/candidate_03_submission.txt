#include <stdint.h>
#include <float.h>
#include <immintrin.h>
#include <omp.h>

void tsvc_2_s3110_fp64(const double *restrict aa, double *restrict bb, const int64_t LEN_2D) {
    if (LEN_2D <= 0) {
        bb[0] = 0.0;
        return;
    }

    const int64_t n = LEN_2D;
    double gmaxv = aa[0];
    int64_t gidx = 0;

    #pragma omp parallel
    {
        const int nt = omp_get_num_threads();
        const int tid = omp_get_thread_num();
        const int64_t row0 = (n * tid) / nt;
        const int64_t row1 = (n * (tid + 1)) / nt;

        const __m512d negmax = _mm512_set1_pd(-DBL_MAX);
        const __m512i badidx = _mm512_set1_epi64(INT64_MAX);
        __m512d vmax = negmax;
        __m512i vidx = badidx;

        for (int64_t i = row0; i < row1; ++i) {
            const double *restrict row = aa + i * n;
            const int64_t base = i * n;
            int64_t j = 0;
            __m512i cur_idx = _mm512_set_epi64(
                base + 7, base + 6, base + 5, base + 4,
                base + 3, base + 2, base + 1, base + 0);
            for (; j + 8 <= n; j += 8) {
                const __m512d v = _mm512_loadu_pd(row + j);
                const __mmask8 gt = _mm512_cmp_pd_mask(v, vmax, _CMP_GT_OQ);
                vmax = _mm512_mask_blend_pd(gt, vmax, v);
                vidx = _mm512_mask_blend_epi64(gt, vidx, cur_idx);
                cur_idx = _mm512_add_epi64(cur_idx, _mm512_set1_epi64(8));
            }
            if (j < n) {
                const __mmask8 mask = (__mmask8)((1u << (uint32_t)(n - j)) - 1u);
                const __m512d v = _mm512_mask_loadu_pd(negmax, mask, row + j);
                const __mmask8 gt = _mm512_cmp_pd_mask(v, vmax, _CMP_GT_OQ);
                vmax = _mm512_mask_blend_pd(gt, vmax, v);
                vidx = _mm512_mask_blend_epi64(gt, vidx, cur_idx);
            }
        }

        alignas(64) double maxv_buf[8];
        alignas(64) int64_t idx_buf[8];
        _mm512_storeu_pd(maxv_buf, vmax);
        _mm512_storeu_si512((__m512i *)idx_buf, vidx);

        double lmaxv = -DBL_MAX;
        int64_t lidx = INT64_MAX;
        for (int p = 0; p < 8; ++p) {
            const double v = maxv_buf[p];
            const int64_t k = idx_buf[p];
            if (v > lmaxv || (v == lmaxv && k < lidx)) {
                lmaxv = v;
                lidx = k;
            }
        }

        #pragma omp critical
        {
            if (lmaxv > gmaxv || (lmaxv == gmaxv && lidx < gidx)) {
                gmaxv = lmaxv;
                gidx = lidx;
            }
        }
    }

    const int64_t xindex = gidx / n;
    const int64_t yindex = gidx % n;
    bb[0] = gmaxv + (double)xindex + (double)yindex;
}
