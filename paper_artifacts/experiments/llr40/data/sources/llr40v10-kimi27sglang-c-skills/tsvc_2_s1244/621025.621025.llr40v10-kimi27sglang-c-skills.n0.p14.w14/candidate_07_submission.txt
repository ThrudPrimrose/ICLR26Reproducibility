#include <stdint.h>
#include <stdlib.h>
#include <omp.h>

void tsvc_2_s1244_fp64(double *restrict a, const double *restrict b, const double *restrict c, double *restrict d,
                       const int64_t LEN_1D) {
    if (LEN_1D <= 1) {
        return;
    }

    // Small inputs stay serial to avoid parallel overhead.
    if (LEN_1D < 4096) {
        #pragma omp simd
        for (int64_t i = 0; i < LEN_1D - 1; i++) {
            a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
            d[i] = a[i] + a[i + 1];
        }
        return;
    }

    int T = omp_get_max_threads();
    double *restrict boundary = (double *restrict)malloc((size_t)T * sizeof(double));
    if (boundary == nullptr) {
        #pragma omp simd
        for (int64_t i = 0; i < LEN_1D - 1; i++) {
            a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
            d[i] = a[i] + a[i + 1];
        }
        return;
    }

    #pragma omp parallel
    {
        const int t = omp_get_thread_num();
        const int nt = omp_get_num_threads();
        const int64_t niter = LEN_1D - 1;
        const int64_t start = (int64_t)t * niter / nt;
        const int64_t end = (int64_t)(t + 1) * niter / nt;

        // Save the one element that this thread reads from the next thread's territory.
        boundary[t] = a[end];

        #pragma omp barrier

        if (end > start) {
            const int64_t last = end - 1;

            #pragma omp simd
            for (int64_t i = start; i < last; i++) {
                a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
                d[i] = a[i] + a[i + 1];
            }

            a[last] = b[last] + c[last] * c[last] + b[last] * b[last] + c[last];
            d[last] = a[last] + boundary[t];
        }
    }

    free(boundary);
}
