#include <stdint.h>
#include <stdlib.h>
#include <omp.h>

void tsvc_2_s1244_fp64(double *restrict a, const double *restrict b, const double *restrict c, double *restrict d,
                       const int64_t LEN_1D) {
    if (LEN_1D <= 1) {
        for (int64_t i = 0; i < LEN_1D - 1; i++) {
            a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
            d[i] = a[i] + a[i + 1];
        }
        return;
    }

    size_t bytes = ((size_t)LEN_1D * sizeof(double) + 63u) & ~63u;
    double *restrict tmp = (double *restrict)aligned_alloc(64, bytes);
    if (tmp == nullptr) {
        for (int64_t i = 0; i < LEN_1D - 1; i++) {
            a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
            d[i] = a[i] + a[i + 1];
        }
        return;
    }

    #pragma omp parallel
    {
        #pragma omp for schedule(static)
        for (int64_t i = 0; i < LEN_1D; i++) {
            tmp[i] = a[i];
        }

        #pragma omp for simd schedule(static)
        for (int64_t i = 0; i < LEN_1D - 1; i++) {
            a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
            d[i] = a[i] + tmp[i + 1];
        }
    }

    free(tmp);
}
