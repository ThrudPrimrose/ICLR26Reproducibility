#include <stdint.h>
#include <stdlib.h>
#include <omp.h>

void tsvc_2_s1244_fp64(double *restrict a, const double *restrict b, const double *restrict c, double *restrict d,
                       const int64_t LEN_1D) {
    if (LEN_1D <= 1) {
        return;
    }

    constexpr int64_t CHUNK = 16384;

    // Small inputs: one chunk, no boundary buffer needed.
    if (LEN_1D <= CHUNK) {
        #pragma omp simd
        for (int64_t i = 0; i < LEN_1D - 1; i++) {
            a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
            d[i] = a[i] + a[i + 1];
        }
        return;
    }

    const int64_t num_chunks = (LEN_1D + CHUNK - 1) / CHUNK;
    double *restrict boundary = (double *restrict)malloc(((size_t)num_chunks - 1u) * sizeof(double));
    if (boundary == nullptr) {
        #pragma omp simd
        for (int64_t i = 0; i < LEN_1D - 1; i++) {
            a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
            d[i] = a[i] + a[i + 1];
        }
        return;
    }

    #pragma omp parallel
    {
        // Save the one element per chunk boundary that the next chunk needs original.
        #pragma omp for schedule(static)
        for (int64_t k = 0; k < num_chunks - 1; k++) {
            boundary[k] = a[(k + 1) * CHUNK];
        }

        // Compute each chunk left-to-right.  Within a chunk a[i+1] is still original
        // because iteration i reads it before iteration i+1 writes it.
        #pragma omp for schedule(static)
        for (int64_t k = 0; k < num_chunks; k++) {
            const int64_t start = k * CHUNK;
            int64_t end = start + CHUNK;
            if (end > LEN_1D) {
                end = LEN_1D;
            }

            // Internal iterations: i+1 stays inside the chunk.
            const int64_t iend = (end - 1 < LEN_1D - 1) ? end - 1 : LEN_1D - 1;
            #pragma omp simd
            for (int64_t i = start; i < iend; i++) {
                a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
                d[i] = a[i] + a[i + 1];
            }

            // The chunk's last iteration needs the saved original a[end].
            if (end < LEN_1D) {
                const int64_t i = end - 1;
                a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
                d[i] = a[i] + boundary[k];
            }
        }
    }

    free(boundary);
}
