#include <stdint.h>
#include <omp.h>

void tsvc_2_s1244_fp64(double *restrict a, const double *restrict b, const double *restrict c, double *restrict d,
                       const int64_t LEN_1D, double *restrict workspace) {
    // Copy original a into scratch so the a[i+1] read uses pre-loop values.
    #pragma omp parallel for simd schedule(static)
    for (int64_t i = 0; i < LEN_1D; i++) {
        workspace[i] = a[i];
    }

    // Parallel, vectorizable loop with no loop-carried dependence.
    #pragma omp parallel for simd schedule(static)
    for (int64_t i = 0; i < LEN_1D - 1; i++) {
        a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
        d[i] = a[i] + workspace[i + 1];
    }
}
