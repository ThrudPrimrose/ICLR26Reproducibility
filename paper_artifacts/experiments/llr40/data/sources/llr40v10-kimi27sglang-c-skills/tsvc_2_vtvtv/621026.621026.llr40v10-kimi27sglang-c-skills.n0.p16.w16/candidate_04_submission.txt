#include <stdint.h>
#include <omp.h>
#include <immintrin.h>

void tsvc_2_vtvtv_fp64(double *restrict a, const double *restrict b, const double *restrict c, const int64_t LEN_1D) {
    #pragma omp parallel
    {
        const int64_t nt = omp_get_num_threads();
        const int64_t tid = omp_get_thread_num();
        const int64_t chunk = (LEN_1D + nt - 1) / nt;
        int64_t i = tid * chunk;
        const int64_t end = (i + chunk < LEN_1D) ? (i + chunk) : LEN_1D;

        while (i < end && ((uintptr_t)(a + i) & 31)) {
            a[i] = a[i] * b[i] * c[i];
            ++i;
        }

        for (; i + 3 < end; i += 4) {
            __m256d va = _mm256_load_pd(a + i);
            __m256d vb = _mm256_loadu_pd(b + i);
            __m256d vc = _mm256_loadu_pd(c + i);
            va = _mm256_mul_pd(va, vb);
            va = _mm256_mul_pd(va, vc);
            _mm256_store_pd(a + i, va);
        }

        for (; i < end; ++i) {
            a[i] = a[i] * b[i] * c[i];
        }
    }
}
