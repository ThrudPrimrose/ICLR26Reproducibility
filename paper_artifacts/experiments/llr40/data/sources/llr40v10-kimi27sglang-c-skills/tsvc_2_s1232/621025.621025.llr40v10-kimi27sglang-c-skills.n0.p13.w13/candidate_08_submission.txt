#include <stdint.h>
#include <omp.h>

static void kernel_positive(double *restrict aa, const double *restrict bb, const double *restrict cc,
                            const int64_t LEN_2D, const int64_t VLEN) {
#pragma omp parallel for schedule(guided)
  for (int64_t i = 0; i < LEN_2D; ++i) {
    int64_t jmax = i / VLEN + 1;
    double *restrict a_row = aa + i * LEN_2D;
    const double *restrict b_row = bb + i * LEN_2D;
    const double *restrict c_row = cc + i * LEN_2D;
#pragma omp simd
    for (int64_t j = 0; j < jmax; ++j) {
      a_row[j] = b_row[j] + c_row[j];
    }
  }
}

static void kernel_zero(double *restrict aa, const double *restrict bb, const double *restrict cc,
                        const int64_t LEN_2D) {
#pragma omp parallel for schedule(guided)
  for (int64_t i = 0; i < LEN_2D; ++i) {
    double *restrict a_row = aa + i * LEN_2D;
    const double *restrict b_row = bb + i * LEN_2D;
    const double *restrict c_row = cc + i * LEN_2D;
#pragma omp simd
    for (int64_t j = 0; j < LEN_2D; ++j) {
      a_row[j] = b_row[j] + c_row[j];
    }
  }
}

void tsvc_2_s1232_fp64(double *restrict aa, const double *restrict bb, const double *restrict cc, const int64_t LEN_2D,
                       const int64_t VLEN) {
  if (VLEN > 0) {
    kernel_positive(aa, bb, cc, LEN_2D, VLEN);
  } else {
    kernel_zero(aa, bb, cc, LEN_2D);
  }
}
