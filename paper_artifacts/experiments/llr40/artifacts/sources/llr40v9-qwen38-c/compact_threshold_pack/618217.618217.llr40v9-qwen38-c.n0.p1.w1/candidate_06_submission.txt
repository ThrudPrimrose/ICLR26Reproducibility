/* compact_threshold_pack -- one-pass: count+stage in phase 1, NT flush in phase 2.
 * Phase 1 reads src and weight ONCE, multiplies and compacts the survivors into
 * a per-thread staging buffer (L2-resident, 1 MB), and counts them.  After the
 * barrier the thread's output slice [base, base+cnt) is known, so it flushes the
 * staging with 64 B non-temporal aligned stores (no write-allocate traffic, and
 * src is never re-read).  Head/tail (< 8 doubles) use plain stores.
 */
#include <stdint.h>
#include <immintrin.h>
#include <omp.h>

#ifndef CTP_B
#define CTP_B 262144
#endif

static int64_t ctp_lut[256][8] __attribute__((aligned(64)));
__attribute__((constructor)) static void ctp_lut_build(void) {
  for (int m = 0; m < 256; m++) {
    int dset = 0;
    for (int b = 0; b < 8; b++)
      if (m & (1 << b)) ctp_lut[m][dset++] = 0x3FF0000000000000LL + b;
    for (int d = dset; d < 8; d++) ctp_lut[m][d] = 0x3FF0000000000000LL + d;
  }
}

static double ctp_stage[32][CTP_B] __attribute__((aligned(64)));

void compact_threshold_pack_fp64(int64_t *restrict out_count,
                                 double *restrict packed,
                                 const double *restrict src,
                                 const double *restrict weight,
                                 const int64_t LEN_1D) {
  if (LEN_1D <= 0) {
    out_count[0] = 0;
    return;
  }

  int nt = omp_get_max_threads();
  if (nt < 1) nt = 1;
  if (nt > 32) nt = 32;
  const int64_t B = CTP_B;
  const int64_t nblocks = (LEN_1D + B - 1) / B;
  static int64_t cur[2][32];
  const __m512d zero = _mm512_setzero_pd();

#pragma omp parallel num_threads(nt)
  {
    int tid = omp_get_thread_num();
    double *st = ctp_stage[tid];
    int64_t run_total = 0;
    for (int64_t r = 0; r * nt < nblocks; r++) {
      int64_t b = r * nt + (int64_t)tid;
      int64_t n_b = 0;
      const double *s = 0, *w = 0;
      if (b < nblocks) {
        int64_t lo = b * B;
        n_b = (lo + B > LEN_1D) ? (LEN_1D - lo) : B;
        s = src + lo;
        w = weight + lo;
      }

      /* phase 1: count + compact into staging (src & weight read once) */
      int64_t cnt = 0, sw = 0;
      {
        int64_t i = 0;
        for (; i + 16 <= n_b; i += 16) {
          __m512d sA = _mm512_loadu_pd(s + i);
          __m512d sB = _mm512_loadu_pd(s + i + 8);
          __mmask8 mA = _mm512_cmp_pd_mask(sA, zero, _CMP_GT_OQ);
          __mmask8 mB = _mm512_cmp_pd_mask(sB, zero, _CMP_GT_OQ);
          __m512d pA = _mm512_mul_pd(sA, _mm512_loadu_pd(w + i));
          __m512d pB = _mm512_mul_pd(sB, _mm512_loadu_pd(w + i + 8));
          __m512d cA = _mm512_permutexvar_pd(_mm512_loadu_si512((const void *)ctp_lut[mA]), pA);
          __m512d cB = _mm512_permutexvar_pd(_mm512_loadu_si512((const void *)ctp_lut[mB]), pB);
          int cntA = __builtin_popcount(mA);
          int cntB = __builtin_popcount(mB);
          _mm512_mask_storeu_pd(st + sw, (__mmask8)((1u << cntA) - 1u), cA);
          _mm512_mask_storeu_pd(st + sw + cntA, (__mmask8)((1u << cntB) - 1u), cB);
          sw += (int64_t)cntA + (int64_t)cntB;
          cnt = sw;
        }
        for (; i < n_b; i++)
          if (s[i] > 0.0) st[sw++] = s[i] * w[i];
        cnt = sw;
      }
      cur[r & 1][tid] = cnt;

      #pragma omp barrier

      if (n_b) {
        int64_t base = run_total;
        for (int t = 0; t < tid; t++) base += cur[r & 1][t];
        /* alignment class of packed[base] in 8 B units (st rows are 64 B
         * aligned, so source side stays aligned for every class below) */
        int64_t u = (((size_t)packed >> 3) + base) & 7;
        int64_t sw = 0;
        if (u == 0) {
          for (; sw + 8 <= cnt; sw += 8)
            _mm512_stream_pd(packed + base + sw, _mm512_load_pd(st + sw));
        } else if ((u & 3) == 0) {
          for (; sw + 4 <= cnt; sw += 4)
            _mm256_stream_pd(packed + base + sw, _mm256_load_pd(st + sw));
        } else if ((u & 1) == 0) {
          for (; sw + 2 <= cnt; sw += 2)
            _mm_stream_pd(packed + base + sw, _mm_load_pd(st + sw));
        } /* u odd: scalar below */
        for (; sw < cnt; sw++) packed[base + sw] = st[sw];
      }
      for (int t = 0; t < nt; t++) run_total += cur[r & 1][t];
    }
        _mm_mfence();
    if (tid == 0) out_count[0] = run_total;
  }
}
