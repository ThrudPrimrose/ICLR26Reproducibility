#include <stdint.h>
#include <stdatomic.h>
#include <omp.h>

void scatter_accum_dup_fp64(double *restrict bins, int32_t *restrict ip, double *restrict src, const int64_t LEN_1D, uint8_t *restrict workspace, const int64_t workspace_bytes) {
    (void)workspace;
    (void)workspace_bytes;
    _Atomic double *restrict abins = (_Atomic double *)bins;
    #pragma omp parallel for schedule(guided, 2048)
    for (int64_t i = 0; i < LEN_1D; ++i) {
        abins[ip[i]] += src[i];
    }
}
