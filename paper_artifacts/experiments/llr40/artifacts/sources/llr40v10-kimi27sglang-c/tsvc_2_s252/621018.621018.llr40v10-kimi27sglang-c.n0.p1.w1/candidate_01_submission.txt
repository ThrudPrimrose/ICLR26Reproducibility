#include <stdint.h>

void tsvc_2_s252_fp64(double *restrict a, const double *restrict b, const double *restrict c, const int64_t LEN_1D) {
  // The recurrence a[i] = b[i]*c[i] + previous_s is equivalent to:
  // let s[i] = b[i]*c[i];
  // a[0] = s[0];
  // a[i] = s[i] + s[i-1] for i >= 1.
  // This removes the loop-carried dependency and exposes full parallelism.
  if (LEN_1D <= 0) return;

  a[0] = b[0] * c[0];

  #pragma omp parallel for
  for (int64_t i = 1; i < LEN_1D; ++i) {
    const double si = b[i] * c[i];
    a[i] = si + b[i - 1] * c[i - 1];
  }
}
