#include <stdint.h>
#include <omp.h>

#ifndef BLOCK
#define BLOCK 256
#endif

void tsvc_2_s231_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
    const int64_t n = LEN_2D;
    if (n <= 0) return;

    if (n < 64) {
        #pragma omp parallel for schedule(static)
        for (int64_t i = 0; i < n; ++i) {
            for (int64_t j = 1; j < n; ++j) {
                aa[j * n + i] = aa[(j - 1) * n + i] + bb[j * n + i];
            }
        }
        return;
    }

    const int64_t BS = BLOCK;
    #pragma omp parallel for schedule(static)
    for (int64_t ib = 0; ib < n; ib += BS) {
        const int64_t ie = (ib + BS < n) ? ib + BS : n;
        for (int64_t j = 1; j < n; ++j) {
            const double *restrict prev = &aa[(j - 1) * n + ib];
            const double *restrict b = &bb[j * n + ib];
            double *restrict cur = &aa[j * n + ib];
            for (int64_t i = 0; i < ie - ib; ++i) {
                cur[i] = prev[i] + b[i];
            }
        }
    }
}
