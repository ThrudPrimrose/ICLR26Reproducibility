#include <stdint.h>
#include <immintrin.h>

void tsvc_2_s4112_fp64(double *restrict a, const double *restrict b, const int32_t *restrict ip, const int64_t LEN_1D) {
    const __m512d two = _mm512_set1_pd(2.0);
    int64_t i = 0;
    #pragma omp parallel for schedule(static)
    for (i = 0; i < LEN_1D - 7; i += 8) {
        __m256i idx = _mm256_loadu_si256((const __m256i *)(ip + i));
        __m512d bv = _mm512_i32gather_pd(idx, b, 8);
        __m512d av = _mm512_loadu_pd(a + i);
        av = _mm512_fmadd_pd(bv, two, av);
        _mm512_storeu_pd(a + i, av);
    }
    for (i = LEN_1D & ~7; i < LEN_1D; ++i) {
        a[i] += b[ip[i]] * 2.0;
    }
}
