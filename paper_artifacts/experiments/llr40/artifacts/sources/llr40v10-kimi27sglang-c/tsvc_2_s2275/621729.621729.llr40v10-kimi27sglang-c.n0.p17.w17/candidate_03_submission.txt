#include <stdint.h>
#include <omp.h>

void tsvc_2_s2275_fp64(double *restrict a, double *restrict aa, const double *restrict b, const double *restrict bb,
                       const double *restrict c, const double *restrict cc, const double *restrict d,
                       const int64_t LEN_2D) {

    const int64_t n2 = LEN_2D * LEN_2D;

    double *restrict aaa = __builtin_assume_aligned(aa, 64);
    const double *restrict bbb = __builtin_assume_aligned(bb, 64);
    const double *restrict ccc = __builtin_assume_aligned(cc, 64);
    double *restrict aaa2 = __builtin_assume_aligned(a, 64);
    const double *restrict bbb2 = __builtin_assume_aligned(b, 64);
    const double *restrict ccc2 = __builtin_assume_aligned(c, 64);
    const double *restrict ddd = __builtin_assume_aligned(d, 64);

#pragma omp parallel for
    for (int64_t k = 0; k < n2; ++k) {
        aaa[k] = aaa[k] + bbb[k] * ccc[k];
    }

#pragma omp parallel for
    for (int64_t i = 0; i < LEN_2D; ++i) {
        aaa2[i] = bbb2[i] + ccc2[i] * ddd[i];
    }
}
