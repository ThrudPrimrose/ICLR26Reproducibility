#include <stdint.h>
#include <stdlib.h>
#include <omp.h>

void ext_war_unit_fp64(double *restrict a, const double *restrict b, const int64_t LEN_1D) {
    const int64_t N = LEN_1D - 1;
    if (N <= 0) return;

    if (N < 200000) {
        for (int64_t i = 0; i < N; ++i)
            a[i] = a[i + 1] + b[i];
        return;
    }

    size_t bytes = (size_t)N * sizeof(double);
    size_t aligned_bytes = (bytes + 63) & ~((size_t)63);
    double *restrict tmp = (double *)aligned_alloc(64, aligned_bytes);
    if (!tmp) return;

    #pragma omp parallel
    {
        #pragma omp for schedule(static)
        for (int64_t i = 0; i < N; ++i)
            tmp[i] = a[i + 1];

        #pragma omp barrier

        #pragma omp for schedule(static)
        for (int64_t i = 0; i < N; ++i)
            a[i] = tmp[i] + b[i];
    }

    free(tmp);
}
