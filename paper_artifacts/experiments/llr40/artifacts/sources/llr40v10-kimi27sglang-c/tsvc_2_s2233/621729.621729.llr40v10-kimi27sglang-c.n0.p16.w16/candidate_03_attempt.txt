#include <stdint.h>

void tsvc_2_s2233_fp64(double *restrict aa, double *restrict bb, const double *restrict cc, const int64_t LEN_2D) {

  /* aa update: interchanged loops so the independent dimension (i) is inner.
   * Parallelize the inner dimension across threads; implicit barrier after
   * each row guarantees row j-1 is fully written before row j is read. */
  #pragma omp parallel
  {
    for (int64_t j = 8; j < LEN_2D; ++j) {
      #pragma omp for schedule(static)
      for (int64_t i = 8; i < LEN_2D; ++i) {
        aa[j * LEN_2D + i] = aa[(j - 1) * LEN_2D + i] + cc[j * LEN_2D + i];
      }
    }

    /* bb update: dependency along i, so keep i outer and parallelize the
     * independent inner dimension j. */
    for (int64_t i = 8; i < LEN_2D; ++i) {
      #pragma omp for schedule(static)
      for (int64_t j = 8; j < LEN_2D; ++j) {
        bb[i * LEN_2D + j] = bb[(i - 1) * LEN_2D + j] + cc[i * LEN_2D + j];
      }
    }
  }
}
