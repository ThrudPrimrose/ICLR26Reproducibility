#include <stdint.h>
#include <omp.h>

void segment_reduce_ragged_fp64(double* __restrict out,
                                int64_t* __restrict row_ptr,
                                double* __restrict val,
                                double* __restrict w,
                                int64_t NSEG,
                                uint8_t* __restrict workspace,
                                int64_t workspace_size) {
    (void)workspace;
    (void)workspace_size;
    #pragma omp parallel for schedule(static)
    for (int64_t s = 0; s < NSEG; s++) {
        double acc = 0.0;
        int64_t start = row_ptr[s];
        int64_t end = row_ptr[s + 1];
        for (int64_t e = start; e < end; e++) {
            acc += val[e] * w[e];
        }
        out[s] = acc;
    }
}
