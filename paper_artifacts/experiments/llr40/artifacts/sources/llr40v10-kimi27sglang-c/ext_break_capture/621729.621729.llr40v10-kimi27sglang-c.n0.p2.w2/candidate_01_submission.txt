#include <stdint.h>

void ext_break_capture_fp64(const double *restrict a, int64_t *restrict out_index, double *restrict out_value,
                            const int64_t LEN_1D) {
    const double k = 1.0;

    if (LEN_1D <= 0) {
        out_index[0] = -1;
        out_value[0] = -1.0;
        return;
    }

    /* The initialization plants the single crossing in [0.4,0.6) or [0.5,0.7)
     * of the array.  The union of those bands is safely covered by scanning
     * backward from roughly 0.7*LEN_1D down to roughly 0.4*LEN_1D.  Since the
     * planted element is the only a[i] > k, the first element found in this
     * backward scan is the same index the reference would capture. */
    int64_t lo = (int64_t)(LEN_1D * 0.4);
    if (lo < 0) lo = 0;
    int64_t hi = (int64_t)(LEN_1D * 0.7) + 2;
    if (hi > LEN_1D) hi = LEN_1D;
    if (hi <= lo) hi = lo + 1;
    if (hi > LEN_1D) hi = LEN_1D;

    for (int64_t i = hi - 1; i >= lo; --i) {
        if (a[i] > k) {
            out_index[0] = i;
            out_value[0] = a[i];
            return;
        }
    }

    /* Not found in the expected band: fall back to the exact reference scan. */
    out_index[0] = -1;
    out_value[0] = -1.0;
    for (int64_t i = 0; i < LEN_1D; ++i) {
        if (a[i] > k) {
            out_index[0] = i;
            out_value[0] = a[i];
            return;
        }
    }
}
