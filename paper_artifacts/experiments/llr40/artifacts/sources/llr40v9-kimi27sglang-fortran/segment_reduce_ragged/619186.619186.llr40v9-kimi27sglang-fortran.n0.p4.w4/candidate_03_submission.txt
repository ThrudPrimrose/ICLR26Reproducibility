subroutine segment_reduce_ragged_fp64(out, row_ptr, val, w, NSEG, workspace, workspace_size) bind(c, name="segment_reduce_ragged_fp64")
  use iso_c_binding, only: c_int64_t, c_double, c_int8_t
  implicit none
  real(c_double), intent(inout) :: out(*)
  integer(c_int64_t), intent(in) :: row_ptr(*)
  real(c_double), intent(in) :: val(*)
  real(c_double), intent(in) :: w(*)
  integer(c_int64_t), intent(in), value :: NSEG
  integer(c_int8_t), intent(inout) :: workspace(*)
  integer(c_int64_t), intent(in), value :: workspace_size
  integer(c_int64_t) :: s, e
  real(c_double) :: acc
  !$omp parallel do schedule(guided, 4) private(acc, e)
  do s = 1, NSEG
    acc = 0.0_c_double
    do e = row_ptr(s) + 1, row_ptr(s+1)
      acc = acc + val(e) * w(e)
    end do
    out(s) = acc
  end do
  !$omp end parallel do
end subroutine segment_reduce_ragged_fp64
