#include <stdint.h>
#include <stdlib.h>
#include <stdbool.h>
#include <string.h>
#include <math.h>
#include <complex.h>
#include <omp.h>

void scatter_accum_dup_fp64(double *restrict bins, const int32_t *restrict ip, const double *restrict src, const int64_t LEN_1D) {
    if (LEN_1D < 2048) {
        for (int64_t i = 0; i < LEN_1D; ++i) {
            const int64_t idx = (int64_t)ip[i];
            bins[idx] = bins[idx] + src[i];
        }
        return;
    }
    #pragma omp parallel for schedule(guided, 4096)
    for (int64_t i = 0; i < LEN_1D; ++i) {
        const int64_t idx = (int64_t)ip[i];
        if (i + 16 < LEN_1D) {
            __builtin_prefetch(&bins[(int64_t)ip[i + 16]], 1, 0);
        }
        #pragma omp atomic update relaxed
        bins[idx] += src[i];
    }
}
