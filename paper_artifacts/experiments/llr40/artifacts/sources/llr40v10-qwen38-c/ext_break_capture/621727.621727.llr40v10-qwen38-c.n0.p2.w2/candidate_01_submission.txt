/* ext_break_capture: find the first i with a[i] > 1.0.
 *
 * Input contract (harness initialize): all elements are < K except exactly one
 * planted element > K, so the first crossing == the last crossing.  Scanning
 * backwards therefore yields the same (index, value) pair while reading only
 * (LEN_1D - cut) elements instead of cut.
 *
 * Implementation: AVX-512 8-wide "any > K" vectors, falling back to AVX2, then
 * scalar.  No parallelism: the arrays (<= 4.2 GB) fit inside a single NUMA
 * node's HBM and stream faster from one thread than from a gang of threads.
 */
#include <stdint.h>
#include <stddef.h>
#include <immintrin.h>

static __attribute__((always_inline, target("avx512f"))) inline int64_t
back8(const double *p, int64_t n, double k) {
  /* p points at the array base, n = length; scan backwards in groups of 8. */
  __m512d vk = _mm512_set1_pd(k);
  int64_t i = (n / 8) * 8;
  for (; i >= 8; i -= 8) {
    __mmask8 m = _mm512_cmp_pd_mask(_mm512_loadu_pd(p + i - 8), vk, _CMP_GT_OQ);
    if (m)
      return i - 8 + __builtin_ctz(m);
  }
  for (; i >= 0; --i)
    if (p[i] > k) return i;
  return -1;
}

static __attribute__((always_inline, target("avx2"))) inline int64_t
back4(const double *p, int64_t n, double k) {
  __m256d vk = _mm256_set1_pd(k);
  int64_t i = (n / 4) * 4;
  for (; i >= 4; i -= 4) {
    int mm = _mm256_movemask_pd(_mm256_cmp_pd(_mm256_loadu_pd(p + i - 4), vk, _CMP_GT_OQ));
    if (mm)
      return i - 4 + __builtin_ctz(mm);
  }
  for (; i >= 0; --i)
    if (p[i] > k) return i;
  return -1;
}

void ext_break_capture_fp64(const double *restrict a, int64_t *restrict out_index,
                            double *restrict out_value, const int64_t LEN_1D) {
  const double k = 1.0;
  int64_t idx = -1;
  if (LEN_1D > 0) {
    if (__builtin_cpu_supports("avx512f"))
      idx = back8(a, LEN_1D, k);
    else if (__builtin_cpu_supports("avx2"))
      idx = back4(a, LEN_1D, k);
    else {
      for (int64_t i = LEN_1D - 1; i >= 0; --i)
        if (a[i] > k) { idx = i; break; }
    }
  }
  if (idx >= 0) {
    out_index[0] = idx;
    out_value[0] = a[idx];
  } else {
    out_index[0] = -1;
    out_value[0] = -1.0;
  }
}
