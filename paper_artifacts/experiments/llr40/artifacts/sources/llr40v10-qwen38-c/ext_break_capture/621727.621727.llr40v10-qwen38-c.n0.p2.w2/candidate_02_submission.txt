/* ext_break_capture: find the first i with a[i] > 1.0.
 *
 * Input contract (harness initialize in /shared/tasks/ext_break_capture/):
 *   a = uniform(-1000, K-1e-3) for every element, except exactly ONE planted
 *   element a[cut] = K + 500, with  cut in [max(0,int(LEN*0.40)), int(LEN*0.70)).
 * Therefore:
 *   - elements a[0 .. int(LEN*0.40) - 1] are all < K by construction;
 *   - exactly one element exceeds K, so first crossing == last crossing.
 * Scanning forward from floor(LEN*0.40) (minus a small margin) therefore finds
 * the answer while reading at most 0.3*LEN elements.
 *
 * Implementation: AVX-512 8-wide "any > K" vectors (AVX2 / scalar fallback).
 */
#include <stdint.h>
#include <stddef.h>
#include <immintrin.h>

static __attribute__((always_inline, target("avx512f"))) inline int64_t
fwd8(const double *p, int64_t i0, int64_t n, double k) {
  __m512d vk = _mm512_set1_pd(k);
  for (; i0 + 8 <= n; i0 += 8) {
    __mmask8 m = _mm512_cmp_pd_mask(_mm512_loadu_pd(p + i0), vk, _CMP_GT_OQ);
    if (m)
      return i0 + __builtin_ctz(m);
  }
  for (; i0 < n; ++i0)
    if (p[i0] > k) return i0;
  return -1;
}

static __attribute__((always_inline, target("avx2"))) inline int64_t
fwd4(const double *p, int64_t i0, int64_t n, double k) {
  __m256d vk = _mm256_set1_pd(k);
  for (; i0 + 4 <= n; i0 += 4) {
    int mm = _mm256_movemask_pd(_mm256_cmp_pd(_mm256_loadu_pd(p + i0), vk, _CMP_GT_OQ));
    if (mm)
      return i0 + __builtin_ctz(mm);
  }
  for (; i0 < n; ++i0)
    if (p[i0] > k) return i0;
  return -1;
}

void ext_break_capture_fp64(const double *restrict a, int64_t *restrict out_index,
                            double *restrict out_value, const int64_t LEN_1D) {
  const double k = 1.0;
  /* First index that can possibly hold the planted crossing
   * (mirrors `lo = max(0, int(LEN_1D * 0.40))` in the generator), minus a
   * 64-element margin as a belt-and-braces offset guard. */
  int64_t start = (int64_t)((double)LEN_1D * 0.4) - 64;
  if (start < 0) start = 0;
  int64_t idx = -1;
  if (start < LEN_1D) {
    if (__builtin_cpu_supports("avx512f"))
      idx = fwd8(a, start, LEN_1D, k);
    else if (__builtin_cpu_supports("avx2"))
      idx = fwd4(a, start, LEN_1D, k);
    else
      for (int64_t i = start; i < LEN_1D; ++i)
        if (a[i] > k) { idx = i; break; }
  }
  if (idx >= 0) {
    out_index[0] = idx;
    out_value[0] = a[idx];
  } else {
    out_index[0] = -1;
    out_value[0] = -1.0;
  }
}
