#include <stdint.h>
#include <omp.h>

void tsvc_2_s311_fp64(const double *restrict a, double *restrict sum_out, const int64_t LEN_1D) {
  double total = 0.0;
  #pragma omp parallel reduction(+:total)
  {
    #pragma omp for
    for (int64_t i = 0; i < LEN_1D; i++) total += a[i];
  }
  sum_out[0] = total;
}
