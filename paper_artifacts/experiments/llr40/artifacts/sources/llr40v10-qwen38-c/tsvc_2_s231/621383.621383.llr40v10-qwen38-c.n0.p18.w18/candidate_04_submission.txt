#include <stdint.h>
#include <omp.h>
#include <stdlib.h>

/*
 * s231: aa[j][i] = aa[j-1][i] + bb[j][i]  (row-major, j=1..N-1, all i)
 * = per-column scan along rows: aa[j] = aa[0] + sum_{k=1..j} bb[k].
 *
 * Parallel (T = nthreads chunks), 3*N^2*8 bytes of DRAM traffic:
 *   Pass A (parallel): chunk_sum[c] = sum of bb rows in chunk c (read bb only).
 *   Bases (sequential, tiny): baseval[0]=aa[0]; baseval[c]=baseval[c-1]+chunk_sum[c-1]
 *                             => baseval[c] = aa[0] + sum_{k=1..s_c-1} bb[k] (true base).
 *   Pass B (parallel): cur = baseval[c]; for j in chunk c: cur += bb[j]; aa[j] = cur.
 *                      (read bb, write aa; cur stays in L2). No aa re-reads, no 2nd pass.
 * The only fp reassociation is the chunk-sum grouping in the bases (a few ulps).
 */
void tsvc_2_s231_fp64(double *restrict aa, const double *restrict bb, const int64_t LEN_2D) {
  const int64_t N = LEN_2D;
  if (N <= 1) return;

  int T = omp_get_max_threads();
  if (T < 1) T = 1;
  if (T > N - 1) T = (int)(N - 1);
  const int64_t base = (N - 1) / T;

  static double *csum = NULL, *baseval = NULL;
  static int64_t cached_N = 0, cached_T = 0;
  if (N != cached_N || T != cached_T) {
    free(csum); free(baseval);
    csum    = (double *)malloc((size_t)T * (size_t)N * sizeof(double));
    baseval = (double *)malloc((size_t)T * (size_t)N * sizeof(double));
    cached_N = N; cached_T = T;
  }

  // Pass A: per-chunk row-sum of bb (parallel over chunks, serial within)
  #pragma omp parallel for schedule(static, 1)
  for (int c = 0; c < T; ++c) {
    int64_t s = 1 + c * base;
    int64_t e = (c == T - 1) ? (N - 1) : (s + base - 1);
    double *cs = csum + (size_t)c * N;
    for (int64_t i = 0; i < N; ++i) cs[i] = 0.0;
    for (int64_t j = s; j <= e; ++j) {
      if (j + 3 < N) {
        __builtin_prefetch(bb + (j + 1) * N, 0, 1);
        __builtin_prefetch(bb + (j + 2) * N, 0, 1);
        __builtin_prefetch(bb + (j + 3) * N, 0, 1);
      }
      const double *b = bb + j * N;
      for (int64_t i = 0; i < N; ++i) cs[i] += b[i];
    }
  }

  // Bases: baseval[0] = aa[0]; baseval[c] = baseval[c-1] + csum[c-1]
  {
    double *b0 = baseval;               // chunk 0 base = aa[0]
    const double *row0 = aa;
    for (int64_t i = 0; i < N; ++i) b0[i] = row0[i];
    for (int c = 1; c < T; ++c) {
      double *bc = baseval + (size_t)c * N;
      const double *bcp1 = baseval + (size_t)(c - 1) * N;
      const double *cs = csum + (size_t)(c - 1) * N;
      for (int64_t i = 0; i < N; ++i) bc[i] = bcp1[i] + cs[i];
    }
  }

  // Pass B: local scan from true base (parallel over chunks); cur in L2
  #pragma omp parallel for schedule(static, 1)
  for (int c = 0; c < T; ++c) {
    int64_t s = 1 + c * base;
    int64_t e = (c == T - 1) ? (N - 1) : (s + base - 1);
    double *cur = baseval + (size_t)c * N;   // running total (in place)
    for (int64_t j = s; j <= e; ++j) {
      if (j + 3 < N) {
        __builtin_prefetch(bb + (j + 1) * N, 0, 1);
        __builtin_prefetch(bb + (j + 2) * N, 0, 1);
        __builtin_prefetch(bb + (j + 3) * N, 0, 1);
      }
      const double *b = bb + j * N;
      double *a = aa + j * N;
      for (int64_t i = 0; i < N; ++i) {
        double v = cur[i] + b[i];
        cur[i] = v;
        a[i] = v;
      }
    }
  }
}
