#include <stdint.h>

void tsvc_2_s1244_fp64(double *restrict a, const double *restrict b, const double *restrict c,
                       double *restrict d, const int64_t LEN_1D) {
  const int64_t n = LEN_1D - 1;
  if (n <= 0) return;

  /* a[i] := v[i] for i in [0,n); d[i] = a[i](new) + a[i+1](original).
     Since iteration i never writes a[i+1] before reading it, a[i+1] here is
     the ORIGINAL input value, so the two statements touch disjoint index
     sets and both loops below are fully independent. */
  #pragma omp parallel for schedule(static)
  for (int64_t i = 0; i < n; i++) {
    d[i] = (b[i] + c[i] * c[i] + b[i] * b[i] + c[i]) + a[i + 1];
  }
  #pragma omp parallel for schedule(static)
  for (int64_t i = 0; i < n; i++) {
    a[i] = b[i] + c[i] * c[i] + b[i] * b[i] + c[i];
  }
}
